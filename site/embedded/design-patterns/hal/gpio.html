    توصیه‌هایی برای رابط‌های GPIO

    نوع‌های پایه به‌طور پیش‌فرض اندازهٔ صفر دارند (C-ZST-PIN)

    رابط‌های GPIO که HAL در اختیار می‌گذارد باید برای هر پایه در هر رابط یا درگاه، نوع‌های اختصاصی با اندازهٔ صفر فراهم کنند، که وقتی همهٔ تخصیص‌های پایه به‌صورت ایستا معلوم باشند، انتزاعی با هزینهٔ صفر برای GPIO به دست می‌دهد.

    هر رابط یا درگاه GPIO باید یک متد split پیاده‌سازی کند که ساختاری شامل همهٔ پایه‌ها برگرداند.

    مثال:

    pub struct PA0;
pub struct PA1;
// ...

pub struct PortA;

impl PortA {
    pub fn split(self) -> PortAPins {
        PortAPins {
            pa0: PA0,
            pa1: PA1,
            // ...
        }
    }
}

pub struct PortAPins {
    pub pa0: PA0,
    pub pa1: PA1,
    // ...
}

    نوع‌های پایه متدهایی برای پاک‌کردن اطلاعات پایه و درگاه فراهم می‌کنند (C-ERASED-PIN)

    پایه‌ها باید متدهای پاک‌سازی نوع (type erasure) فراهم کنند که ویژگی‌هایشان را از زمان کامپایل به زمان اجرا منتقل می‌کند و انعطاف بیشتری در برنامه‌ها ممکن می‌سازد.

    مثال:

    /// Port A, pin 0.
pub struct PA0;

impl PA0 {
    pub fn erase_pin(self) -> PA {
        PA { pin: 0 }
    }
}

/// A pin on port A.
pub struct PA {
    /// The pin number.
    pin: u8,
}

impl PA {
    pub fn erase_port(self) -> Pin {
        Pin {
            port: Port::A,
            pin: self.pin,
        }
    }
}

pub struct Pin {
    port: Port,
    pin: u8,
    // (these fields can be packed to reduce the memory footprint)
}

enum Port {
    A,
    B,
    C,
    D,
}

    وضعیت پایه باید به‌صورت پارامترهای نوع رمزگذاری شود (C-PIN-STATE)

    بسته به تراشه یا خانواده، پایه‌ها ممکن است با ویژگی‌های مختلف به‌صورت ورودی یا خروجی پیکربندی شوند. این وضعیت باید در سامانهٔ نوع رمزگذاری شود تا از استفاده از پایه‌ها در وضعیت‌های نادرست جلوگیری شود.

    وضعیت اضافی مختص تراشه (مثلاً قدرت راه‌اندازی یا drive strength) نیز ممکن است با استفاده از پارامترهای نوع اضافی به همین شیوه رمزگذاری شود.

    متدهای تغییر وضعیت پایه باید به‌صورت متدهای into_input و into_output فراهم شوند.

    افزون بر این، متدهای with_{input,output}_state باید فراهم شوند که یک پایه را به‌طور موقت در وضعیتی متفاوت پیکربندی مجدد می‌کنند، بی‌آنکه آن را منتقل کنند.

    متدهای زیر باید برای هر نوع پایه فراهم شوند (یعنی هم نوع‌های پایهٔ پاک‌شده و هم پاک‌نشده باید همین API را ارائه دهند):

    • pub fn into_input<N: InputState>(self, input: N) -> Pin<N>
    • pub fn into_output<N: OutputState>(self, output: N) -> Pin<N>
    • pub fn with_input_state<N: InputState, R>(
    &mut self,
    input: N,
    f: impl FnOnce(&mut PA1<N>) -> R,
) -> R

    • pub fn with_output_state<N: OutputState, R>(
    &mut self,
    output: N,
    f: impl FnOnce(&mut PA1<N>) -> R,
) -> R

    وضعیت پایه باید با صفت‌های مهرشده (sealed trait) کران‌دار شود. کاربران HAL نباید نیازی به افزودن وضعیت خودشان داشته باشند. این صفت‌ها می‌توانند متدهای مختص HAL را که برای پیاده‌سازی API وضعیت پایه لازم‌اند فراهم کنند.

    مثال:

    use std::marker::PhantomData;
mod sealed {
    pub trait Sealed {}
}

pub trait PinState: sealed::Sealed {}
pub trait OutputState: sealed::Sealed {}
pub trait InputState: sealed::Sealed {
    // ...
}

pub struct Output<S: OutputState> {
    _p: PhantomData<S>,
}

impl<S: OutputState> PinState for Output<S> {}
impl<S: OutputState> sealed::Sealed for Output<S> {}

pub struct PushPull;
pub struct OpenDrain;

impl OutputState for PushPull {}
impl OutputState for OpenDrain {}
impl sealed::Sealed for PushPull {}
impl sealed::Sealed for OpenDrain {}

pub struct Input<S: InputState> {
    _p: PhantomData<S>,
}

impl<S: InputState> PinState for Input<S> {}
impl<S: InputState> sealed::Sealed for Input<S> {}

pub struct Floating;
pub struct PullUp;
pub struct PullDown;

impl InputState for Floating {}
impl InputState for PullUp {}
impl InputState for PullDown {}
impl sealed::Sealed for Floating {}
impl sealed::Sealed for PullUp {}
impl sealed::Sealed for PullDown {}
