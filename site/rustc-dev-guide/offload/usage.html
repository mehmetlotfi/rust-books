    Usage

    این قابلیت در حال توسعه است و آمادهٔ استفاده نیست. دستورالعمل‌های اینجا برای مشارکت‌کنندگان یا افراد علاقه‌مند به دنبال‌کردن آخرین پیشرفت است. فعلاً روی اجرای kernel زیر در Rust روی GPU کار می‌کنیم. برای دنبال‌کردن، آن را در فایل src/lib.rs کپی کنید.

    #![allow(internal_features)]
#![feature(gpu_offload)]
#![cfg_attr(target_os = "linux", feature(core_intrinsics))]
#![cfg_attr(target_arch = "amdgpu", feature(stdarch_amdgpu, abi_gpu_kernel))]
#![cfg_attr(target_arch = "nvptx64", feature(stdarch_nvptx, abi_gpu_kernel))]
#![no_std]

#[cfg(target_os = "linux")]
extern crate libc;

use core::offload::offload_kernel;

#[panic_handler]
fn panic(_: &core::panic::PanicInfo) -> ! {
    loop {}
}

#[cfg(target_arch = "amdgpu")]
use core::arch::amdgpu::{workgroup_id_x as block_idx_x, workitem_id_x as thread_idx_x};
#[cfg(target_arch = "nvptx64")]
use core::arch::nvptx::{
    _block_dim_x as block_dim_x, _block_idx_x as block_idx_x, _thread_idx_x as thread_idx_x,
};

// Kernels can be generic, like any other Rust function.
// The concrete instantiations required by the host code are collected in a
// manifest, which the device compilation then reads (see below).
#[offload_kernel]
fn kernel<T: Copy>(x: *mut [T; 256], value: T) {
    unsafe {
        let n = (*x).len();
        let i = (thread_idx_x() + block_idx_x() * block_dim_x()) as usize;
        if i < n {
            (*x)[i] = value;
        }
    }
}

#[cfg(target_os = "linux")]
#[unsafe(no_mangle)]
fn main() {
    let mut x = [0.0f64; 256];
    core::offload::offload! {
        kernel = kernel::<f64>,
        workgroup_dim = [256, 1, 1],
        args = (&mut x as *mut [f64; 256], 2.5),
    }
    for i in 0..x.len() {
        assert_eq!(x[i], 2.5);
    }
    unsafe { libc::printf(c"all checks passed".as_ptr()); }
}

    Compile instructions

    مهم است از clangی استفاده کنید که روی همان LLVM مربوط به rustc ساخته شده است. فراخوانی clang بدون مسیر کامل احتمالاً از clang سیستمی شما استفاده می‌کند که احتمالاً ناسازگار خواهد بود. پس یا فراخوانی‌های clang/lld زیر را با مسیر مطلق جایگزین کنید، یا PATH خود را متناسب تنظیم کنید.

    کامپایل سه گذر اجرا می‌کند:

    1. HostMetadata: کد میزبان را کامپایل می‌کند و مانیفستی می‌نویسد که نمونه‌سازی‌های kernel (از جمله جنریک‌ها) مورد نیاز کد میزبان را فهرست می‌کند.
    2. Device: kernelها را برای GPU کامپایل می‌کند و مانیفست را می‌خواند تا نمونه‌سازی‌های جنریک ثبت‌شده codegen شوند.
    3. Host: کد نهایی میزبان را تولید می‌کند و آرتیفکت device را در آن جاسازی می‌کند.

    target-cpu (gfx90a) را با کد درست برای GPU خود جایگزین کنید. این‌ها اغلب «نام‌های هدف LLVM» نامیده می‌شوند1.

    ابتدا مانیفست را از کد میزبان تولید می‌کنیم:

    RUSTFLAGS="--emit=llvm-bc,llvm-ir -Csave-temps -Zoffload=HostMetadata=/absolute/path/to/offload.manifest -Zunstable-options" cargo +offload build -r

    این گذر فقط مانیفست را می‌نویسد.

    حالا کد device (GPU) را با دادن مانیفست تولید می‌کنیم:

    RUSTFLAGS="-Ctarget-cpu=gfx90a --emit=llvm-bc,llvm-ir -Zoffload=Device=/absolute/path/to/offload.manifest -Csave-temps -Zunstable-options" cargo +offload build -Zunstable-options -r -v --target amdgcn-amd-amdhsa -Zbuild-std=core

    سپس کد میزبان (CPU) را تولید می‌کنیم.

    RUSTFLAGS="--emit=llvm-bc,llvm-ir -Csave-temps -Zoffload=Host=$PWD/target/amdgcn-amd-amdhsa/release/deps/device.bin -Zunstable-options" cargo +offload build -r

    در گام نهایی، اکنون می‌توانید فایل دودویی خود را اجرا کنید

    ./target/x86_64-unknown-linux-gnu/release/binary-name
all checks passed!

    این سه گام به‌زودی در یک دستور واحد جمع می‌شوند، وقتی زمان بیشتری برای تست همهٔ گام‌ها داشته باشیم.

    برای دریافت اطلاعات بیشتر دربارهٔ انتقال حافظه، می‌توانید با افزودن LIBOMPTARGET_INFO=-1 پیش از فراخوانی فایل دودویی چاپ اطلاعات را فعال کنید.

    1. https://rocm.docs.amd.com/en/latest/reference/gpu-arch-specs.html یا https://developer.nvidia.com/cuda/gpus. یا می‌توانید rustc --print target-cpus را بررسی کنید. ↩
