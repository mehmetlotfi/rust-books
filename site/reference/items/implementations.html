    پیاده‌سازی‌ها

    [items.impl.syntax]

    Syntax
    Implementation → InherentImpl | TraitImpl

    InherentImpl →
        impl GenericParams? Type WhereClause? {
            InnerAttribute*
            AssociatedItem*
        }

    TraitImpl →
        unsafe? impl GenericParams? !? TypePath for Type
        WhereClause?
        {
            InnerAttribute*
            AssociatedItem*
        }

    [items.impl.intro]

    پیاده‌سازی آیتمی است که آیتم‌ها را به یک نوع پیاده‌سازی‌کننده وابسته می‌کند. پیاده‌سازی با کلیدواژهٔ impl تعریف می‌شود و تابع‌هایی دارد که به نمونه‌ای از نوع مورد پیاده‌سازی یا به‌صورت ایستا به خود نوع تعلق دارند.

    [items.impl.kinds]

    دو نوع پیاده‌سازی وجود دارد:

    • پیاده‌سازی ذاتی
    • پیاده‌سازی صفت
    [items.impl.inherent]

    پیاده‌سازی‌های ذاتی

    [items.impl.inherent.intro]

    پیاده‌سازی ذاتی از کلیدواژهٔ impl، اعلان‌های نوع عمومی، مسیری به یک نوع اسمی، بند where و مجموعه‌ای از آیتم‌های قابل وابستگی درون آکولاد تشکیل می‌شود.

    [items.impl.inherent.implementing-type]

    نوع اسمی نوع پیاده‌سازی‌کننده و آیتم‌های قابل وابستگی، آیتم‌های وابسته به آن نوع نامیده می‌شوند.

    [items.impl.inherent.associated-items]

    پیاده‌سازی ذاتی آیتم‌های درون خود را به نوع پیاده‌سازی‌کننده وابسته می‌کند.

    [items.impl.inherent.allowed-items]

    پیاده‌سازی‌های ذاتی می‌توانند شامل تابع‌های وابسته (از جمله متدها) و ثابت‌های وابسته باشند.

    [items.impl.inherent.type-alias]

    نمی‌تواند نام مستعار نوع وابسته داشته باشد.

    [items.impl.inherent.associated-item-path]

    مسیر یک آیتم وابسته، هر مسیری به نوع پیاده‌سازی‌کننده است که شناسهٔ آیتم وابسته به‌عنوان آخرین جزء مسیر پس از آن آمده باشد.

    [items.impl.inherent.coherence]

    یک نوع می‌تواند چند پیاده‌سازی ذاتی داشته باشد. نوع پیاده‌سازی‌کننده باید در همان crate تعریف اصلی نوع تعریف شده باشد.

    pub mod color {
    pub struct Color(pub u8, pub u8, pub u8);

    impl Color {
        pub const WHITE: Color = Color(255, 255, 255);
    }
}

mod values {
    use super::color::Color;
    impl Color {
        pub fn red() -> Color {
            Color(255, 0, 0)
        }
    }
}

pub use self::color::Color;
fn main() {
    // Actual path to the implementing type and impl in the same module.
    color::Color::WHITE;

    // Impl blocks in different modules are still accessed through a path to the type.
    color::Color::red();

    // Re-exported paths to the implementing type also work.
    Color::red();

    // Does not work, because use in `values` is not pub.
    // values::Color::red();
}
    [items.impl.trait]

    پیاده‌سازی صفت

    [items.impl.trait.intro]

    پیاده‌سازی صفت مانند پیاده‌سازی ذاتی تعریف می‌شود، با این تفاوت که پس از اعلان‌های اختیاری نوع عمومی یک صفت، سپس کلیدواژهٔ for و بعد مسیری به نوع اسمی می‌آید.

    [items.impl.trait.implemented-trait]

    این trait با نام trait پیاده‌سازی‌شده (implemented trait) شناخته می‌شود. نوع پیاده‌ساز، trait پیاده‌سازی‌شده را پیاده‌سازی می‌کند.

    [items.impl.trait.def-requirement]

    پیاده‌سازی صفت باید همهٔ آیتم‌های وابستهٔ بدون پیش‌فرض اعلان‌شده توسط صفت را تعریف کند، می‌تواند آیتم‌های پیش‌فرض صفت را دوباره تعریف کند و نمی‌تواند آیتم دیگری تعریف کند.

    [items.impl.trait.associated-item-path]

    مسیر آیتم وابسته از <، سپس مسیر نوع پیاده‌سازی‌کننده، ‏as، مسیر صفت، ‏> به‌عنوان جزء مسیر و در پایان جزء مسیر آیتم وابسته تشکیل می‌شود.

    [items.impl.trait.safety]

    صفت‌های ناامن لازم می‌دانند پیاده‌سازی صفت با کلیدواژهٔ unsafe آغاز شود.

    #[derive(Copy, Clone)]
struct Point {x: f64, y: f64};
type Surface = i32;
struct BoundingBox {x: f64, y: f64, width: f64, height: f64};
trait Shape { fn draw(&self, s: Surface); fn bounding_box(&self) -> BoundingBox; }
fn do_draw_circle(s: Surface, c: Circle) { }
struct Circle {
    radius: f64,
    center: Point,
}

impl Copy for Circle {}

impl Clone for Circle {
    fn clone(&self) -> Circle { *self }
}

impl Shape for Circle {
    fn draw(&self, s: Surface) { do_draw_circle(s, *self); }
    fn bounding_box(&self) -> BoundingBox {
        let r = self.radius;
        BoundingBox {
            x: self.center.x - r,
            y: self.center.y - r,
            width: 2.0 * r,
            height: 2.0 * r,
        }
    }
}
    [items.impl.trait.coherence]

    انسجام پیاده‌سازی صفت

    [items.impl.trait.coherence.intro]

    دو پیاده‌سازی صفت زمانی هم‌پوشانی دارند که اشتراک صفت‌های مورد پیاده‌سازی تهی نباشد و بتوان پیاده‌سازی‌ها را با نوعی یکسان نمونه‌سازی کرد.

    [items.impl.trait.coherence.overlapping]

    دو پیاده‌سازی trait زمانی همپوشانی (overlap) دارند که اشتراک غیرتهی میان traitهایی که پیاده‌سازی برای آن‌هاست وجود داشته باشد و پیاده‌سازی‌ها بتوانند با نوع یکسانی نمونه‌سازی شوند.

    [items.impl.trait.orphan-rule]

    قواعد Orphan

    [items.impl.trait.orphan-rule.intro]

    قاعدهٔ orphan می‌گوید پیاده‌سازی صفت فقط زمانی مجاز است که خود صفت یا دست‌کم یکی از نوع‌های پیاده‌سازی در crate جاری تعریف شده باشد. این قاعده از پیاده‌سازی‌های متعارض صفت در crateهای مختلف جلوگیری می‌کند و برای تضمین انسجام ضروری است.

    پیاده‌سازی orphan یک صفت خارجی را برای نوع خارجی پیاده‌سازی می‌کند. اگر این کار آزادانه مجاز بود، دو crate می‌توانستند یک صفت را برای نوعی یکسان به‌شکل‌های ناسازگار پیاده‌سازی کنند؛ سپس افزودن یا به‌روزرسانی وابستگی می‌توانست به‌دلیل تعارض، کامپایل را بشکند.

    قاعدهٔ orphan به نویسندگان کتابخانه اجازه می‌دهد بدون ترس از شکستن کد پایین‌دستی، پیاده‌سازی‌های جدیدی به صفت‌های خود بیفزایند. بدون این محدودیت، کتابخانه نمی‌توانست پیاده‌سازی‌ای مانند impl<T: Display> MyTrait for T اضافه کند، زیرا ممکن بود با پیاده‌سازی پایین‌دستی تعارض کند.

    [items.impl.trait.orphan-rule.def]

    برای impl<P1..=Pn> Trait<T1..=Tn> for T0، یک impl فقط در صورت برقراری دست‌کم یکی از موارد زیر معتبر است:

    • Trait یک صفت محلی باشد؛
    • همهٔ موارد زیر
      • دست‌کم یکی از نوع‌های T0..=Tn یک نوع محلی باشد. Ti را نخستین نوع محلی در نظر بگیرید.
      • هیچ پارامتر نوع پوشش‌نیافته‌ای از P1..=Pn در T0..Ti—به‌جز خود Ti—ظاهر نشود.
    [items.impl.trait.uncovered-param]

    فقط ظاهرشدن پارامترهای نوع پوشش‌نیافته محدود شده است.

    [items.impl.trait.fundamental]

    برای انسجام، نوع‌های بنیادی ویژه‌اند. T در Box<T> پوشش‌یافته محسوب نمی‌شود و Box<LocalType> محلی در نظر گرفته می‌شود.

    [items.impl.generics]

    پیاده‌سازی‌های عمومی

    [items.impl.generics.intro]

    پیاده‌سازی می‌تواند پارامترهای عمومی بگیرد که در ادامهٔ پیاده‌سازی استفاده می‌شوند. پارامترها مستقیماً پس از کلیدواژهٔ impl نوشته می‌شوند.

    trait Seq<T> { fn dummy(&self, _: T) { } }
impl<T> Seq<T> for Vec<T> {
    /* ... */
}
impl Seq<bool> for u32 {
    /* Treat the integer as a sequence of bits */
}
    [items.impl.generics.use]

    پارامتر عمومی زمانی یک پیاده‌سازی را مقید می‌کند که دست‌کم یک بار در یکی از موارد زیر ظاهر شود:

    • صفت پیاده‌سازی‌شده، اگر وجود داشته باشد؛
    • نوع پیاده‌سازی‌کننده؛
    • به‌عنوان نوع وابسته در کران‌های نوعی که پارامتر مقیدکنندهٔ دیگری را در خود دارد.
    [items.impl.generics.constrain]

    پارامترهای نوع و const همیشه باید پیاده‌سازی را مقید کنند. طول‌عمرها اگر در نوع وابسته استفاده شوند باید پیاده‌سازی را مقید کنند.

    نمونه‌های موقعیت‌های مقیدکننده:

    trait Trait{}
trait GenericTrait<T> {}
trait HasAssocType { type Ty; }
struct Struct;
struct GenericStruct<T>(T);
struct ConstGenericStruct<const N: usize>([(); N]);
// T constrains by being an argument to GenericTrait.
impl<T> GenericTrait<T> for i32 { /* ... */ }

// T constrains by being an argument to GenericStruct
impl<T> Trait for GenericStruct<T> { /* ... */ }

// Likewise, N constrains by being an argument to ConstGenericStruct
impl<const N: usize> Trait for ConstGenericStruct<N> { /* ... */ }

// T constrains by being in an associated type in a bound for type `U` which is
// itself a generic parameter constraining the trait.
impl<T, U> GenericTrait<U> for u32 where U: HasAssocType<Ty = T> { /* ... */ }

// Like previous, except the type is `(U, isize)`. `U` appears inside the type
// that includes `T`, and is not the type itself.
impl<T, U> GenericStruct<U> where (U, isize): HasAssocType<Ty = T> { /* ... */ }

    نمونه‌های موقعیت‌های غیرمقیدکننده:

    // The rest of these are errors, since they have type or const parameters that
// do not constrain.

// T does not constrain since it does not appear at all.
impl<T> Struct { /* ... */ }

// N does not constrain for the same reason.
impl<const N: usize> Struct { /* ... */ }

// Usage of T inside the implementation does not constrain the impl.
impl<T> Struct {
    fn uses_t(t: &T) { /* ... */ }
}

// T is used as an associated type in the bounds for U, but U does not constrain.
impl<T, U> Struct where U: HasAssocType<Ty = T> { /* ... */ }

// T is used in the bounds, but not as an associated type, so it does not constrain.
impl<T, U> GenericTrait<U> for u32 where U: GenericTrait<T> {}

    نمونهٔ پارامتر طول‌عمر غیرمقیدکنندهٔ مجاز:

    struct Struct;
impl<'a> Struct {}

    نمونهٔ پارامتر طول‌عمر غیرمقیدکنندهٔ غیرمجاز:

    struct Struct;
trait HasAssocType { type Ty; }
impl<'a> HasAssocType for Struct {
    type Ty = &'a Struct;
}

    ویژگی‌های روی پیاده‌سازی‌ها

    پیاده‌سازی‌ها می‌توانند پیش از کلیدواژهٔ impl ویژگی‌های بیرونی و درون آکولاد دربردارندهٔ آیتم‌های وابسته، ویژگی‌های درونی داشته باشند. ویژگی‌های درونی باید پیش از همهٔ آیتم‌های وابسته بیایند. ویژگی‌های معنادار در اینجا cfg، ‏deprecated، ‏doc و ویژگی‌های بررسی lint هستند.
