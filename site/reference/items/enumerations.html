    [items.enum.intro]

    یک برشمردنی که با نام enum نیز شناخته می‌شود، هم‌زمان یک نوع برشمرده اسمی و مجموعه‌ای از سازنده‌ها را تعریف می‌کند. این سازنده‌ها برای ایجاد یا تطبیق الگو با مقادیر نوع برشمردهٔ متناظر به کار می‌روند.

    [items.enum.decl]

    برشمردنی‌ها با کلیدواژهٔ enum اعلان می‌شوند.

    [items.enum.namespace]

    اعلان enum نوع برشمردنی را در فضای نام نوع ماژول یا بلوکی که در آن قرار دارد تعریف می‌کند.

    نمونه‌ای از یک قلم enum و شیوهٔ استفاده از آن:

    enum Animal {
    Dog,
    Cat,
}

let mut a: Animal = Animal::Dog;
a = Animal::Cat;
    [items.enum.constructor]

    سازنده‌های برشمردنی می‌توانند میدان‌های نام‌دار یا بی‌نام داشته باشند:

    enum Animal {
    Dog(String, f64),
    Cat { name: String, weight: f64 },
}

let mut a: Animal = Animal::Dog("Cocoa".to_string(), 37.2);
a = Animal::Cat { name: "Spotty".to_string(), weight: 2.7 };

    در این مثال، Cat یک گونهٔ برشمردنی ساختارمانند است، در حالی که Dog صرفاً یک گونهٔ برشمردنی نامیده می‌شود.

    [items.enum.fieldless]

    برشمردنی‌ای که هیچ‌یک از سازنده‌هایش میدان نداشته باشند، برشمردنی بدون میدان نامیده می‌شود. برای نمونه، این یک برشمردنی بدون میدان است:

    enum Fieldless {
    Tuple(),
    Struct{},
    Unit,
}
    [items.enum.unit-only]

    اگر یک برشمردنی بدون میدان فقط شامل گونه‌های واحد باشد، برشمردنی فقط‌واحد نامیده می‌شود. برای نمونه:

    enum Enum {
    Foo = 3,
    Bar = 2,
    Baz = 1,
}
    [items.enum.constructor-names]

    سازنده‌های گونه (variant) شبیه تعریف‌های struct هستند و می‌توان آن‌ها را با مسیری که از نام enum آغاز می‌شود ارجاع داد، از جمله در اعلان‌های use.

    [items.enum.constructor-namespace]

    هر گونه، نوع خود را در فضای نام نوع تعریف می‌کند؛ با این حال آن نوع را نمی‌توان به‌عنوان مشخص‌کنندهٔ نوع به کار برد. گونه‌های تاپل‌مانند و واحدمانند همچنین سازنده‌ای در فضای نام مقدار تعریف می‌کنند.

    [items.enum.struct-expr]

    یک گونهٔ ساختارمانند را می‌توان با یک عبارت ساختار نمونه‌سازی کرد.

    [items.enum.tuple-expr]

    یک گونهٔ تاپل‌مانند را می‌توان با یک عبارت فراخوانی یا عبارت ساختار نمونه‌سازی کرد.

    [items.enum.path-expr]

    یک گونهٔ واحدمانند را می‌توان با یک عبارت مسیر یا عبارت ساختار نمونه‌سازی کرد. برای نمونه:

    enum Examples {
    UnitLike,
    TupleLike(i32),
    StructLike { value: i32 },
}

use Examples::*; // Creates aliases to all variants.
let x = UnitLike; // Path expression of the const item.
let x = UnitLike {}; // Struct expression.
let y = TupleLike(123); // Call expression.
let y = TupleLike { 0: 123 }; // Struct expression using integer field names.
let z = StructLike { value: 123 }; // Struct expression.

    [items.enum.discriminant]

    متمایزکننده‌ها

    [items.enum.discriminant.intro]

    هر نمونهٔ برشمردنی یک متمایزکننده دارد: عدد صحیحی که از نظر منطقی به آن وابسته است و برای تعیین گونه‌ای که نمونه در خود دارد به کار می‌رود.

    [items.enum.discriminant.type]

    برشمردنی‌های بدون بازنمایی ابتدایی متمایزکننده‌هایی از نوع isize دارند. با این حال، کامپایلر ممکن است در چیدمان واقعی حافظه از نوعی کوچک‌تر یا روش دیگری برای تشخیص گونه‌ها استفاده کند.

    use core::mem::size_of;
enum E {
    V1 = 0isize, // OK: `isize` is the discriminant type.
    V2,
}

assert!(size_of::<E>() <= size_of::<isize>());
    enum E {
    V = 0u8, // ERROR: Expected `isize`, found `u8`.
}
    [items.enum.discriminant.type-primitive]

    برشمردنی‌های دارای بازنمایی ابتدایی متمایزکننده‌هایی از نوع نام‌برده‌شده در بازنمایی دارند. این قاعده دربارهٔ برشمردنی‌هایی که بازنمایی C را با یک بازنمایی ابتدایی ترکیب می‌کنند نیز برقرار است (بخش layout.repr.primitive-c را ببینید).

    #[repr(u8)]
enum E {
    V = 0u8, // OK: `u8` is the discriminant type.
}
    #[repr(u8)]
enum E {
    V = 0isize, // ERROR: Expected `u8`, found `isize`.
}

    مقداردهی متمایزکننده‌ها

    [items.enum.discriminant.explicit]

    متمایزکننده‌های صریح

    [items.enum.discriminant.explicit.intro]

    در دو وضعیت می‌توان متمایزکنندهٔ یک گونه را صریحاً با قرار دادن = و یک عبارت ثابت پس از نام گونه تعیین کرد:

    [items.enum.discriminant.explicit.unit-only]
    1. اگر برشمردنی «فقط‌واحد» باشد.
    [items.enum.discriminant.explicit.primitive-repr]

    \۲. اگر بازنمایی اولیه به کار رفته باشد. برای نمونه:

    #[repr(u8)]
enum Enum {
    Unit = 3,
    Tuple(u16),
    Struct {
        a: u8,
        b: u16,
    } = 1,
}
    [items.enum.discriminant.implicit]

    متمایزکننده‌های ضمنی

    اگر متمایزکنندهٔ یک گونه مشخص نشده باشد، مقدار آن یک واحد بیشتر از متمایزکنندهٔ گونهٔ پیشین در اعلان خواهد بود. اگر متمایزکنندهٔ نخستین گونهٔ اعلان مشخص نشده باشد، مقدار آن صفر است.

    enum Foo {
    Bar,            // 0
    Baz = 123,      // 123
    Quux,           // 124
}

let baz_discriminant = Foo::Baz as u32;
assert_eq!(baz_discriminant, 123);
    [items.enum.discriminant.restrictions]

    محدودیت‌ها

    [items.enum.discriminant.restrictions.same-discriminant]

    اشتراک یک متمایزکننده میان دو گونه خطاست.

    enum SharedDiscriminantError {
    SharedA = 1,
    SharedB = 1,
}

enum SharedDiscriminantError2 {
    Zero,       // 0
    One,        // 1
    OneToo = 1, // 1 (collision with previous!)
}
    [items.enum.discriminant.restrictions.above-max-discriminant]

    وجود یک متمایزکنندهٔ نامشخص پس از متمایزکننده‌ای که بیشینهٔ مقدار قابل نمایش در اندازهٔ متمایزکننده است نیز خطاست.

    #[repr(u8)]
enum OverflowingDiscriminantError {
    Max = 255,
    MaxPlusOne, // Would be 256, but that overflows the enum.
}

#[repr(u8)]
enum OverflowingDiscriminantError2 {
    MaxMinusOne = 254, // 254
    Max,               // 255
    MaxPlusOne,        // Would be 256, but that overflows the enum.
}
    [items.enum.discriminant.restrictions.generics]

    مقداردهنده‌های صریح متمایزکنندهٔ برشمردنی نمی‌توانند از پارامترهای عمومیِ برشمردنی دربرگیرنده استفاده کنند.

    #[repr(u32)]
enum E<'a, T, const N: u32> {
    Lifetime(&'a T) = {
        let a: &'a (); // ERROR.
        1
    },
    Type(T) = {
        let x: T; // ERROR.
        2
    },
    Const = N, // ERROR.
}

    دسترسی به متمایزکننده

    از راه mem::discriminant

    [items.enum.discriminant.access-opaque]

    تابع std::mem::discriminant ارجاعی کدر به متمایزکنندهٔ یک مقدار برشمردنی برمی‌گرداند که قابل مقایسه است. از این ارجاع نمی‌توان مقدار متمایزکننده را به دست آورد.

    [items.enum.discriminant.coercion]

    تبدیل نوع

    [items.enum.discriminant.coercion.intro]

    اگر یک برشمردنی فقط‌واحد باشد (هیچ گونهٔ تاپلی یا ساختاری نداشته باشد)، می‌توان مستقیماً با یک تبدیل عددی به متمایزکنندهٔ آن دسترسی یافت؛ برای نمونه:

    enum Enum {
    Foo,
    Bar,
    Baz,
}

assert_eq!(0, Enum::Foo as isize);
assert_eq!(1, Enum::Bar as isize);
assert_eq!(2, Enum::Baz as isize);
    [items.enum.discriminant.coercion.fieldless]

    برشمردنی‌های بدون میدان را زمانی می‌توان تبدیل کرد که متمایزکنندهٔ صریح نداشته باشند، یا فقط گونه‌های واحد آن‌ها صریح باشند.

    enum Fieldless {
    Tuple(),
    Struct{},
    Unit,
}

assert_eq!(0, Fieldless::Tuple() as isize);
assert_eq!(1, Fieldless::Struct{} as isize);
assert_eq!(2, Fieldless::Unit as isize);

#[repr(u8)]
enum FieldlessWithDiscriminants {
    First = 10,
    Tuple(),
    Second = 20,
    Struct{},
    Unit,
}

assert_eq!(10, FieldlessWithDiscriminants::First as u8);
assert_eq!(11, FieldlessWithDiscriminants::Tuple() as u8);
assert_eq!(20, FieldlessWithDiscriminants::Second as u8);
assert_eq!(21, FieldlessWithDiscriminants::Struct{} as u8);
assert_eq!(22, FieldlessWithDiscriminants::Unit as u8);

    تبدیل اشاره‌گر

    [items.enum.discriminant.access-memory]

    اگر برشمردنی یک بازنمایی ابتدایی مشخص کرده باشد، می‌توان به‌شکلی قابل اتکا و از راه تبدیل ناامن اشاره‌گر به متمایزکننده دسترسی یافت:

    #[repr(u8)]
enum Enum {
    Unit,
    Tuple(bool),
    Struct{a: bool},
}

impl Enum {
    fn discriminant(&self) -> u8 {
        unsafe { *(self as *const Self as *const u8) }
    }
}

let unit_like = Enum::Unit;
let tuple_like = Enum::Tuple(true);
let struct_like = Enum::Struct{a: false};

assert_eq!(0, unit_like.discriminant());
assert_eq!(1, tuple_like.discriminant());
assert_eq!(2, struct_like.discriminant());
    [items.enum.empty]

    برشمردنی‌های بدون گونه

    [items.enum.empty.intro]

    برشمردنی‌هایی با صفر گونه، برشمردنی بدون گونه نامیده می‌شوند. از آنجا که هیچ مقدار معتبری ندارند، نمی‌توان از آن‌ها نمونه ساخت.

    enum ZeroVariants {}
    [items.enum.empty.uninhabited]

    برشمردنی‌های بدون گونه معادل نوع هرگز هستند، اما نمی‌توان آن‌ها را به نوع‌های دیگر واداشت.

    enum ZeroVariants {}
let x: ZeroVariants = panic!();
let y: u32 = x; // mismatched type error
    [items.enum.variant-visibility]

    نمایانی گونه‌ها

    گونه‌های برشمردنی از نظر نحوی یک حاشیه‌نویسی Visibility را می‌پذیرند، اما هنگام اعتبارسنجی برشمردنی رد می‌شوند. این ویژگی اجازه می‌دهد اقلام با نحوی یکپارچه در بافت‌های مختلفی که استفاده می‌شوند تجزیه شوند.

    macro_rules! mac_variant {
    ($vis:vis $name:ident) => {
        enum $name {
            $vis Unit,

            $vis Tuple(u8, u16),

            $vis Struct { f: u8 },
        }
    }
}

// Empty `vis` is allowed.
mac_variant! { E }

// This is allowed, since it is removed before being validated.
#[cfg(false)]
enum E {
    pub U,
    pub(crate) T(u8),
    pub(super) T { f: String },
}
