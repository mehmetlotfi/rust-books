    [items.use]

    اعلان‌های use

    [items.use.syntax]

    Syntax
    UseDeclaration → use UseTree ;

    UseTree →
          ( SimplePath? :: )? *
        | ( SimplePath? :: )? { ( UseTree ( , UseTree )* ,? )? }
        | SimplePath ( as ( IDENTIFIER | _ ) )?

    [items.use.intro]

    یک اعلان use یک یا چند مقیدسازی نام محلی ایجاد می‌کند که مترادف مسیر دیگری هستند. معمولاً از اعلان use برای کوتاه کردن مسیر لازم جهت ارجاع به قلمی در ماژول استفاده می‌شود. این اعلان‌ها می‌توانند در ماژول‌ها و بلوک‌ها، معمولاً در ابتدای آن‌ها، ظاهر شوند. اعلان use گاهی واردسازی و اگر عمومی باشد بازصادرسازی نیز نامیده می‌شود.

    [items.use.forms]

    اعلان‌های use از چند میان‌بر کاربردی پشتیبانی می‌کنند:

    [items.use.forms-multiple]
    • مقید کردن هم‌زمان فهرستی از مسیرها با پیشوند مشترک، با استفاده از نحو آکولاد: use a::b::{c, d, e::f, g::h::i};
    [items.use.forms-self]
    • مقید کردن هم‌زمان فهرستی از مسیرها با پیشوند مشترک و ماژول والد مشترکشان، با استفاده از کلیدواژهٔ self؛ مانند use a::b::{self, c, d::e};
    [items.use.forms-as]
    • مقید کردن دوبارهٔ نام هدف با یک نام محلی جدید، با نحو use p::q::r as x;. این روش را می‌توان همراه دو قابلیت پیشین نیز به کار برد: use a::b::{self as ab, c as abc}.
    [items.use.forms-glob]
    • مقید کردن همهٔ مسیرهای مطابق یک پیشوند مشخص، با نحو نویسهٔ عام ستاره: use a::b::*;.
    [items.use.forms-nesting]
    • تودرتو کردن چندبارهٔ گروه‌هایی از قابلیت‌های پیشین؛ مانند use a::b::{self as ab, c, d::{*, e::f}};

    نمونه‌ای از اعلان‌های use:

    use std::collections::hash_map::{self, HashMap};

fn foo<T>(_: T){}
fn bar(map1: HashMap<String, usize>, map2: hash_map::HashMap<String, usize>){}

fn main() {
    // use declarations can also exist inside of functions
    use std::option::Option::{Some, None};

    // Equivalent to 'foo(vec![std::option::Option::Some(1.0f64),
    // std::option::Option::None]);'
    foo(vec![Some(1.0f64), None]);

    // Both `hash_map` and `HashMap` are in scope.
    let map1 = HashMap::new();
    let map2 = hash_map::HashMap::new();
    bar(map1, map2);
}
    [items.use.visibility]

    نمایانی use

    [items.use.visibility.intro]

    اعلان‌های use نیز مانند آیتم‌ها، به‌طور پیش‌فرض برای ماژول دربرگیرندهٔ خود خصوصی هستند. همچنین مانند آیتم‌ها، اگر اعلان use با کلیدواژهٔ pub مشخص شود، می‌تواند عمومی باشد. چنین اعلان useای برای بازصادرکردن (re-export) یک نام به‌کار می‌رود. بنابراین یک اعلان use عمومی می‌تواند یک نام عمومی را به تعریف هدفی دیگر هدایت کند: حتی تعریفی با مسیر کانونی خصوصی در ماژولی دیگر.

    [items.use.visibility.unambiguous]

    اگر دنباله‌ای از این هدایت‌ها چرخه بسازد یا نتوان آن را بدون ابهام تفکیک کرد، خطای زمان کامپایل رخ می‌دهد.

    نمونه‌ای از بازصادرسازی:

    mod quux {
    pub use self::foo::{bar, baz};
    pub mod foo {
        pub fn bar() {}
        pub fn baz() {}
    }
}

fn main() {
    quux::bar();
    quux::baz();
}

    در این مثال، ماژول quux دو نام عمومی تعریف‌شده در foo را بازصادر می‌کند.

    [items.use.path]

    مسیرهای use

    [items.use.path.intro]

    مسیرهای مجاز در یک قلم use از دستور زبان SimplePath پیروی می‌کنند و مشابه مسیرهایی هستند که در یک عبارت به کار می‌روند. آن‌ها می‌توانند برای موارد زیر مقیدسازی بسازند:

    • آیتم‌های قابل نام‌گذاری
    • نوع‌های داخلی
    • ویژگی‌ها
    • ویژگی‌ها
    • ماکروهای مشتق‌سازی
    • macro_rules
    [items.use.path.disallowed]

    آن‌ها نمی‌توانند اقلام وابسته، پارامترهای عمومی، متغیرهای محلی، مسیرهای دارای Self یا ویژگی‌های ابزار را وارد کنند. محدودیت‌های بیشتر در ادامه شرح داده شده‌اند.

    [items.use.path.namespace]

    use برای همهٔ فضاهای نام موجودیت‌های واردشده مقیدسازی می‌سازد؛ تنها استثنا این است که واردسازی self فقط از فضای نام نوع وارد می‌کند، چنان‌که در ادامه شرح داده شده است. برای نمونه، کد زیر ساخت مقیدسازی برای یک نام یکسان در دو فضای نام را نشان می‌دهد:

    mod stuff {
    pub struct Foo(pub i32);
}

// Imports the `Foo` type and the `Foo` constructor.
use stuff::Foo;

fn example() {
    let ctor = Foo; // Uses `Foo` from the value namespace.
    let x: Foo = ctor(123); // Uses `Foo` From the type namespace.
}
    [items.use.path.edition2018]

    تفاوت‌های ویرایش 2018 در ویرایش ۲۰۱۵، مسیرهای use نسبت به ریشهٔ کریت هستند. برای نمونه:

    mod foo {
    pub mod example { pub mod iter {} }
    pub mod baz { pub fn foobaz() {} }
}
mod bar {
    // Resolves `foo` from the crate root.
    use foo::example::iter;
    // The `::` prefix explicitly resolves `foo`
    // from the crate root.
    use ::foo::baz::foobaz;
}

fn main() {}

    ویرایش ۲۰۱۵ اجازه نمی‌دهد اعلان‌های use به پیش‌درآمد خارجی ارجاع دهند. بنابراین در ویرایش ۲۰۱۵ برای ارجاع به یک کریت خارجی در اعلان use همچنان به اعلان extern crate نیاز است. از ویرایش ۲۰۱۸ به بعد، اعلان‌های use می‌توانند وابستگی به یک کریت خارجی را به همان شیوهٔ extern crate مشخص کنند.

    [items.use.as]

    تغییر نام با as

    با کلیدواژهٔ as می‌توان نام موجودیت واردشده را تغییر داد. برای نمونه:

    // Creates a non-public alias `bar` for the function `foo`.
use inner::foo as bar;

mod inner {
    pub fn foo() {}
}
    [items.use.multiple-syntax]

    نحو آکولاد

    [items.use.multiple-syntax.intro]

    می‌توان در آخرین قطعهٔ مسیر از آکولاد استفاده کرد تا چند موجودیت از قطعهٔ پیشین، یا اگر قطعهٔ پیشینی وجود ندارد از دامنهٔ جاری، وارد شوند. آکولادها می‌توانند تودرتو باشند و درختی از مسیرها بسازند؛ هر گروه قطعه از نظر منطقی با والد خود ترکیب می‌شود تا مسیری کامل بسازد.

    // Creates bindings to:
// - `std::collections::BTreeSet`
// - `std::collections::hash_map`
// - `std::collections::hash_map::HashMap`
use std::collections::{BTreeSet, hash_map::{self, HashMap}};
    [items.use.multiple-syntax.empty]

    آکولاد خالی چیزی وارد نمی‌کند، هرچند مسیر پیش از آن برای در دسترس بودن اعتبارسنجی می‌شود.

    [items.use.multiple-syntax.edition2018]

    تفاوت‌های ویرایش 2018 در ویرایش ۲۰۱۵، مسیرها نسبت به ریشهٔ کریت هستند؛ بنابراین واردسازی‌ای مانند use {foo, bar}; نام‌های foo و bar را از ریشهٔ کریت وارد می‌کند. از ویرایش ۲۰۱۸ به بعد، این نام‌ها نسبت به دامنهٔ جاری هستند.

    [items.use.self]

    واردسازی‌های self

    [items.use.self.intro]

    در نحو آکولاد می‌توان از کلیدواژهٔ self برای ساخت مقیدسازی موجودیت والد با نام خودش استفاده کرد.

    mod stuff {
    pub fn foo() {}
    pub fn bar() {}
}
mod example {
    // Creates a binding for `stuff` and `foo`.
    use crate::stuff::{self, foo};
    pub fn baz() {
        foo();
        stuff::bar();
    }
}
fn main() {}

    Note

    نکته self را می‌توان به‌عنوان نخستین قطعهٔ یک مسیر نیز به کار برد. کاربرد self به‌عنوان نخستین قطعه و درون آکولاد use از نظر منطقی یکسان است: منظور ماژول جاریِ قطعهٔ والد، یا در نبود قطعهٔ والد، ماژول جاری است. برای اطلاعات بیشتر دربارهٔ معنای self ابتدایی، بخش self در فصل مسیرها را ببینید.

    [items.use.self.trailing]

    self می‌تواند به‌عنوان آخرین قطعهٔ مسیر use و پس از :: ظاهر شود. مسیر P::self معادل P::{self} و P::self as name معادل P::{self as name} است.

    mod m {
    pub enum E { V1, V2 }
}
use m::self as _; // Equivalent to `use m::{self as _};`.
use m::E::self; // Equivalent to `use m::E::{self};`.
fn main() {}

    Note

    نکته برای محدودیت‌های مسیر پیشین، paths.qualifiers.mod-self.trailing را ببینید.

    [items.use.self.module]

    وقتی self درون نحو آکولاد به کار می‌رود، مسیر پیش از گروه آکولاد باید به یک ماژول، برشمردنی یا صفت تفکیک شود.

    mod m {
    pub enum E { V1, V2 }
    pub trait Tr { fn f(&self); }
}
use m::{self as _}; // OK: Modules can be parents of `self`.
use m::E::{self, V1}; // OK: Enums can be parents of `self`.
use m::Tr::{self}; // OK: Traits can be parents of `self`.
fn main() {}
    struct S {}
use S::{self as _}; // ERROR: Structs cannot be parents of `self`.
fn main() {}
    [items.use.self.namespace]

    self فقط از فضای نام نوع موجودیت والد مقیدسازی می‌سازد. برای نمونه، در کد زیر تنها ماژول foo وارد می‌شود:

    mod bar {
    pub mod foo {}
    pub fn foo() {}
}

// This only imports the module `foo`. The function `foo` lives in
// the value namespace and is not imported.
use bar::foo::{self};

fn main() {
    foo(); //~ ERROR `foo` is a module
}
    [items.use.glob]

    واردسازی‌های سراسری

    [items.use.glob.intro]

    نویسهٔ * را می‌توان به‌عنوان آخرین قطعهٔ مسیر use به کار برد تا همهٔ موجودیت‌های قابل واردسازی از موجودیت قطعهٔ پیشین وارد شوند. برای نمونه:

    // Creates a non-public alias to `bar`.
use foo::*;

mod foo {
    fn i_am_private() {}
    enum Example {
        V1,
        V2,
    }
    pub fn bar() {
        // Creates local aliases to `V1` and `V2`
        // of the `Example` enum.
        use Example::*;
        let x = V1;
    }
}
    [items.use.glob.shadowing]

    اقلام و واردسازی‌های نام‌دار اجازه دارند در همان فضای نام روی نام‌های حاصل از واردسازی سراسری سایه بیندازند. یعنی اگر نامی پیش‌تر به‌وسیلهٔ قلم دیگری در همان فضای نام تعریف شده باشد، واردسازی سراسری زیر سایه قرار می‌گیرد. برای نمونه:

    // This creates a binding to the `clashing::Foo` tuple struct
// constructor, but does not import its type because that would
// conflict with the `Foo` struct defined here.
//
// Note that the order of definition here is unimportant.
use clashing::*;
struct Foo {
    field: f32,
}

fn do_stuff() {
    // Uses the constructor from `clashing::Foo`.
    let f1 = Foo(123);
    // The struct expression uses the type from
    // the `Foo` struct defined above.
    let f2 = Foo { field: 1.0 };
    // `Bar` is also in scope due to the glob import.
    let z = Bar {};
}

mod clashing {
    pub struct Foo(pub i32);
    pub struct Bar {}
}

    Note

    برای ناحیه‌هایی که سایه‌اندازی در آن‌ها مجاز نیست، ابهام‌های تفکیک نام را ببینید.

    [items.use.glob.last-segment-only]

    * را نمی‌توان به‌عنوان نخستین یا یکی از قطعه‌های میانی به کار برد.

    [items.use.glob.self-import]

    * را نمی‌توان برای وارد کردن محتوای یک ماژول به خودش به کار برد؛ مانند use self::*;.

    [items.use.glob.edition2018]

    تفاوت‌های ویرایش 2018 در ویرایش ۲۰۱۵ مسیرها نسبت به ریشهٔ کریت هستند؛ بنابراین واردسازی‌ای مانند use *; معتبر است و همه‌چیز را از ریشهٔ کریت وارد می‌کند. این شکل را نمی‌توان در خود ریشهٔ کریت به کار برد.

    [items.use.as-underscore]

    واردسازی با زیرخط

    [items.use.as-underscore.intro]

    می‌توان با استفاده از زیرخط به شکل use path as _، اقلام را بدون مقید کردن به یک نام وارد کرد. این روش به‌ویژه برای وارد کردن یک صفت سودمند است تا بتوان بدون وارد کردن نماد صفت از متدهایش استفاده کرد؛ برای نمونه وقتی نماد صفت ممکن است با نماد دیگری تداخل داشته باشد. نمونهٔ دیگر، پیوند به یک کریت خارجی بدون وارد کردن نام آن است.

    mod foo {
    pub trait Zoo {
        fn zoo(&self) {}
    }

    impl<T> Zoo for T {}
}

use self::foo::Zoo as _;
struct Zoo;  // Underscore import avoids name conflict with this item.

fn main() {
    let z = Zoo;
    z.zoo();
}
    [items.use.as-underscore.glob]

    واردسازی‌های سراسری ستاره‌ای، اقلام واردشده با _ را در شکل نام‌ناپذیرشان وارد می‌کنند.

    [items.use.as-underscore.macro]

    نمادهای یکتا و نام‌ناپذیر پس از گسترش ماکرو ساخته می‌شوند تا ماکروها بتوانند با ایمنی چند ارجاع به واردسازی‌های _ تولید کنند. برای نمونه، کد زیر نباید خطایی ایجاد کند:

    macro_rules! m {
    ($item: item) => { $item $item }
}

m!(use std as _;);
// This expands to:
// use std as _;
// use std as _;
    [items.use.restrictions]

    محدودیت‌ها

    [items.use.restrictions.intro]

    قواعد زیر محدودیت‌های اعلان‌های معتبر use هستند.

    [items.use.restrictions.crate-alias]

    هنگام استفاده از crate برای وارد کردن کریت جاری، باید برای تعریف نام مقیدسازی از as استفاده کنید.

    مثال

    use crate as root;
use crate::{self as root2};

// Not allowed:
// use crate;
// use crate::{self};
    [items.use.restrictions.macro-crate-alias]

    هنگام استفاده از $crate در رونویس ماکرو برای وارد کردن کریت جاری، باید برای تعریف نام مقیدسازی از as استفاده کنید.

    مثال

    macro_rules! import_crate_root {
    () => {
        use $crate as my_crate;
        use $crate::{self as my_crate2};
    };
}
    [items.use.restrictions.self-alias]

    هنگام استفاده از self برای وارد کردن ماژول جاری، باید برای تعریف نام مقیدسازی از as استفاده کنید.

    مثال

    use {self as this_module};
use self as this_module2;
use self::{self as this_module3};

// Not allowed:
// use {self};
// use self;
// use self::{self};
    [items.use.restrictions.super-alias]

    هنگام استفاده از super برای وارد کردن یک ماژول والد، باید برای تعریف نام مقیدسازی از as استفاده کنید.

    مثال

    mod a {
    mod b {
        use super as parent;
        use super::{self as parent2};
        use self::super as parent3;
        use super::super as grandparent;
        use super::super::{self as grandparent2};

        // Not allowed:
        // use super;
        // use super::{self};
        // use self::super;
        // use super::super;
        // use super::super::{self};
    }
}
    [items.use.restrictions.extern-prelude]

    :: به‌عنوان پیش‌درآمد خارجی قابل وارد کردن نیست.

    مثال

    use ::{self as root}; //~ Error

    تفاوت‌های ویرایش 2018 در ویرایش ۲۰۱۵، پیشوند :: به ریشهٔ کریت اشاره می‌کند؛ بنابراین use ::{self as root}; مجاز است، زیرا با use crate::{self as root}; یکسان است. از ویرایش ۲۰۱۸ به بعد، پیشوند :: به پیش‌درآمد خارجی اشاره می‌کند که نمی‌توان آن را مستقیماً وارد کرد.

    use ::{self as root}; //~ Ok
    [items.use.restrictions.duplicate-name]

    مانند تعریف هر قلم دیگر، واردسازی‌های use نمی‌توانند در یک ماژول یا بلوک، مقیدسازی‌های تکراری با نام یکسان در همان فضای نام بسازند.

    [items.use.restrictions.variant]

    مسیرهای use نمی‌توانند از راه یک نام مستعار نوع به گونه‌های برشمردنی ارجاع دهند.

    مثال

    enum MyEnum {
  MyVariant
}
type TypeAlias = MyEnum;

use MyEnum::MyVariant; //~ OK
use TypeAlias::MyVariant; //~ ERROR
