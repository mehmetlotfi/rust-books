    [items.generics]

    پارامترهای عمومی

    [items.generics.syntax]

    Syntax
    GenericParams → < ( GenericParam ( , GenericParam )* ,? )? >

    GenericParam → OuterAttribute* ( LifetimeParam | TypeParam | ConstParam )

    LifetimeParam → Lifetime ( : LifetimeBounds? )?

    TypeParam → IDENTIFIER ( : Bounds? )? ( = Type )?

    ConstParam →
        const IDENTIFIER : Type
        ( = ( BlockExpression | IDENTIFIER | -? LiteralExpression ) )?

    [items.generics.intro]

    تابع‌ها، نام‌های مستعار نوع، structها، enumها، unionها، traitها و پیاده‌سازی‌ها می‌توانند با نوع‌ها، ثابت‌ها و طول‌عمرها پارامتری شوند. این پارامترها درون براکت‌های زاویه‌ای (<...>) فهرست می‌شوند، معمولاً بلافاصله پس از نام آیتم و پیش از تعریف آن. در پیاده‌سازی‌ها که نام ندارند، این پارامترها مستقیماً پس از impl می‌آیند.

    [items.generics.decl-order]

    ترتیب پارامترهای عمومی محدود است: ابتدا پارامترهای طول‌عمر و سپس پارامترهای نوع و const به‌شکل درهم‌آمیخته می‌آیند.

    [items.generics.duplicate-params]

    یک نام پارامتر را نمی‌توان بیش از یک بار در فهرست GenericParams اعلان کرد.

    چند نمونه از آیتم‌های دارای پارامتر نوع، const و طول‌عمر:

    fn foo<'a, T>() {}
trait A<U> {}
struct Ref<'a, T> where T: 'a { r: &'a T }
struct InnerArray<T, const N: usize>([T; N]);
struct EitherOrderWorks<const N: bool, U>(U);
    [items.generics.scope]

    پارامترهای عمومی در تعریف آیتمی که در آن اعلان شده‌اند در حوزه هستند. آن‌ها برای آیتم‌های اعلان‌شده در بدنهٔ تابع در حوزه نیستند؛ اعلان آیتم‌ها را ببینید. جزئیات بیشتر در حوزه‌های پارامتر عمومی آمده است.

    [items.generics.builtin-generic-types]

    ارجاع‌ها، اشاره‌گرهای خام، آرایه‌ها، sliceها، تاپل‌ها و اشاره‌گرهای تابع نیز پارامترهای طول عمر یا نوع دارند، اما با نحو مسیر به آن‌ها ارجاع داده نمی‌شود.

    [items.generics.invalid-lifetimes]

    '_ و 'static نام معتبر پارامتر طول‌عمر نیستند.

    [items.generics.const]

    عمومی‌های Const

    [items.generics.const.intro]

    پارامترهای عمومی const اجازه می‌دهند آیتم‌ها روی مقادیر ثابت عمومی باشند.

    [items.generics.const.namespace]

    شناسهٔ const نامی برای پارامتر ثابت در فضای نام مقدار معرفی می‌کند و همهٔ نمونه‌های آیتم باید با مقداری از نوع داده‌شده نمونه‌سازی شوند.

    [items.generics.const.allowed-types]

    تنها نوع‌های مجاز پارامتر const عبارت‌اند از u8، ‏u16، ‏u32، ‏u64، ‏u128، ‏usize، ‏i8، ‏i16، ‏i32، ‏i64، ‏i128، ‏isize، ‏char و bool.

    [items.generics.const.use]

    پارامتر const را می‌توان هرجا آیتم const مجاز است به‌کار برد؛ با این استثنا که در نوع یا عبارت تکرار آرایه باید به‌تنهایی ظاهر شود، همان‌گونه که در ادامه توضیح داده می‌شود. محل‌های مجاز:

    1. به‌عنوان const اعمال‌شده به هر نوعی که بخشی از امضای آیتم موردنظر است.
    2. به‌عنوان بخشی از عبارت const برای تعریف ثابت وابسته یا پارامتر یک نوع وابسته.
    3. به‌عنوان مقدار در هر عبارت زمان اجرا در بدنهٔ تابع‌های آیتم.
    4. به‌عنوان پارامتر هر نوع استفاده‌شده در بدنهٔ تابع‌های آیتم.
    5. به‌عنوان بخشی از نوع هر فیلد آیتم.
    // Examples where const generic parameters can be used.

// Used in the signature of the item itself.
fn foo<const N: usize>(arr: [i32; N]) {
    // Used as a type within a function body.
    let x: [i32; N];
    // Used as an expression.
    println!("{}", N * 2);
}

// Used as a field of a struct.
struct Foo<const N: usize>([i32; N]);

impl<const N: usize> Foo<N> {
    // Used as an associated constant.
    const CONST: usize = N * 4;
}

trait Trait {
    type Output;
}

impl<const N: usize> Trait for Foo<N> {
    // Used as an associated type.
    type Output = [i32; N];
}
    // Examples where const generic parameters cannot be used.
fn foo<const N: usize>() {
    // Cannot use in item definitions within a function body.
    const BAD_CONST: [usize; N] = [1; N];
    static BAD_STATIC: [usize; N] = [1; N];
    fn inner(bad_arg: [usize; N]) {
        let bad_value = N * 2;
    }
    type BadAlias = [usize; N];
    struct BadStruct([usize; N]);
}
    [items.generics.const.standalone]

    به‌عنوان محدودیت بیشتر، پارامترهای const درون نوع یا عبارت تکرار آرایه فقط می‌توانند آرگومانی مستقل باشند. در این بافت‌ها فقط به‌شکل عبارت مسیر تک‌قطعه‌ای و احتمالاً درون بلوک، مانند N یا {N}، مجازند و نمی‌توان آن‌ها را با عبارت‌های دیگر ترکیب کرد.

    // Examples where const parameters may not be used.

// Not allowed to combine in other expressions in types, such as the
// arithmetic expression in the return type here.
fn bad_function<const N: usize>() -> [u8; {N + 1}] {
    // Similarly not allowed for array repeat expressions.
    [1; {N + 1}]
}
    [items.generics.const.argument]

    آرگومان const در یک مسیر مقدار ثابتی را مشخص می‌کند که باید برای آیتم استفاده شود.

    [items.generics.const.argument-const-expr]

    آرگومان باید یا const استنتاج‌شده باشد یا عبارت const از نوع نسبت‌داده‌شده به پارامتر const. عبارت const باید عبارت بلوکی محصور در آکولاد باشد، مگر اینکه یک قطعهٔ مسیر منفرد (IDENTIFIER) یا لفظ با - آغازین اختیاری باشد.

    Note

    نکته این محدودیت نحوی برای جلوگیری از نیاز به پیش‌نگری نامتناهی هنگام تجزیهٔ عبارت درون نوع ضروری است.

    struct S<const N: i64>;
const C: i64 = 1;
fn f<const N: i64>() -> S<N> { S }

let _ = f::<1>(); // Literal.
let _ = f::<-1>(); // Negative literal.
let _ = f::<{ 1 + 2 }>(); // Constant expression.
let _ = f::<C>(); // Single segment path.
let _ = f::<{ C + 1 }>(); // Constant expression.
let _: S<1> = f::<_>(); // Inferred const.
let _: S<1> = f::<(((_)))>(); // Inferred const.

    Note

    نکته در فهرست آرگومان عمومی، const استنتاج‌شده به‌عنوان نوع استنتاج‌شده تجزیه می‌شود، اما از نظر معنایی نوع جداگانه‌ای از آرگومان عمومی const است.

    [items.generics.const.inferred]

    جایی که آرگومان const انتظار می‌رود، می‌توان از _ با هر تعداد پرانتز متناظر اختیاری استفاده کرد که const استنتاج‌شده نام دارد؛ قواعد مسیر و قواعد عبارت آرایه را ببینید. این نحو از کامپایلر می‌خواهد در صورت امکان آرگومان const را از اطلاعات پیرامون استنتاج کند.

    fn make_buf<const N: usize>() -> [u8; N] {
    [0; _]
    //  ^ Infers `N`.
}
let _: [u8; 1024] = make_buf::<_>();
//                             ^ Infers `1024`.

    Note

    نکته const استنتاج‌شده از نظر معنایی یک عبارت نیست و درون آکولاد پذیرفته نمی‌شود.

    fn f<const N: usize>() -> [u8; N] { [0; _] }
let _: [_; 1] = f::<{ _ }>();
//                    ^ ERROR `_` not allowed here
    [items.generics.const.inferred-constraint]

    const استنتاج‌شده را نمی‌توان در امضای آیتم به‌کار برد.

    fn f<const N: usize>(x: [u8; N]) -> [u8; _] { x }
//                                       ^ ERROR not allowed
    [items.generics.const.type-ambiguity]

    هرگاه ابهامی وجود داشته باشد که یک آرگومان عمومی می‌تواند به‌عنوان آرگومان نوع تفسیر شود یا آرگومان ثابت، همیشه به‌عنوان نوع تفسیر می‌شود. قرار دادن آرگومان در یک عبارت بلوکی می‌تواند آن را وادار کند که به‌عنوان آرگومان ثابت تفسیر شود.

    type N = u32;
struct Foo<const N: usize>;
// The following is an error, because `N` is interpreted as the type alias `N`.
fn foo<const N: usize>() -> Foo<N> { todo!() } // ERROR
// Can be fixed by wrapping in braces to force it to be interpreted as the `N`
// const parameter:
fn bar<const N: usize>() -> Foo<{ N }> { todo!() } // ok
    [items.generics.const.variance]

    برخلاف پارامترهای نوع و طول‌عمر، پارامترهای const را می‌توان بدون استفاده در آیتم پارامتری اعلان کرد؛ به‌جز پیاده‌سازی‌ها طبق توضیح پیاده‌سازی‌های عمومی:

    // ok
struct Foo<const N: usize>;
enum Bar<const M: usize> { A, B }

// ERROR: unused parameter
struct Baz<T>;
struct Biz<'a>;
struct Unconstrained;
impl<const N: usize> Unconstrained {}
    [items.generics.const.exhaustiveness]

    هنگام تفکیک تعهد کران صفت، جامع‌بودن همهٔ پیاده‌سازی‌های پارامتر const برای تشخیص برآورده‌شدن کران در نظر گرفته نمی‌شود. در مثال زیر، با اینکه همهٔ مقادیر ممکن bool پیاده‌سازی شده‌اند، برآورده‌نشدن کران صفت همچنان خطاست:

    struct Foo<const B: bool>;
trait Bar {}
impl Bar for Foo<true> {}
impl Bar for Foo<false> {}

fn needs_bar(_: impl Bar) {}
fn generic<const B: bool>() {
    let v = Foo::<B>;
    needs_bar(v); // ERROR: trait bound `Foo<B>: Bar` is not satisfied
}
    [items.generics.where]

    بندهای Where

    [items.generics.where.syntax]

    Syntax
    WhereClause → where ( WhereClauseItem , )* WhereClauseItem?

    WhereClauseItem →
          LifetimeWhereClauseItem
        | TypeBoundWhereClauseItem

    LifetimeWhereClauseItem → Lifetime : LifetimeBounds?

    TypeBoundWhereClauseItem → ForLifetimes? Type : Bounds?

    [items.generics.where.intro]

    بندهای where راه دیگری برای مشخص‌کردن کران پارامترهای نوع و طول‌عمر و نیز کران روی نوع‌هایی هستند که پارامتر نوع نیستند.

    [items.generics.where.higher-ranked-lifetimes]

    کلیدواژهٔ for می‌تواند طول‌عمرهای رتبه‌بالاتر را معرفی کند و فقط پارامترهای LifetimeParam را می‌پذیرد.

    struct A<T>
where
    T: Iterator,            // Could use A<T: Iterator> instead
    T::Item: Copy,          // Bound on an associated type
    String: PartialEq<T>,   // Bound on `String`, using the type parameter
    i32: Default,           // Allowed, but not useful
{
    f: T,
}
    [items.generics.attributes]

    ویژگی‌ها

    پارامترهای عمومی طول‌عمر و نوع، ویژگی‌ها را می‌پذیرند. هیچ ویژگی داخلی در این موقعیت اثری ندارد، هرچند ویژگی‌های derive سفارشی می‌توانند به آن معنا دهند.

    این مثال استفاده از ویژگی derive سفارشی برای تغییر معنای پارامتر عمومی را نشان می‌دهد.

    // Assume that the derive for MyFlexibleClone declared `my_flexible_clone` as
// an attribute it understands.
#[derive(MyFlexibleClone)]
struct Foo<#[my_flexible_clone(unbounded)] H> {
    a: *const H
}
