    [paths]

    مسیرها

    [paths.intro]

    مسیر دنباله‌ای از یک یا چند قطعهٔ مسیر است که با توکن‌های :: از هم جدا شده‌اند. مسیرها برای ارجاع به اقلام، مقادیر، نوع‌ها، ماکروها و ویژگی‌ها به کار می‌روند.

    دو نمونه از مسیرهای ساده که فقط از قطعه‌های شناسه تشکیل شده‌اند:

    x;
x::y::z;

    گونه‌های مسیر

    [paths.simple]

    مسیرهای ساده

    [paths.simple.syntax]

    Syntax
    SimplePath →
        ::? SimplePathSegment ( :: SimplePathSegment )*

    SimplePathSegment →
        IDENTIFIER | super | self | crate | $crate

    [paths.simple.intro]

    مسیرهای ساده در نشانگرهای نمایانی، ویژگی‌ها، ماکروها و اقلام use به کار می‌روند. برای نمونه:

    use std::io::{self, Write};
mod m {
    #[clippy::cyclomatic_complexity = "0"]
    pub (in super) fn f1() {}
}
    [paths.expr]

    مسیرها در عبارت‌ها

    [paths.expr.syntax]

    Syntax
    PathInExpression →
        ::? PathExprSegment ( :: PathExprSegment )*

    PathExprSegment →
        PathIdentSegment ( :: GenericArgs )?

    PathIdentSegment →
        IDENTIFIER | super | self | Self | crate | $crate

    GenericArgs →
          < GenericArgList? >
        | ( TypeList? ) ( -> TypeNoBounds )?

    GenericArgList →
        ( GenericArg , )* GenericArg ,?

    TypeList →
        ( Type , )* Type ,?

    GenericArg →
        Lifetime | Type | GenericArgsConst | GenericArgsBinding | GenericArgsBounds

    GenericArgsConst →
          BlockExpression
        | LiteralExpression
        | - LiteralExpression
        | SimplePathSegment

    GenericArgsBinding →
        TypePathSegment = Type

    GenericArgsBounds →
        TypePathSegment : Bounds?

    [paths.expr.intro]

    مسیرهای درون عبارت اجازه می‌دهند مسیرهای دارای آرگومان عمومی مشخص شوند. این مسیرها در جایگاه‌های گوناگون عبارت‌ها و الگوها به کار می‌روند.

    [paths.expr.turbofish]

    برای جلوگیری از ابهام با عملگر کوچک‌تر از، پیش از < بازِ آرگومان‌های عمومی وجود توکن :: الزامی است. این شکل در گفتار نحو «توربوفیش» نامیده می‌شود.

    (0..10).collect::<Vec<_>>();
Vec::<u8>::with_capacity(1024);
    [paths.expr.argument-order]

    ترتیب آرگومان‌های عمومی به آرگومان‌های طول‌عمر، سپس آرگومان‌های نوع، پس از آن آرگومان‌های ثابت و در پایان قیدهای برابری محدود است.

    [paths.expr.complex-const-params]

    آرگومان‌های ثابت باید درون آکولاد قرار گیرند، مگر آنکه یک لفظ، ثابت استنباط‌شده یا مسیر تک‌قطعه‌ای باشند. یک ثابت استنباط‌شده نباید درون آکولاد قرار گیرد.

    mod m {
    pub const C: usize = 1;
}
const C: usize = m::C;
fn f<const N: usize>() -> [u8; N] { [0; N] }

let _ = f::<1>(); // Literal.
let _: [_; 1] = f::<_>(); // Inferred const.
let _: [_; 1] = f::<(((_)))>(); // Inferred const.
let _ = f::<C>(); // Single segment path.
let _ = f::<{ m::C }>(); // Multi-segment path must be braced.
    fn f<const N: usize>() -> [u8; N] { [0; _] }
let _: [_; 1] = f::<{ _ }>();
//                    ^ ERROR `_` not allowed here

    Note

    در فهرست آرگومان‌های عمومی، یک ثابت استنباط‌شده به‌صورت یک نوع استنباط‌شده تجزیه می‌شود، اما سپس از نظر معنایی گونه‌ای جداگانه از آرگومان ثابت عمومی در نظر گرفته می‌شود.

    [paths.expr.impl-trait-params]

    پارامترهای نوع ساختگی متناظر با نوع‌های impl Trait ضمنی هستند و نمی‌توان آن‌ها را صریحاً مشخص کرد.

    [paths.qualified]

    مسیرهای واجد شرایط

    [paths.qualified.syntax]

    Syntax
    QualifiedPathInExpression → QualifiedPathType ( :: PathExprSegment )+

    QualifiedPathType → < Type ( as TypePath )? >

    QualifiedPathInType → QualifiedPathType ( :: TypePathSegment )+

    [paths.qualified.intro]

    مسیرهای کاملاً واجد شرایط امکان رفع ابهام مسیر را برای پیاده‌سازی‌های صفت و مشخص کردن مسیرهای معیار فراهم می‌کنند. هنگام استفاده در مشخصات نوع، از نحو نوعِ آمده در ادامه پشتیبانی می‌کنند.

    struct S;
impl S {
    fn f() { println!("S"); }
}
trait T1 {
    fn f() { println!("T1 f"); }
}
impl T1 for S {}
trait T2 {
    fn f() { println!("T2 f"); }
}
impl T2 for S {}
S::f();  // Calls the inherent impl.
<S as T1>::f();  // Calls the T1 trait function.
<S as T2>::f();  // Calls the T2 trait function.
    [paths.type]

    مسیرها در نوع‌ها

    [paths.type.syntax]

    Syntax
    TypePath → ::? TypePathSegment ( :: TypePathSegment )*

    TypePathSegment → PathIdentSegment ( ::? GenericArgs )?

    [paths.type.intro]

    مسیرهای نوع درون تعریف نوع‌ها، کران‌های صفت و مسیرهای واجد شرایط به کار می‌روند.

    [paths.type.turbofish]

    هرچند توکن :: پیش از آرگومان‌های عمومی مجاز است، الزامی نیست؛ زیرا ابهامی مانند PathInExpression وجود ندارد.

    mod ops {
    pub struct Range<T> {f1: T}
    pub trait Index<T> {}
    pub struct Example<'a> {f1: &'a i32}
}
struct S;
impl ops::Index<ops::Range<usize>> for S { /*...*/ }
fn i<'a>() -> impl Iterator<Item = ops::Example<'a>> {
    // ...
   const EXAMPLE: Vec<ops::Example<'static>> = Vec::new();
   EXAMPLE.into_iter()
}
type G = std::boxed::Box<dyn std::ops::FnOnce(isize) -> isize>;
    [paths.qualifiers]

    تعیین‌کننده‌های مسیر

    [paths.qualifiers.intro]

    مسیرها می‌توانند با تعیین‌کننده‌های ابتدایی گوناگونی نوشته شوند تا معنای شیوهٔ تفکیکشان تغییر کند.

    Note

    نکته اعلان‌های use برای self، super، crate و $crate رفتارها و محدودیت‌های بیشتری دارند.

    [paths.qualifiers.global-root]

    ::

    [paths.qualifiers.global-root.intro]

    مسیرهایی که با :: آغاز می‌شوند مسیرهای سراسری به شمار می‌آیند و تفکیک قطعه‌های مسیر از جایی آغاز می‌شود که بسته به ویرایش متفاوت است. هر شناسه در مسیر باید به یک قلم تفکیک شود.

    [paths.qualifiers.global-root.edition2018]

    تفاوت‌های ویرایش 2018 در ویرایش ۲۰۱۵، شناسه‌ها از «ریشهٔ کریت» (معادل crate:: در ویرایش ۲۰۱۸) تفکیک می‌شوند. ریشهٔ کریت شامل اقلام گوناگونی است، از جمله کریت‌های خارجی، کریت‌های پیش‌فرض مانند std یا core و اقلام سطح بالای کریت، از جمله واردسازی‌های use.

    از ویرایش ۲۰۱۸ به بعد، مسیرهای آغازشده با :: از کریت‌های موجود در پیش‌درآمد خارجی تفکیک می‌شوند؛ یعنی پس از آن‌ها باید نام یک کریت بیاید.

    pub fn foo() {
    // In the 2018 edition, this accesses `std` via the extern prelude.
    // In the 2015 edition, this accesses `std` via the crate root.
    let now = ::std::time::Instant::now();
    println!("{:?}", now);
}
    // 2015 Edition
mod a {
    pub fn foo() {}
}
mod b {
    pub fn foo() {
        ::a::foo(); // call `a`'s foo function
        // In Rust 2018, `::a` would be interpreted as the crate `a`.
    }
}
fn main() {}
    [paths.qualifiers.mod-self]

    self

    [paths.qualifiers.mod-self.intro]

    self مسیر را نسبت به ماژول جاری تفکیک می‌کند.

    [paths.qualifiers.mod-self.restriction]

    self فقط می‌تواند نخستین قطعهٔ مسیر، بدون :: پیش از آن، یا آخرین قطعهٔ مسیر، با :: پیش از آن، باشد.

    [paths.qualifiers.mod-self.trailing]

    وقتی self آخرین قطعهٔ مسیر باشد، به موجودیتی اشاره می‌کند که قطعهٔ پیش از آن نام‌گذاری کرده است. مسیر پیشین باید به یک ماژول، برشمردنی یا صفت تفکیک شود.

    mod m {
    pub enum E { V1 }
    pub trait Tr {}
    pub(in crate::m::self) fn g() {} // OK: Modules can be parents of `self`.
}
type Ty = m::E::self; // OK: Enumerations can be parents of `self`.
fn f<T: m::Tr::self>() {} // OK: Traits can be parents of `self`.
fn main() { let _: Ty = m::E::V1; }
    struct S;
type Ty = S::self; // ERROR: Structs cannot be parents of `self`.
fn main() {}

    Note

    نکته برای قواعد بیشتر دربارهٔ self در اعلان‌های use، بخش items.use.self را ببینید.

    [paths.qualifiers.self-pat]

    در بدنهٔ متد، مسیری که تنها از یک قطعهٔ self تشکیل شده باشد به پارامتر self متد تفکیک می‌شود.

    fn foo() {}
fn bar() {
    self::foo();
}
struct S(bool);
impl S {
  fn baz(self) {
        self.0;
    }
}
fn main() {}
    [paths.qualifiers.type-self]

    Self

    [paths.qualifiers.type-self.intro]

    Self با حرف بزرگ «S» برای ارجاع به نوع جاریِ در حال پیاده‌سازی یا تعریف به کار می‌رود. از آن می‌توان در موقعیت‌های زیر استفاده کرد:

    [paths.qualifiers.type-self.trait]
    • در تعریف یک صفت، به نوعی اشاره می‌کند که صفت را پیاده‌سازی می‌کند.
    [paths.qualifiers.type-self.impl]
    • در یک پیاده‌سازی، به نوع در حال پیاده‌سازی اشاره می‌کند. هنگام پیاده‌سازی یک ساختار تاپلی یا واحد، به سازنده در فضای نام مقدار نیز اشاره می‌کند.
    [paths.qualifiers.type-self.type]
    • در تعریف یک ساختار، برشمردنی یا اجتماع، به نوع در حال تعریف اشاره می‌کند. تعریف اجازه ندارد بی‌نهایت بازگشتی باشد و باید یک سطح غیرمستقیم‌سازی وجود داشته باشد.
    [paths.qualifiers.type-self.scope]

    دامنهٔ Self رفتاری مشابه یک پارامتر عمومی دارد؛ برای جزئیات بیشتر بخش دامنهٔ Self را ببینید.

    [paths.qualifiers.type-self.allowed-positions]

    Self فقط می‌تواند نخستین قطعه و بدون :: پیش از آن باشد.

    [paths.qualifiers.type-self.no-generics]

    مسیر Self نمی‌تواند آرگومان عمومی داشته باشد، مانند Self::<i32>.

    trait T {
    type Item;
    const C: i32;
    // `Self` will be whatever type that implements `T`.
    fn new() -> Self;
    // `Self::Item` will be the type alias in the implementation.
    fn f(&self) -> Self::Item;
}
struct S;
impl T for S {
    type Item = i32;
    const C: i32 = 9;
    fn new() -> Self {           // `Self` is the type `S`.
        S
    }
    fn f(&self) -> Self::Item {  // `Self::Item` is the type `i32`.
        Self::C                  // `Self::C` is the constant value `9`.
    }
}

// `Self` is in scope within the generics of a trait definition,
// to refer to the type being defined.
trait Add<Rhs = Self> {
    type Output;
    // `Self` can also reference associated items of the
    // type being implemented.
    fn add(self, rhs: Rhs) -> Self::Output;
}

struct NonEmptyList<T> {
    head: T,
    // A struct can reference itself (as long as it is not
    // infinitely recursive).
    tail: Option<Box<Self>>,
}
    [paths.qualifiers.super]

    super

    [paths.qualifiers.super.intro]

    super در یک مسیر به ماژول والد تفکیک می‌شود.

    [paths.qualifiers.super.allowed-positions]

    این کلیدواژه فقط در قطعه‌های ابتدایی مسیر و احتمالاً پس از یک قطعهٔ self ابتدایی قابل استفاده است.

    mod a {
    pub fn foo() {}
}
mod b {
    pub fn foo() {
        super::a::foo(); // call a's foo function
    }
}
fn main() {}
    [paths.qualifiers.super.repetition]

    برای ارجاع به ماژول‌های نیا می‌توان super را پس از نخستین super یا self چند بار تکرار کرد.

    mod a {
    fn foo() {}

    mod b {
        mod c {
            fn foo() {
                super::super::foo(); // call a's foo function
                self::super::super::foo(); // call a's foo function
            }
        }
    }
}
fn main() {}
    [paths.qualifiers.crate]

    crate

    [paths.qualifiers.crate.intro]

    crate مسیر را نسبت به کریت جاری تفکیک می‌کند.

    [paths.qualifiers.crate.allowed-positions]

    crate فقط می‌تواند نخستین قطعه و بدون :: پیش از آن باشد.

    fn foo() {}
mod a {
    fn bar() {
        crate::foo();
    }
}
fn main() {}
    [paths.qualifiers.macro-crate]

    $crate

    [paths.qualifiers.macro-crate.allowed-positions]

    $crate فقط در رونویس‌های ماکرو استفاده می‌شود و تنها می‌تواند نخستین قطعه و بدون :: پیش از آن باشد.

    [paths.qualifiers.macro-crate.hygiene]

    $crate به مسیری گسترش می‌یابد که از سطح بالای کریتِ تعریف‌کنندهٔ ماکرو به اقلام دسترسی می‌یابد، مستقل از اینکه ماکرو در کدام کریت فراخوانی شود.

    pub fn increment(x: u32) -> u32 {
    x + 1
}

#[macro_export]
macro_rules! inc {
    ($x:expr) => ( $crate::increment($x) )
}
fn main() { }
    [paths.canonical]

    مسیرهای معیار

    [paths.canonical.intro]

    هر قلم تعریف‌شده در یک ماژول یا پیاده‌سازی، مسیر معیاری دارد که با محل تعریف آن درون کریت متناظر است.

    [paths.canonical.alias]

    همهٔ مسیرهای دیگر به این اقلام نام مستعار هستند.

    [paths.canonical.def]

    مسیر معیار به‌صورت یک پیشوند مسیر تعریف می‌شود که قطعهٔ مسیر تعریف‌شده به‌وسیلهٔ خود قلم به انتهای آن افزوده شده است.

    [paths.canonical.non-canonical]

    پیاده‌سازی‌ها و اعلان‌های use مسیر معیار ندارند، هرچند اقلامی که پیاده‌سازی‌ها تعریف می‌کنند دارای مسیر معیارند. اقلام تعریف‌شده در عبارت‌های بلوک مسیر معیار ندارند. اقلام تعریف‌شده در ماژولی که مسیر معیار ندارد نیز مسیر معیار ندارند. اقلام وابستهٔ تعریف‌شده در پیاده‌سازی‌ای که به قلمی بدون مسیر معیار ارجاع می‌دهد ــ برای نمونه به‌عنوان نوع پیاده‌ساز، صفت در حال پیاده‌سازی، یک پارامتر نوع یا کرانی روی پارامتر نوع ــ مسیر معیار ندارند.

    [paths.canonical.module-prefix]

    پیشوند مسیر ماژول‌ها همان مسیر معیار آن ماژول است.

    [paths.canonical.bare-impl-prefix]

    برای پیاده‌سازی‌های بدون صفت، پیشوند مسیر همان مسیر معیار قلم در حال پیاده‌سازی است که درون براکت‌های زاویه‌دار (<>) قرار گرفته است.

    [paths.canonical.trait-impl-prefix]

    برای پیاده‌سازی‌های صفت، پیشوند مسیر همان مسیر معیار قلم در حال پیاده‌سازی، سپس as و پس از آن مسیر معیار صفت است که همگی درون براکت‌های زاویه‌دار (<>) قرار گرفته‌اند.

    [paths.canonical.local-canonical-path]

    مسیر معیار تنها درون یک کریت مشخص معنا دارد. فضای نام سراسری میان کریت‌ها وجود ندارد؛ مسیر معیار یک قلم فقط آن را درون کریت شناسایی می‌کند.

    // Comments show the canonical path of the item.

mod a { // crate::a
    pub struct Struct; // crate::a::Struct

    pub trait Trait { // crate::a::Trait
        fn f(&self); // crate::a::Trait::f
    }

    impl Trait for Struct {
        fn f(&self) {} // <crate::a::Struct as crate::a::Trait>::f
    }

    impl Struct {
        fn g(&self) {} // <crate::a::Struct>::g
    }
}

mod without { // crate::without
    fn canonicals() { // crate::without::canonicals
        struct OtherStruct; // None

        trait OtherTrait { // None
            fn g(&self); // None
        }

        impl OtherTrait for OtherStruct {
            fn g(&self) {} // None
        }

        impl OtherTrait for crate::a::Struct {
            fn g(&self) {} // None
        }

        impl crate::a::Trait for OtherStruct {
            fn f(&self) {} // None
        }
    }
}

fn main() {}
