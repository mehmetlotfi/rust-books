    [asm]

    اسمبلی درون‌خطی

    [asm.intro]

    پشتیبانی از اسمبلی درون‌خطی از طریق ماکروهای asm!، naked_asm! و global_asm! فراهم می‌شود. با آن‌ها می‌توان کد اسمبلیِ دست‌نویس را در خروجی اسمبلی تولیدشده توسط کامپایلر گنجاند.

    [asm.stable-targets]

    پشتیبانی از اسمبلی درون‌خطی برای معماری‌های زیر پایدار است:

    • x86 و x86-64
    • ARM
    • AArch64 و Arm64EC
    • RISC-V
    • LoongArch
    • s390x
    • PowerPC و PowerPC64

    اگر ماکروی اسمبلی روی هدفی استفاده شود که از آن پشتیبانی نمی‌کند، کامپایلر خطا می‌دهد.

    [asm.example]

    مثال

    #[cfg(target_arch = "x86_64")] {
use std::arch::asm;

// Multiply x by 6 using shifts and adds
let mut x: u64 = 4;
unsafe {
    asm!(
        "mov {tmp}, {x}",
        "shl {tmp}, 1",
        "shl {x}, 2",
        "add {x}, {tmp}",
        x = inout(reg) x,
        tmp = out(reg) _,
    );
}
assert_eq!(x, 4 * 6);
}
    [asm.syntax]

    نحو

    دستور زبان زیر آرگومان‌هایی را مشخص می‌کند که می‌توان به ماکروهای asm!، global_asm! و naked_asm! داد.

    Syntax
    AsmArgs → AsmAttrFormatString ( , AsmAttrFormatString )* ( , AsmAttrOperand )* ,?

    FormatString → STRING_LITERAL | RAW_STRING_LITERAL | MacroInvocation

    AsmAttrFormatString → ( OuterAttribute )* FormatString

    AsmOperand →
          ClobberAbi
        | AsmOptions
        | RegOperand

    AsmAttrOperand → ( OuterAttribute )* AsmOperand

    ClobberAbi → clobber_abi ( Abi ( , Abi )* ,? )

    AsmOptions →
        options ( ( AsmOption ( , AsmOption )* ,? )? )

    AsmOption →
          pure
        | nomem
        | readonly
        | preserves_flags
        | noreturn
        | nostack
        | att_syntax
        | raw

    RegOperand → ( ParamName = )?
        (
              DirSpec ( RegSpec ) Expression
            | DualDirSpec ( RegSpec ) DualDirSpecExpression
            | sym PathExpression
            | const Expression
            | label { Statements? }
        )

    ParamName → IDENTIFIER_OR_KEYWORD | RAW_IDENTIFIER

    DualDirSpecExpression →
          Expression
        | Expression => Expression

    RegSpec → RegisterClass | ExplicitRegister

    RegisterClass → IDENTIFIER_OR_KEYWORD

    ExplicitRegister → STRING_LITERAL

    DirSpec →
          in
        | out
        | lateout

    DualDirSpec →
          inout
        | inlateout

    [asm.scope]

    دامنه

    [asm.scope.intro]

    اسمبلی درون‌خطی را می‌توان به یکی از سه روش زیر استفاده کرد.

    [asm.scope.asm]

    با ماکروی asm!، کد اسمبلی در دامنهٔ یک تابع تولید و با کد اسمبلیِ تولیدشده توسط کامپایلر برای آن تابع ادغام می‌شود. این کد باید از قواعد سخت‌گیرانه‌ای پیروی کند تا رفتار تعریف‌نشده رخ ندهد. توجه کنید که در برخی موارد کامپایلر ممکن است کد اسمبلی را به‌صورت تابعی جداگانه تولید کند و فراخوانی آن را بسازد.

    #[cfg(target_arch = "x86_64")] {
unsafe { core::arch::asm!("/* {} */", in(reg) 0); }
}
    [asm.scope.naked_asm]

    با ماکروی naked_asm!، کد اسمبلی در دامنهٔ یک تابع تولید می‌شود و تمام کد اسمبلیِ آن تابع را تشکیل می‌دهد. ماکروی naked_asm! فقط در تابع‌های برهنه مجاز است.

    #[cfg(target_arch = "x86_64")] {
#[unsafe(naked)]
extern "C" fn wrapper() {
core::arch::naked_asm!("/* {} */", const 0);
}
}
    [asm.scope.global_asm]

    با ماکروی global_asm!، کد اسمبلی در دامنهٔ سراسری و خارج از تابع تولید می‌شود. با این روش می‌توان تابع‌های کامل را به‌صورت دستی با اسمبلی نوشت؛ همچنین معمولاً آزادی بسیار بیشتری برای استفاده از رجیسترهای دلخواه و دستورهای راهنمای اسمبلر وجود دارد.

    fn main() {}
#[cfg(target_arch = "x86_64")]
core::arch::global_asm!("/* {} */", const 0);
    [asm.ts-args]

    آرگومان‌های رشتهٔ قالب

    [asm.ts-args.syntax]

    قالب اسمبلر از همان نحو رشته‌های قالب‌بندی استفاده می‌کند؛ یعنی جای‌نگهدارها با آکولاد مشخص می‌شوند.

    [asm.ts-args.order]

    آرگومان‌های متناظر به‌ترتیب، با اندیس، یا با نام در دسترس‌اند.

    #[cfg(target_arch = "x86_64")] {
let x: i64;
let y: i64;
let z: i64;
// This
unsafe { core::arch::asm!("mov {}, {}", out(reg) x, in(reg) 5); }
// ... this
unsafe { core::arch::asm!("mov {0}, {1}", out(reg) y, in(reg) 5); }
// ... and this
unsafe { core::arch::asm!("mov {out}, {in}", out = out(reg) z, in = in(reg) 5); }
// all have the same behavior
assert_eq!(x, y);
assert_eq!(y, z);
}
    [asm.ts-args.no-implicit]

    بااین‌حال، آرگومان‌های نام‌دار ضمنی که در RFC شمارهٔ ۲۷۹۵ معرفی شدند، پشتیبانی نمی‌شوند.

    #[cfg(target_arch = "x86_64")] {
let x = 5;
// We can't refer to `x` from the scope directly, we need an operand like `in(reg) x`
unsafe { core::arch::asm!("/* {x} */"); } // ERROR: no argument named x
}
#[cfg(not(target_arch = "x86_64"))] core::compile_error!("Test not supported on this arch");
    [asm.ts-args.one-or-more]

    یک فراخوانی asm! می‌تواند یک یا چند آرگومان رشتهٔ قالب (template string) داشته باشد؛ asm! با چند آرگومان رشتهٔ قالب طوری در نظر گرفته می‌شود که گویی همهٔ رشته‌ها با یک \n میان آن‌ها به هم متصل شده‌اند. کاربرد مورد انتظار این است که هر آرگومان رشتهٔ قالب با یک خط از کد اسمبلی متناظر باشد.

    #[cfg(target_arch = "x86_64")] {
let x: i64;
let y: i64;
// We can separate multiple strings as if they were written together
unsafe { core::arch::asm!("mov eax, 5", "mov ecx, eax", out("rax") x, out("rcx") y); }
assert_eq!(x, y);
}
    [asm.ts-args.before-other-args]

    تمام آرگومان‌های رشتهٔ قالب باید پیش از آرگومان‌های دیگر بیایند.

    let x = 5;
#[cfg(target_arch = "x86_64")] {
// The template strings need to appear first in the asm invocation
unsafe { core::arch::asm!("/* {x} */", x = const 5, "ud2"); } // ERROR: unexpected token
}
#[cfg(not(target_arch = "x86_64"))] core::compile_error!("Test not supported on this arch");
    [asm.ts-args.positional-first]

    همانند رشته‌های قالب‌بندی، آرگومان‌های موقعیتی باید پیش از آرگومان‌های نام‌دار و عملوندهای رجیستر صریح بیایند.

    #[cfg(target_arch = "x86_64")] {
// Named operands need to come after positional ones
unsafe { core::arch::asm!("/* {x} {} */", x = const 5, in(reg) 5); }
// ERROR: positional arguments cannot follow named arguments or explicit register arguments
}
#[cfg(not(target_arch = "x86_64"))] core::compile_error!("Test not supported on this arch");
    #[cfg(target_arch = "x86_64")] {
// We also can't put explicit registers before positional operands
unsafe { core::arch::asm!("/* {} */", in("eax") 0, in(reg) 5); }
// ERROR: positional arguments cannot follow named arguments or explicit register arguments
}
#[cfg(not(target_arch = "x86_64"))] core::compile_error!("Test not supported on this arch");
    [asm.ts-args.register-operands]

    جای‌نگهدارهای رشتهٔ قالب نمی‌توانند از عملوندهای رجیستر صریح استفاده کنند.

    #[cfg(target_arch = "x86_64")] {
// Explicit register operands don't get substituted, use `eax` explicitly in the string
unsafe { core::arch::asm!("/* {} */", in("eax") 5); }
// ERROR: invalid reference to argument at index 0
}
#[cfg(not(target_arch = "x86_64"))] core::compile_error!("Test not supported on this arch");
    [asm.ts-args.at-least-once]

    همهٔ عملوندهای نام‌دار و موقعیتیِ دیگر باید دست‌کم یک بار در رشتهٔ قالب ظاهر شوند؛ در غیر این صورت خطای کامپایل رخ می‌دهد.

    #[cfg(target_arch = "x86_64")] {
// We have to name all of the operands in the format string
unsafe { core::arch::asm!("", in(reg) 5, x = const 5); }
// ERROR: multiple unused asm arguments
}
#[cfg(not(target_arch = "x86_64"))] core::compile_error!("Test not supported on this arch");
    [asm.ts-args.opaque]

    نحو دقیق کد اسمبلی به هدف بستگی دارد و برای کامپایلر نامشخص است؛ تنها استثنا شیوهٔ جای‌گذاری عملوندها در رشتهٔ قالب برای ساختن کدی است که به اسمبلر داده می‌شود.

    [asm.ts-args.llvm-syntax]

    در حال حاضر، همهٔ هدف‌های پشتیبانی‌شده از نحو کد اسمبلیِ اسمبلر داخلی LLVM پیروی می‌کنند که معمولاً با نحو اسمبلر GNU یا GAS مطابقت دارد. در x86، حالت .intel_syntax noprefix از GAS به‌صورت پیش‌فرض به کار می‌رود. در ARM، حالت .syntax unified استفاده می‌شود. این هدف‌ها محدودیت دیگری نیز بر کد اسمبلی اعمال می‌کنند: هر وضعیت اسمبلر، مانند بخش جاری که با .section تغییر می‌کند، باید در پایان رشتهٔ اسمبلی به مقدار اولیه بازگردد. کدی که با نحو GAS سازگار نباشد، رفتاری وابسته به اسمبلر خواهد داشت. محدودیت‌های بیشتر دربارهٔ دستورهای راهنمای مورد استفاده در اسمبلی درون‌خطی در بخش پشتیبانی از دستورهای راهنما آمده است.

    [asm.attributes]

    ویژگی‌ها

    [asm.attributes.supported-attributes]

    از نظر معنایی، روی رشته‌های قالب و عملوندهای اسمبلی درون‌خطی فقط ویژگی‌های cfg و cfg_attr پذیرفته می‌شوند. ویژگی‌های دیگر تجزیه می‌شوند، اما هنگام بسط ماکروی اسمبلی رد می‌شوند.

    fn main() {}
#[cfg(target_arch = "x86_64")]
core::arch::global_asm!(
    #[cfg(not(panic = "abort"))]
    ".cfi_startproc",
    // ...
    "ret",
    #[cfg(not(panic = "abort"))]
    ".cfi_endproc",
);

    Note

    نکته در rustc، ماکروهای اسمبلی رسیدگی به این ویژگی‌ها را جدا از سامانهٔ معمولِ رسیدگی به ویژگی‌های مشابه در زبان پیاده‌سازی می‌کنند. محدود بودن انواع ویژگی‌های پشتیبانی‌شده از همین موضوع ناشی می‌شود و ممکن است تفاوت‌های رفتاری ظریفی نیز ایجاد کند.

    [asm.attributes.starts-with-template]

    از نظر نحوی، باید پیش از نخستین عملوند دست‌کم یک رشتهٔ قالب وجود داشته باشد.

    // This is rejected because `a = out(reg) x` does not parse as a
// template string.
core::arch::asm!(
    #[cfg(false)]
    a = out(reg) x, // ERROR.
    "",
);
    [asm.operand-type]

    نوع عملوند

    [asm.operand-type.supported-operands]

    چند نوع عملوند پشتیبانی می‌شود:

    [asm.operand-type.supported-operands-in]
    • in(<reg>) <expr>
      • <reg> می‌تواند یک ردهٔ رجیستر یا رجیستری صریح را مشخص کند. نام رجیستر تخصیص‌یافته در رشتهٔ قالب اسمبلی جای‌گذاری می‌شود.
      • رجیستر تخصیص‌یافته در آغاز کد اسمبلی حاوی مقدار <expr> خواهد بود.
      • رجیستر تخصیص‌یافته باید در پایان کد اسمبلی همان مقدار را داشته باشد، مگر آنکه یک lateout به همان رجیستر تخصیص داده شده باشد.
    #[cfg(target_arch = "x86_64")] {
// ``in` can be used to pass values into inline assembly...
unsafe { core::arch::asm!("/* {} */", in(reg) 5); }
}

    Note

    نکته اگر نوع مقدار از رجیستر کوچک‌تر باشد، مقدار بیت‌های بالایی به سکو بستگی دارد. برخی هدف‌ها بیت‌های بالایی را صفر می‌کنند و برخی دیگر آن‌ها را دست‌نخورده می‌گذارند.

    [asm.operand-type.supported-operands-out]
    • out(<reg>) <expr>
      • <reg> می‌تواند یک ردهٔ رجیستر یا رجیستری صریح را مشخص کند. نام رجیستر تخصیص‌یافته در رشتهٔ قالب اسمبلی جای‌گذاری می‌شود.
      • رجیستر تخصیص‌یافته در آغاز کد اسمبلی حاوی مقداری نامعین خواهد بود.
      • <expr> باید یک عبارت مکان باشد که ممکن است هنوز مقداردهی نشده باشد. در پایان کد اسمبلی، محتوای رجیستر تخصیص‌یافته در آن نوشته می‌شود.
      • می‌توان به‌جای عبارت، زیرخط (_) قرار داد تا محتوای رجیستر در پایان کد اسمبلی کنار گذاشته شود؛ این حالت عملاً تغییرِ بدون حفظ مقدار رجیستر را اعلام می‌کند.
    #[cfg(target_arch = "x86_64")] {
let x: i64;
// and `out` can be used to pass values back to rust.
unsafe { core::arch::asm!("/* {} */", out(reg) x); }
}
    [asm.operand-type.supported-operands-lateout]
    • lateout(<reg>) <expr>
      • مانند out است، با این تفاوت که تخصیص‌دهندهٔ رجیستر می‌تواند از رجیستری که به یک in تخصیص یافته است دوباره استفاده کند.
      • فقط پس از خواندن همهٔ ورودی‌ها باید در رجیستر بنویسید؛ وگرنه ممکن است مقدار یک ورودی را از بین ببرید.
    #[cfg(target_arch = "x86_64")] {
let x: i64;
// `lateout` is the same as `out`
// but the compiler knows we don't care about the value of any inputs by the
// time we overwrite it.
unsafe { core::arch::asm!("mov {}, 5", lateout(reg) x); }
assert_eq!(x, 5)
}
    [asm.operand-type.supported-operands-inout]
    • inout(<reg>) <expr>
      • <reg> می‌تواند یک ردهٔ رجیستر یا رجیستری صریح را مشخص کند. نام رجیستر تخصیص‌یافته در رشتهٔ قالب اسمبلی جای‌گذاری می‌شود.
      • رجیستر تخصیص‌یافته در آغاز کد اسمبلی حاوی مقدار <expr> خواهد بود.
      • <expr> باید یک عبارت مکانِ تغییرپذیر و مقداردهی‌شده باشد. در پایان کد اسمبلی، محتوای رجیستر تخصیص‌یافته در آن نوشته می‌شود.
    #[cfg(target_arch = "x86_64")] {
let mut x: i64 = 4;
// `inout` can be used to modify values in-register
unsafe { core::arch::asm!("inc {}", inout(reg) x); }
assert_eq!(x, 5);
}
    [asm.operand-type.supported-operands-inout-arrow]
    • inout(<reg>) <in expr> => <out expr>
      • مانند inout است، با این تفاوت که مقدار اولیهٔ رجیستر از مقدار <in expr> گرفته می‌شود.
      • <out expr> باید یک عبارت مکان باشد که ممکن است هنوز مقداردهی نشده باشد. در پایان کد اسمبلی، محتوای رجیستر تخصیص‌یافته در آن نوشته می‌شود.
      • می‌توان برای <out expr> به‌جای عبارت، زیرخط (_) قرار داد تا محتوای رجیستر در پایان کد اسمبلی کنار گذاشته شود؛ این حالت عملاً تغییرِ بدون حفظ مقدار رجیستر را اعلام می‌کند.
      • نوع‌های <in expr> و <out expr> می‌توانند متفاوت باشند.
    #[cfg(target_arch = "x86_64")] {
let x: i64;
// `inout` can also move values to different places
unsafe { core::arch::asm!("inc {}", inout(reg) 4u64=>x); }
assert_eq!(x, 5);
}
    [asm.operand-type.supported-operands-inlateout]
    • inlateout(<reg>) <expr> / inlateout(<reg>) <in expr> => <out expr>
      • مانند inout است، با این تفاوت که تخصیص‌دهندهٔ رجیستر می‌تواند از رجیستر تخصیص‌یافته به یک in دوباره استفاده کند؛ این اتفاق ممکن است زمانی رخ دهد که کامپایلر بداند مقدار اولیهٔ in و inlateout یکسان است.
      • فقط پس از خواندن همهٔ ورودی‌ها باید در رجیستر بنویسید؛ وگرنه ممکن است مقدار یک ورودی را از بین ببرید.
    #[cfg(target_arch = "x86_64")] {
let mut x: i64 = 4;
// `inlateout` is `inout` using `lateout`
unsafe { core::arch::asm!("inc {}", inlateout(reg) x); }
assert_eq!(x, 5);
}
    [asm.operand-type.supported-operands-sym]
    • sym <path>
      • <path> باید به یک fn یا static اشاره کند.
      • نام نمادِ تبدیل‌شده‌ای که به آن آیتم اشاره دارد در رشتهٔ قالب اسمبلی جای‌گذاری می‌شود.
      • رشتهٔ جای‌گذاری‌شده هیچ اصلاح‌گری، مانند GOT، PLT یا اطلاعات بازجای‌گذاری، ندارد.
      • <path> می‌تواند به یک متغیر ایستای #[thread_local] اشاره کند. در این حالت کد اسمبلی می‌تواند نماد را با بازجای‌گذاری‌هایی مانند @plt یا @TPOFF ترکیب کند تا داده‌های محلیِ نخ را بخواند.
    #[cfg(target_arch = "x86_64")] {
extern "C" fn foo() {
    println!("Hello from inline assembly")
}
// `sym` can be used to refer to a function (even if it doesn't have an
// external name we can directly write)
unsafe { core::arch::asm!("call {}", sym foo, clobber_abi("C")); }
}
    [asm.operand-type.supported-operands-const]
    • const <expr>
      • <expr> باید یک عبارت ثابتِ صحیح باشد. این عبارت از همان قواعد بلوک‌های const درون‌خطی پیروی می‌کند.
      • نوع عبارت می‌تواند هر نوع صحیحی باشد، اما همانند لیترال‌های صحیح، نوع پیش‌فرض آن i32 است.
      • مقدار عبارت به رشته تبدیل و مستقیماً در رشتهٔ قالب اسمبلی جای‌گذاری می‌شود.
    #[cfg(target_arch = "x86_64")] {
// swizzle [0, 1, 2, 3] => [3, 2, 0, 1]
const SHUFFLE: u8 = 0b01_00_10_11;
let x: core::arch::x86_64::__m128 = unsafe { core::mem::transmute([0u32, 1u32, 2u32, 3u32]) };
let y: core::arch::x86_64::__m128;
// Pass a constant value into an instruction that expects an immediate like `pshufd`
unsafe {
    core::arch::asm!("pshufd {xmm}, {xmm}, {shuffle}",
        xmm = inlateout(xmm_reg) x=>y,
        shuffle = const SHUFFLE
    );
}
let y: [u32; 4] = unsafe { core::mem::transmute(y) };
assert_eq!(y, [3, 2, 0, 1]);
}
    [asm.operand-type.supported-operands-label]
    • label <block>
      • نشانی بلوک در رشتهٔ قالب اسمبلی جای‌گذاری می‌شود. کد اسمبلی می‌تواند به این نشانی بپرد.
      • در هدف‌هایی که میان پرش مستقیم و غیرمستقیم تفاوت قائل‌اند، مانند x86-64 با cf-protection فعال، کد اسمبلی نباید به‌صورت غیرمستقیم به این نشانی بپرد.
      • پس از اجرای بلوک، عبارت asm! بازمی‌گردد.
      • نوع بلوک باید واحد یا !، یعنی نوع هرگز، باشد.
      • این بلوک یک بافت ایمنی تازه آغاز می‌کند؛ عملیات ناامن داخل بلوک label باید در یک بلوک داخلی unsafe قرار بگیرند، حتی اگر کل عبارت asm! از قبل در unsafe باشد.
    #[cfg(target_arch = "x86_64")]
unsafe {
    core::arch::asm!("jmp {}", label {
        println!("Hello from inline assembly label");
    });
}
    [asm.operand-type.left-to-right]

    عبارت‌های عملوند، درست مانند آرگومان‌های فراخوانی تابع، از چپ به راست ارزیابی می‌شوند. پس از اجرای asm!، خروجی‌ها نیز از چپ به راست نوشته می‌شوند. اگر دو خروجی به یک محل اشاره کنند، این ترتیب مهم است: آن محل در پایان مقدار راست‌ترین خروجی را خواهد داشت.

    #[cfg(target_arch = "x86_64")] {
let mut y: i64;
// y gets its value from the second output, rather than the first
unsafe { core::arch::asm!("mov {}, 0", "mov {}, 1", out(reg) y, out(reg) y); }
assert_eq!(y, 1);
}
    [asm.operand-type.naked_asm-restriction]

    چون naked_asm! بدنهٔ کامل یک تابع را تعریف می‌کند و کامپایلر نمی‌تواند کد اضافی برای رسیدگی به عملوندها تولید کند، فقط عملوندهای sym و const در آن مجازند.

    [asm.operand-type.global_asm-restriction]

    چون global_asm! خارج از تابع قرار دارد، فقط عملوندهای sym و const در آن مجازند.

    fn main() {}
// register operands aren't allowed, since we aren't in a function
#[cfg(target_arch = "x86_64")]
core::arch::global_asm!("", in(reg) 5);
// ERROR: the `in` operand cannot be used with `global_asm!`
#[cfg(not(target_arch = "x86_64"))] core::compile_error!("Test not supported on this arch");
    fn main() {}
fn foo() {}

#[cfg(target_arch = "x86_64")]
// `const` and `sym` are both allowed, however
core::arch::global_asm!("/* {} {} */", const 0, sym foo);

    عملوندهای رجیستر

    [asm.register-operands.register-or-class]

    عملوندهای ورودی و خروجی را می‌توان به‌صورت یک رجیستر صریح یا یک ردهٔ رجیستر مشخص کرد تا تخصیص‌دهنده از میان آن‌ها رجیستری را انتخاب کند. رجیسترهای صریح با لیترال رشته، مانند "eax"، و رده‌های رجیستر با شناسه، مانند reg، مشخص می‌شوند.

    #[cfg(target_arch = "x86_64")] {
let mut y: i64;
// We can name both `reg`, or an explicit register like `eax` to get an
// integer register
unsafe { core::arch::asm!("mov eax, {:e}", in(reg) 5, lateout("eax") y); }
assert_eq!(y, 5);
}
    [asm.register-operands.equivalence-to-base-register]

    توجه کنید که در رجیسترهای صریح، نام‌های مستعار رجیستر، مانند r14 و lr در ARM، و نماهای کوچک‌تر یک رجیستر، مانند eax و rax، با رجیستر پایه معادل در نظر گرفته می‌شوند.

    [asm.register-operands.error-two-operands]

    استفاده از یک رجیستر صریح یکسان برای دو عملوند ورودی یا دو عملوند خروجی، خطای زمان کامپایل است.

    #[cfg(target_arch = "x86_64")] {
// We can't name eax twice
unsafe { core::arch::asm!("", in("eax") 5, in("eax") 4); }
// ERROR: register `eax` conflicts with register `eax`
}
#[cfg(not(target_arch = "x86_64"))] core::compile_error!("Test not supported on this arch");
    #[cfg(target_arch = "x86_64")] {
// ... even using different aliases
unsafe { core::arch::asm!("", in("ax") 5, in("rax") 4); }
// ERROR: register `rax` conflicts with register `ax`
}
#[cfg(not(target_arch = "x86_64"))] core::compile_error!("Test not supported on this arch");
    [asm.register-operands.error-overlapping]

    همچنین استفاده از رجیسترهای هم‌پوشان، مانند رجیسترهای VFP در ARM، در عملوندهای ورودی یا در عملوندهای خروجی، خطای زمان کامپایل است.

    #[cfg(target_arch = "x86_64")] {

    • اعداد صحیح، علامت‌دار و بدون علامت
    • اعداد ممیز شناور
    • اشاره‌گرها، فقط از نوع باریک
    • اشاره‌گرهای تابع
    • بردارهای SIMD؛ یعنی ساختارهایی که با #[repr(simd)] تعریف شده‌اند و Copy را پیاده‌سازی می‌کنند. این دسته شامل نوع‌های برداریِ ویژهٔ معماری در std::arch، مانند __m128 در x86 یا int8x16_t در ARM نیز می‌شود.
    #[cfg(target_arch = "x86_64")] {

let func: extern "C" fn() = foo;
unsafe { core::arch::asm!("/* {} */", in(reg) func); }

unsafe { core::arch::asm!("/* {} */", in(xmm_reg) z); }
}

    اگر اندازهٔ مقدار از رجیستری که به آن تخصیص یافته کوچک‌تر باشد، بیت‌های بالایی آن رجیستر برای ورودی‌ها مقدار نامعین خواهند داشت و برای خروجی‌ها نادیده گرفته می‌شوند. تنها استثنا ردهٔ freg در RISC-V است که در آن، طبق الزام معماری RISC-V، مقادیر f32 در یک f64 با شیوهٔ بسته‌بندی NaN قرار می‌گیرند.

    #[cfg(target_arch = "x86_64")] {
let mut x: i64;
// Moving a 32-bit value into a 64-bit value, oops.
#[allow(asm_sub_register)] // rustc warns about this behavior
unsafe { core::arch::asm!("mov {}, {}", lateout(reg) x, in(reg) 4i32); }
// top 32-bits are indeterminate
assert_eq!(x, 4); // This assertion is not guaranteed to succeed
assert_eq!(x & 0xFFFFFFFF, 4); // However, this one will succeed
}

    وقتی برای یک عملوند inout عبارت‌های ورودی و خروجی جداگانه مشخص شوند، هر دو عبارت باید نوع یکسانی داشته باشند. تنها استثنا زمانی است که هر دو عملوند اشاره‌گر یا عدد صحیح باشند؛ در این حالت فقط یکسان بودن اندازهٔ آن‌ها الزامی است. دلیل این محدودیت آن است که تخصیص‌دهنده‌های رجیستر در LLVM و GCC گاهی نمی‌توانند عملوندهای وابسته با نوع‌های متفاوت را مدیریت کنند.

    #[cfg(target_arch = "x86_64")] {
// Pointers and integers can mix (as long as they are the same size)
let x: isize = 0;
let y: *mut ();
// Transmute an `isize` to a `*mut ()`, using inline assembly magic
unsafe { core::arch::asm!("/*{}*/", inout(reg) x=>y); }
assert!(y.is_null()); // Extremely roundabout way to make a null pointer
}
    #[cfg(target_arch = "x86_64")] {
let x: i32 = 0;
let y: f32;
// But we can't reinterpret an `i32` to an `f32` like this
unsafe { core::arch::asm!("/* {} */", inout(reg) x=>y); }
// ERROR: incompatible types for asm inout argument
}
#[cfg(not(target_arch = "x86_64"))] core::compile_error!("Test not supported on this arch");

    [asm.abi-clobbers]

    clobberهای ABI

    [asm.abi-clobbers.intro]

    با کلیدواژهٔ clobber_abi می‌توان مجموعهٔ پیش‌فرضی از رجیسترهای تخریب‌شونده را برای کد اسمبلی مشخص کرد. این کار به‌طور خودکار قیود لازم را برای فراخوانی تابعی با قرارداد فراخوانی مشخص درج می‌کند: اگر قرارداد فراخوانی مقدار یک رجیستر را در طول فراخوانی کاملاً حفظ نکند، lateout("...") _ به‌طور ضمنی به فهرست عملوندها افزوده می‌شود؛ به‌جای ... نام آن رجیستر قرار می‌گیرد.

    #[cfg(target_arch = "x86_64")] {
extern "C" fn foo() -> i32 { 0 }

let z: i32;
// To call a function, we have to inform the compiler that we're clobbering
// callee saved registers
unsafe { core::arch::asm!("call {}", sym foo, out("rax") z, clobber_abi("C")); }
assert_eq!(z, 0);
}
    [asm.abi-clobbers.many]

    می‌توان clobber_abi را به هر تعداد بار مشخص کرد. برای هر رجیستر متمایز در اجتماعِ همهٔ قراردادهای فراخوانیِ مشخص‌شده، اعلام تخریب درج می‌شود.

    #[cfg(target_arch = "x86_64")] {
extern "sysv64" fn foo() -> i32 { 0 }
extern "win64" fn bar(x: i32) -> i32 { x + 1 }

let z: i32;
// We can even call multiple functions with different conventions and
// different saved registers
unsafe {
    core::arch::asm!(
        "call {}",
        "mov ecx, eax",
        "call {}",
        sym foo,
        sym bar,
        out("rax") z,
        clobber_abi("sysv64"),
        clobber_abi("win64"),
    );
}
assert_eq!(z, 1);
}
    [asm.abi-clobbers.must-specify]

    هنگام استفاده از clobber_abi، کامپایلر خروجی‌های دارای ردهٔ رجیستر عمومی را نمی‌پذیرد؛ همهٔ خروجی‌ها باید رجیستری صریح مشخص کنند.

    #[cfg(target_arch = "x86_64")] {
extern "C" fn foo(x: i32) -> i32 { 0 }

let z: i32;
// explicit registers must be used to not accidentally overlap.
unsafe {
    core::arch::asm!(
        "mov eax, {:e}",
        "call {}",
        out(reg) z,
        sym foo,
        clobber_abi("C")
    );
    // ERROR: asm with `clobber_abi` must specify explicit registers for outputs
}
assert_eq!(z, 0);
}
#[cfg(not(target_arch = "x86_64"))] core::compile_error!("Test not supported on this arch");
    [asm.abi-clobbers.explicit-have-precedence]

    [asm.options]

    گزینه‌ها

    [asm.options.supported-options]

    پرچم‌ها برای کنترل بیشتر رفتار کد اسمبلی درون‌خطی به کار می‌روند. در حال حاضر گزینه‌های زیر تعریف شده‌اند:

    [asm.options.supported-options-pure]
    • pure: کد اسمبلی اثر جانبی ندارد، باید سرانجام بازگردد و خروجی‌هایش فقط به ورودی‌های مستقیم آن، یعنی خود مقادیر نه آنچه به آن اشاره می‌کنند، یا به مقادیر خوانده‌شده از حافظه وابسته است؛ مگر آنکه گزینهٔ nomem نیز فعال باشد. این گزینه به کامپایلر اجازه می‌دهد کد اسمبلی را کمتر از تعداد دفعات مشخص‌شده در برنامه اجرا کند، مثلاً با بیرون کشیدن آن از حلقه، یا اگر خروجی‌ها استفاده نشوند آن را کاملاً حذف کند. گزینهٔ pure باید همراه با nomem یا readonly باشد؛ وگرنه خطای زمان کامپایل رخ می‌دهد.
    #[cfg(target_arch = "x86_64")] {
let x: i32 = 0;
let z: i32;
// pure can be used to optimize by assuming the assembly has no side effects
unsafe { core::arch::asm!("inc {}", inout(reg) x => z, options(pure, nomem)); }
assert_eq!(z, 1);
}
    #[cfg(target_arch = "x86_64")] {
let x: i32 = 0;
let z: i32;
// Either nomem or readonly must be satisfied, to indicate whether or not
// memory is allowed to be read
unsafe { core::arch::asm!("inc {}", inout(reg) x => z, options(pure)); }
// ERROR: the `pure` option must be combined with either `nomem` or `readonly`
assert_eq!(z, 0);
}
#[cfg(not(target_arch = "x86_64"))] core::compile_error!("Test not supported on this arch");
    [asm.options.supported-options-nomem]
    • nomem: کد اسمبلی هیچ حافظه‌ای را که از بیرون کد اسمبلی در دسترس باشد نمی‌خواند و در آن نمی‌نویسد. در نتیجه کامپایلر می‌تواند مقدار متغیرهای سراسریِ تغییریافته را در طول اجرای کد اسمبلی در رجیسترها نگه دارد، زیرا می‌داند این کد آن‌ها را نمی‌خواند و تغییر نمی‌دهد. کامپایلر همچنین فرض می‌کند که کد اسمبلی هیچ نوع همگام‌سازی با نخ‌های دیگر، مثلاً از طریق حصارهای حافظه، انجام نمی‌دهد.
    #[cfg(target_arch = "x86_64")] {
let mut x = 0i32;
let z: i32;
// Accessing outside memory from assembly when `nomem` is
// specified is disallowed
unsafe {
    core::arch::asm!("mov {val:e}, dword ptr [{ptr}]",
        ptr = in(reg) &mut x,
        val = lateout(reg) z,
        options(nomem)
    )
}

// Writing to outside memory from assembly when `nomem` is
// specified is also undefined behaviour
unsafe {
    core::arch::asm!("mov  dword ptr [{ptr}], {val:e}",
        ptr = in(reg) &mut x,
        val = in(reg) z,
        options(nomem)
    )
}
}
    #[cfg(target_arch = "x86_64")] {
let x: i32 = 0;
let z: i32;
// If we allocate our own memory, such as via `push`, however.
// we can still use it
unsafe {
    core::arch::asm!("push {x}", "add qword ptr [rsp], 1", "pop {x}",
        x = inout(reg) x => z,
        options(nomem)
    );
}
assert_eq!(z, 1);
}
    [asm.options.supported-options-readonly]
    • readonly: کد اسمبلی در هیچ حافظه‌ای که از بیرون آن در دسترس باشد نمی‌نویسد. در نتیجه کامپایلر می‌تواند مقدار متغیرهای سراسریِ تغییرنیافته را در طول اجرای کد اسمبلی در رجیسترها نگه دارد، زیرا می‌داند این کد آن‌ها را تغییر نمی‌دهد. کامپایلر همچنین فرض می‌کند که کد اسمبلی هیچ نوع همگام‌سازی با نخ‌های دیگر، مثلاً از طریق حصارهای حافظه، انجام نمی‌دهد.
    #[cfg(target_arch = "x86_64")] {
let mut x = 0;
// We cannot modify outside memory when `readonly` is specified
unsafe {
    core::arch::asm!("mov dword ptr[{}], 1", in(reg) &mut x, options(readonly))
}
}
    #[cfg(target_arch = "x86_64")] {
let x: i64 = 0;
let z: i64;
// We can still read from it, though
unsafe {
    core::arch::asm!("mov {x}, qword ptr [{x}]",
        x = inout(reg) &x => z,
        options(readonly)
    );
}
assert_eq!(z, 0);
}
    #[cfg(target_arch = "x86_64")] {
let x: i64 = 0;
let z: i64;
// Same exception applies as with nomem.
unsafe {
    core::arch::asm!("push {x}", "add qword ptr [rsp], 1", "pop {x}",
        x = inout(reg) x => z,
        options(readonly)
    );
}
assert_eq!(z, 1);
}
    [asm.options.supported-options-preserves_flags]
    • preserves_flags: کد اسمبلی رجیستر پرچم‌ها را که در قواعد پایین تعریف شده است تغییر نمی‌دهد. این گزینه به کامپایلر اجازه می‌دهد پس از اجرای کد اسمبلی از محاسبهٔ دوبارهٔ پرچم‌های شرط صرف‌نظر کند.
    [asm.options.supported-options-noreturn]
    • noreturn: اجرای کد اسمبلی از انتهای آن به دستور بعدی ادامه پیدا نمی‌کند؛ اگر چنین شود، رفتار تعریف‌نشده است. بااین‌حال، پرش به بلوک‌های label همچنان مجاز است. اگر یکی از بلوک‌های label مقدار واحد بازگرداند، بلوک asm! نیز مقدار واحد بازمی‌گرداند؛ در غیر این صورت نوع بازگشتی آن !، یعنی هرگز، خواهد بود. همانند فراخوانی تابعی که بازنمی‌گردد، متغیرهای محلیِ درون دامنه پیش از اجرای کد اسمبلی تخریب نمی‌شوند.
    fn main() -> ! {
#[cfg(target_arch = "x86_64")] {
    // We can use an instruction to trap execution inside of a noreturn block
    unsafe { core::arch::asm!("ud2", options(noreturn)); }
}
#[cfg(not(target_arch = "x86_64"))] panic!("no return");
}
    #[cfg(target_arch = "x86_64")] {
// You are responsible for not falling past the end of a noreturn asm block
unsafe { core::arch::asm!("", options(noreturn)); }
}
    #[cfg(target_arch = "x86_64")]
let _: () = unsafe {
    // You may still jump to a `label` block
    core::arch::asm!("jmp {}", label {
        println!();
    }, options(noreturn));
};
    [asm.options.supported-options-nostack]
    • nostack: کد اسمبلی داده‌ای روی پشته قرار نمی‌دهد و در ناحیهٔ قرمز پشته، اگر هدف از آن پشتیبانی کند، نمی‌نویسد. اگر این گزینه استفاده نشود، کامپایلر تضمین می‌کند که اشاره‌گر پشته در آغاز کد اسمبلی مطابق ABI هدف برای فراخوانی تابع به‌درستی هم‌تراز باشد.
    #[cfg(target_arch = "x86_64")] {
// `push` and `pop` are UB when used with nostack
unsafe { core::arch::asm!("push rax", "pop rax", options(nostack)); }
}
    [asm.options.supported-options-att_syntax]
    • att_syntax: این گزینه فقط در x86 معتبر است و باعث می‌شود اسمبلر از حالت .att_syntax prefix اسمبلر GNU استفاده کند. عملوندهای رجیستر با یک % در ابتدای نام جای‌گذاری می‌شوند.
    #[cfg(target_arch = "x86_64")] {
let x: i32;
let y = 1i32;
// We need to use AT&T Syntax here. src, dest order for operands
unsafe {
    core::arch::asm!("mov {y:e}, {x:e}",
        x = lateout(reg) x,
        y = in(reg) y,
        options(att_syntax)
    );
}
assert_eq!(x, y);
}
    [asm.options.supported-options-raw]
    • raw: رشتهٔ قالب به‌صورت یک رشتهٔ اسمبلی خام تجزیه می‌شود و { و } معنای ویژه‌ای ندارند. این گزینه به‌ویژه برای گنجاندن کد اسمبلی خام از فایل خارجی با include_str! مفید است.
    [asm.options.checks]

    کامپایلر چند بررسی اضافی نیز روی گزینه‌ها انجام می‌دهد:

    [asm.options.checks-mutually-exclusive]
    • گزینه‌های nomem و readonly ناسازگارند؛ مشخص کردن هم‌زمان هر دو، خطای زمان کامپایل است.
    #[cfg(target_arch = "x86_64")] {
// nomem is strictly stronger than readonly, they can't be specified together
unsafe { core::arch::asm!("", options(nomem, readonly)); }
// ERROR: the `nomem` and `readonly` options are mutually exclusive
}
#[cfg(not(target_arch = "x86_64"))] core::compile_error!("Test not supported on this arch");
    [asm.options.checks-pure]
    • مشخص کردن pure برای بلوک اسمبلیِ بدون خروجی، یا بلوکی که همهٔ خروجی‌های آن کنار گذاشته می‌شوند (_)، خطای زمان کامپایل است.
    #[cfg(target_arch = "x86_64")] {
// pure blocks need at least one output
unsafe { core::arch::asm!("", options(pure)); }
// ERROR: asm with the `pure` option must have at least one output
}
#[cfg(not(target_arch = "x86_64"))] core::compile_error!("Test not supported on this arch");
    [asm.options.checks-noreturn]
    • مشخص کردن noreturn برای بلوک اسمبلی‌ای که خروجی دارد اما برچسب ندارد، خطای زمان کامپایل است.
    #[cfg(target_arch = "x86_64")] {
let z: i32;
// noreturn can't have outputs
unsafe { core::arch::asm!("mov {:e}, 1", out(reg) z, options(noreturn)); }
// ERROR: asm outputs are not allowed with the `noreturn` option
}
#[cfg(not(target_arch = "x86_64"))] core::compile_error!("Test not supported on this arch");
    [asm.options.checks-label-with-outputs]
    • وجود هر بلوک label در بلوک اسمبلیِ دارای خروجی، خطای زمان کامپایل است.
    [asm.options.naked_asm-restriction]

    naked_asm! فقط گزینه‌های att_syntax و raw را پشتیبانی می‌کند. گزینه‌های دیگر معنایی ندارند، زیرا اسمبلی درون‌خطی تمام بدنهٔ تابع را تعریف می‌کند.

    [asm.options.global_asm-restriction]

    global_asm! فقط گزینه‌های att_syntax و raw را پشتیبانی می‌کند. گزینه‌های دیگر برای اسمبلی درون‌خطی در دامنهٔ سراسری معنایی ندارند.

    fn main() {}
#[cfg(target_arch = "x86_64")]
// nomem is useless on global_asm!
core::arch::global_asm!("", options(nomem));
#[cfg(not(target_arch = "x86_64"))] core::compile_error!("Test not supported on this arch");
    [asm.rules]

    قواعد اسمبلی درون‌خطی

    [asm.rules.intro]

    برای جلوگیری از رفتار تعریف‌نشده، هنگام استفاده از اسمبلی درون‌خطی در دامنهٔ تابع (asm!) باید قواعد زیر رعایت شوند:

    [asm.rules.reg-not-input]
    • هر رجیستری که به‌عنوان ورودی مشخص نشده باشد، هنگام ورود به کد اسمبلی مقداری نامعین دارد.
      • «مقدار نامعین» در بافت اسمبلی درون‌خطی یعنی رجیستر می‌تواند به‌صورت غیرقطعی هر یک از مقادیر مجاز در معماری را داشته باشد. به‌ویژه، این مفهوم با undef در LLVM یکسان نیست؛ undef می‌تواند با هر بار خواندن مقدار متفاوتی داشته باشد، اما چنین مفهومی در کد اسمبلی وجود ندارد.
    [asm.rules.reg-not-output]
    • هر رجیستری که به‌عنوان خروجی مشخص نشده باشد باید هنگام خروج از کد اسمبلی همان مقدار زمان ورود را داشته باشد؛ وگرنه رفتار تعریف‌نشده است.
      • این قاعده فقط دربارهٔ رجیسترهایی است که می‌توان آن‌ها را به‌عنوان ورودی یا خروجی مشخص کرد. رجیسترهای دیگر از قواعد ویژهٔ هدف پیروی می‌کنند.
      • توجه کنید که ممکن است lateout به همان رجیستر یک in تخصیص یابد؛ در این حالت این قاعده اعمال نمی‌شود. بااین‌حال، کد نباید به این اتفاق تکیه کند، زیرا به نتیجهٔ تخصیص رجیستر بستگی دارد.
    [asm.rules.unwind]
    • اگر اجرای برنامه با بازکردن پشته از کد اسمبلی خارج شود، رفتار تعریف‌نشده است.
      • این قاعده در صورتی هم برقرار است که کد اسمبلی تابعی را فراخوانی کند و آن تابع پشته را باز کند.
    [asm.rules.mem-same-as-ffi]
    • مجموعهٔ محل‌های حافظه که کد اسمبلی اجازهٔ خواندن و نوشتن آن‌ها را دارد، همان مجموعهٔ مجاز برای یک تابع FFI است.
      • اگر گزینهٔ readonly فعال باشد، فقط خواندن حافظه مجاز است.
      • اگر گزینهٔ nomem فعال باشد، هیچ خواندن یا نوشتنی در حافظه مجاز نیست.
      • این قواعد شامل حافظهٔ خصوصیِ کد اسمبلی، مانند فضای پشته‌ای که درون آن تخصیص می‌یابد، نمی‌شوند.
    [asm.rules.black-box]
    • کامپایلر نمی‌تواند فرض کند دستورهای موجود در کد اسمبلی همان دستورهایی هستند که در نهایت اجرا خواهند شد.
      • در عمل، کامپایلر باید کد اسمبلی را جعبه‌ای سیاه در نظر بگیرد و فقط مشخصات رابط را لحاظ کند، نه خود دستورها را.
      • وصله کردن کد در زمان اجرا با سازوکارهای ویژهٔ هدف مجاز است.
      • بااین‌حال، تضمینی وجود ندارد که هر بلوک اسمبلی در کد منبع مستقیماً با یک نمونهٔ واحد از دستورها در فایل شیء متناظر باشد؛ کامپایلر می‌تواند کد اسمبلیِ بلوک‌های asm! را تکثیر کند یا نمونه‌های تکراری آن را ادغام کند.
    [asm.rules.stack-below-sp]
    • مگر آنکه گزینهٔ nostack فعال باشد، کد اسمبلی می‌تواند از فضای پشته در نشانی‌های پایین‌تر از اشاره‌گر پشته استفاده کند.
      • هنگام ورود به کد اسمبلی، تضمین می‌شود که اشاره‌گر پشته طبق ABI هدف برای فراخوانی تابع به‌درستی هم‌تراز باشد.
      • جلوگیری از سرریز پشته بر عهدهٔ شماست؛ برای نمونه، با وارسی تدریجی پشته مطمئن شوید که به صفحهٔ محافظ برخورد می‌کنید.
      • هنگام تخصیص حافظهٔ پشته باید اشاره‌گر پشته را طبق الزامات ABI هدف تنظیم کنید.
      • پیش از خروج از کد اسمبلی، اشاره‌گر پشته باید به مقدار اولیه بازگردد.
    [asm.rules.stack-above-sp]
    • مگر آنکه گزینهٔ nostack فعال باشد، اگر ABI هدف ذخیرهٔ برخی مقادیر در قاب پشتهٔ فراخواننده را الزامی بداند، کد اسمبلی می‌تواند آن قاب را تغییر دهد؛ برای نمونه هنگام ذخیرهٔ lr در PowerPC64.
    [asm.rules.noreturn]
    • اگر گزینهٔ noreturn فعال باشد و اجرا از انتهای کد اسمبلی به دستور بعدی ادامه پیدا کند، رفتار تعریف‌نشده است.
    [asm.rules.pure]
    • اگر گزینهٔ pure فعال باشد، داشتن اثر جانبی به‌جز خروجی‌های مستقیمِ asm! رفتار تعریف‌نشده است. همچنین اگر دو اجرای کد asm! با ورودی‌های یکسان خروجی‌های متفاوتی تولید کنند، رفتار تعریف‌نشده است.
      • همراه با گزینهٔ nomem، منظور از «ورودی‌ها» فقط ورودی‌های مستقیم asm! است.
      • همراه با گزینهٔ readonly، «ورودی‌ها» شامل ورودی‌های مستقیم کد اسمبلی و هر حافظه‌ای است که اجازهٔ خواندنش را دارد.
    [asm.rules.preserved-registers]
    • اگر گزینهٔ preserves_flags فعال باشد، رجیسترهای پرچم زیر باید هنگام خروج از کد اسمبلی به وضعیت قبلی بازگردند:
      • x86
        • پرچم‌های وضعیت در EFLAGS شامل CF، PF، AF، ZF، SF و OF.
        • تمام کلمهٔ وضعیت ممیز شناور.
        • پرچم‌های استثنای ممیز شناور در MXCSR شامل PE، UE، OE، ZE، DE و IE.
      • ARM
        • پرچم‌های شرط در CPSR شامل N، Z، C و V.
        • پرچم اشباع در CPSR یعنی Q.
        • پرچم‌های بزرگ‌تر یا برابر در CPSR یعنی GE.
        • پرچم‌های شرط در FPSCR شامل N، Z، C و V.
        • پرچم اشباع در FPSCR یعنی QC.
        • پرچم‌های استثنای ممیز شناور در FPSCR شامل IDC، IXC، UFC، OFC، DZC و IOC.
      • AArch64 و Arm64EC
        • پرچم‌های شرط، یعنی رجیستر NZCV.
        • وضعیت ممیز شناور، یعنی رجیستر FPSR.
      • RISC-V
        • پرچم‌های استثنای ممیز شناور در fcsr، یعنی fflags.
        • وضعیت افزونهٔ برداری شامل vtype، vl، vxsat و vxrm.
      • LoongArch
        • پرچم‌های شرط ممیز شناور در $fcc[0-7].
      • PowerPC/PowerPC64
        • بیت‌های وضعیت و ماندگارِ ممیز شناور در fpscr؛ همهٔ فیلدها به‌جز DRN، VE، OE، UE، ZE، XE، NI و RN.
        • بیت‌های وضعیت و ماندگارِ برداری در vscr؛ همهٔ فیلدها به‌جز NJ.
      • PowerPC SPE
        • بیت‌های ماندگار و وضعیت در spefscr؛ همهٔ فیلدها به‌جز FINXE، FINVE، FDBZE، FUNFE، FOVFE و FRMC.
      • s390x
        • رجیستر کد شرط cc.
    [asm.rules.x86-df]
    • در x86، پرچم جهت، یعنی DF در EFLAGS، هنگام ورود به کد اسمبلی صفر است و هنگام خروج نیز باید صفر باشد.
      • اگر پرچم جهت هنگام خروج از کد اسمبلی یک باشد، رفتار تعریف‌نشده است.
    [asm.rules.x86-x87]
    • در x86، پشتهٔ رجیسترهای ممیز شناور x87 باید بدون تغییر بماند، مگر آنکه همهٔ رجیسترهای st([0-7]) با out("st(0)") _, out("st(1)") _, ... به‌عنوان تخریب‌شونده مشخص شده باشند.
      • اگر همهٔ رجیسترهای x87 تخریب‌شونده باشند، تضمین می‌شود که پشتهٔ رجیسترهای x87 هنگام ورود به کد اسمبلی خالی باشد. کد اسمبلی باید مطمئن شود که این پشته هنگام خروج نیز خالی است.
    #[cfg(target_arch = "x86_64")]
pub fn fadd(x: f64, y: f64) -> f64 {
  let mut out = 0f64;
  let mut top = 0u16;
  // we can do complex stuff with x87 if we clobber the entire x87 stack
  unsafe { core::arch::asm!(
    "fld qword ptr [{x}]",
    "fld qword ptr [{y}])",
    "faddp",
    "fstp qword ptr [{out}]",
    "xor eax, eax",
    "fstsw ax",
    "shl eax, 11",
    x = in(reg) &x,
    y = in(reg) &y,
    out = in(reg) &mut out,
    out("st(0)") _, out("st(1)") _, out("st(2)") _, out("st(3)") _,
    out("st(4)") _, out("st(5)") _, out("st(6)") _, out("st(7)") _,
    out("eax") top
  );}

  assert_eq!(top & 0x7, 0);
  out
}

pub fn main() {
#[cfg(target_arch = "x86_64")]{
  assert_eq!(fadd(1.0, 1.0), 2.0);
}
}
    [asm.rules.arm64ec]
    • در arm64ec، هنگام فراخوانی تابع‌ها استفاده از بررسی‌کننده‌های فراخوانی با واسط‌های مناسب الزامی است.
    [asm.rules.only-on-exit]
    • الزام بازگرداندن اشاره‌گر پشته و رجیسترهای غیرخروجی به مقدار اولیه فقط هنگام خروج از کد اسمبلی برقرار است.
      • بنابراین کد اسمبلی‌ای که نه از انتهای خود به دستور بعدی ادامه می‌دهد و نه به بلوک label می‌پرد، حتی اگر noreturn نداشته باشد، لازم نیست این رجیسترها را حفظ کند.
      • هنگام بازگشت به کد اسمبلیِ بلوک asm! متفاوتی از بلوکی که واردش شده‌اید، مثلاً برای تعویض بافت، این رجیسترها باید مقدار زمان ورود به بلوک asm!ای را داشته باشند که اکنون از آن خارج می‌شوید.
        • نمی‌توانید از کد اسمبلی یک بلوک asm! که وارد آن نشده‌اید خارج شوید. همچنین نمی‌توانید از کد اسمبلی یک بلوک asm! که پیش‌تر از کد اسمبلی آن خارج شده‌اید دوباره خارج شوید (بدون آنکه ابتدا دوباره وارد آن شده باشید).
        • تعویض هر وضعیت ویژهٔ هدف، مانند ذخیره‌سازی محلیِ نخ و مرزهای پشته، بر عهدهٔ شماست.
        • نمی‌توانید از نشانی‌ای در یک بلوک asm! به نشانی‌ای در بلوک دیگری بپرید، حتی در همان تابع یا بلوک، مگر آنکه بافت‌هایشان را بالقوه متفاوت و نیازمند تعویض بافت در نظر بگیرید. نباید فرض کنید مقدار خاصی در این بافت‌ها، مانند اشاره‌گر فعلی پشته یا مقادیر موقت پایین‌تر از آن، میان دو بلوک asm! بدون تغییر می‌ماند.
        • مجموعهٔ محل‌های حافظهٔ قابل دسترسی، اشتراک مجموعه‌های مجاز در بلوک‌های asm!ای است که واردشان شده‌اید و از آن‌ها خارج شده‌اید.
    [asm.rules.not-successive]
    • نمی‌توانید فرض کنید دو بلوک asm! مجاور در کد منبع، حتی بدون هیچ کدی میانشان، در فایل باینری نیز در نشانی‌های متوالی و بدون دستور دیگری در میان قرار می‌گیرند.
    [asm.rules.not-exactly-once]
    • نمی‌توانید فرض کنید یک بلوک asm! دقیقاً یک بار در باینری خروجی ظاهر می‌شود. کامپایلر مجاز است چند نسخه از بلوک asm! را نمونه‌سازی (instantiate) کند؛ مثلاً وقتی تابعِ دربرگیرندهٔ آن در چند محل inline شود.
    [asm.rules.x86-prefix-restriction]
    • در x86، اسمبلی درون‌خطی نباید با پیشوند دستوری مانند LOCK پایان یابد که بر دستورهای تولیدشده توسط کامپایلر اعمال شود.
      • به‌دلیل شیوهٔ کامپایل اسمبلی درون‌خطی، کامپایلر اکنون قادر به تشخیص این حالت نیست، اما ممکن است در آینده آن را تشخیص دهد و رد کند.
    [asm.rules.preserves_flags]

    Note

    [asm.naked-rules]

    قواعد اسمبلی درون‌خطی برهنه

    [asm.naked-rules.intro]

    برای جلوگیری از رفتار تعریف‌نشده، هنگام استفاده از اسمبلی درون‌خطی در دامنهٔ تابع‌های برهنه (naked_asm!) باید قواعد زیر رعایت شوند:

    [asm.naked-rules.reg-not-input]
    • هر رجیستری که طبق قرارداد فراخوانی و امضای تابع برای ورودی‌های تابع استفاده نشود، هنگام ورود به بلوک naked_asm! مقداری نامعین دارد.
      • «مقدار نامعین» در بافت اسمبلی درون‌خطی یعنی رجیستر می‌تواند به‌صورت غیرقطعی هر یک از مقادیر مجاز در معماری را داشته باشد. به‌ویژه، این مفهوم با undef در LLVM یکسان نیست؛ undef می‌تواند با هر بار خواندن مقدار متفاوتی داشته باشد، اما چنین مفهومی در کد اسمبلی وجود ندارد.
    [asm.naked-rules.callee-saved-registers]
    • تمام رجیسترهایی که حفظشان بر عهدهٔ تابع فراخوانده‌شده است باید هنگام بازگشت همان مقدار زمان ورود را داشته باشند.
    [asm.naked-rules.caller-saved-registers]
    • رجیسترهای caller-saved را می‌توان آزادانه به کار برد.
    [asm.naked-rules.noreturn]
    • اگر اجرا از انتهای کد اسمبلی عبور کند و به دستور بعدی ادامه دهد، رفتار تعریف‌نشده است.
      • انتظار می‌رود هر مسیر در کد اسمبلی با دستور بازگشت پایان یابد یا واگرا باشد و بازنگردد.
    [asm.naked-rules.mem-same-as-ffi]
    • مجموعهٔ محل‌های حافظه که کد اسمبلی اجازهٔ خواندن و نوشتن آن‌ها را دارد، همان مجموعهٔ مجاز برای یک تابع FFI است.
    [asm.naked-rules.black-box]
    • کامپایلر نمی‌تواند فرض کند دستورهای موجود در بلوک naked_asm! همان دستورهایی هستند که واقعاً اجرا خواهند شد.
      • در عمل، کامپایلر باید naked_asm! را جعبه‌ای سیاه در نظر بگیرد و فقط مشخصات رابط را لحاظ کند، نه خود دستورها را.
      • وصله کردن کد در زمان اجرا با سازوکارهای ویژهٔ هدف مجاز است.
    [asm.naked-rules.unwind]
    • بازکردن پشته برای خروج از بلوک naked_asm! مجاز است.
      • برای رفتار صحیح، باید از دستورهای راهنمای مناسب اسمبلر استفاده شود تا فرادادهٔ بازکردن پشته را تولید کنند.
    #[cfg(target_arch = "x86_64")] {
#[unsafe(naked)]
extern "sysv64-unwind" fn unwinding_naked() {
    core::arch::naked_asm!(
        // "CFI" here stands for "call frame information".
        ".cfi_startproc",
        // The CFA (canonical frame address) is the value of `rsp`
        // before the `call`, i.e. before the return address, `rip`,
        // was pushed to `rsp`, so it's eight bytes higher in memory
        // than `rsp` upon function entry (after `rip` has been
        // pushed).
        //
        // This is the default, so we don't have to write it.
        //".cfi_def_cfa rsp, 8",
        //
        // The traditional thing to do is to preserve the base
        // pointer, so we'll do that.
        "push rbp",
        // Since we've now extended the stack downward by 8 bytes in
        // memory, we need to adjust the offset to the CFA from `rsp`
        // by another 8 bytes.
        ".cfi_adjust_cfa_offset 8",
        // We also then annotate where we've stored the caller's value
        // of `rbp`, relative to the CFA, so that when unwinding into
        // the caller we can find it, in case we need it to calculate
        // the caller's CFA relative to it.
        //
        // Here, we've stored the caller's `rbp` starting 16 bytes
        // below the CFA.  I.e., starting from the CFA, there's first
        // the `rip` (which starts 8 bytes below the CFA and continues
        // up to it), then there's the caller's `rbp` that we just
        // pushed.
        ".cfi_offset rbp, -16",
        // As is traditional, we set the base pointer to the value of
        // the stack pointer.  This way, the base pointer stays the
        // same throughout the function body.
        "mov rbp, rsp",
        // We can now track the offset to the CFA from the base
        // pointer.  This means we don't need to make any further
        // adjustments until the end, as we don't change `rbp`.
        ".cfi_def_cfa_register rbp",
        // We can now call a function that may panic.
        "call {f}",
        // Upon return, we restore `rbp` in preparation for returning
        // ourselves.
        "pop rbp",
        // Now that we've restored `rbp`, we must specify the offset
        // to the CFA again in terms of `rsp`.
        ".cfi_def_cfa rsp, 8",
        // Now we can return.
        "ret",
        ".cfi_endproc",
        f = sym may_panic,
    )
}

extern "sysv64-unwind" fn may_panic() {
    panic!("unwind");
}
}

    درستی و اعتبار

    [asm.validity.necessary-but-not-sufficient]

    علاوه بر تمام قواعد پیشین، آرگومان رشته‌ایِ asm! باید در نهایت، پس از ارزیابی سایر آرگومان‌ها، انجام قالب‌بندی و تبدیل عملوندها، به اسمبلی‌ای تبدیل شود که هم از نظر نحوی درست باشد و هم از نظر معنایی برای معماری هدف معتبر باشد. قواعد قالب‌بندی به کامپایلر امکان می‌دهند اسمبلی با نحو صحیح تولید کند. قواعد مربوط به عملوندها نیز تبدیل معتبر عملوندهای Rust به کد اسمبلی و برعکس را ممکن می‌کنند. رعایت این قواعد برای درستی و اعتبار اسمبلیِ نهاییِ بسط‌یافته لازم است، اما کافی نیست. برای نمونه:

    • ممکن است آرگومان‌ها در جایگاهی قرار بگیرند که پس از قالب‌بندی از نظر نحوی نادرست باشد.
    • ممکن است یک دستور درست نوشته شده باشد، اما عملوندهایی از نظر معماری نامعتبر دریافت کند.
    • ممکن است دستوری که معماری آن را مشخص نکرده است، به کدی نامشخص اسمبل شود.
    • ممکن است مجموعه‌ای از دستورها که هرکدام به‌تنهایی درست و معتبرند، در صورت قرار گرفتن بلافاصله پشت سر هم رفتار تعریف‌نشده ایجاد کنند.
    [asm.validity.non-exhaustive]

    در نتیجه، این قواعد جامع نیستند. کامپایلر موظف نیست درستی و اعتبار رشتهٔ اولیه یا اسمبلی نهاییِ تولیدشده را بررسی کند. اسمبلر ممکن است این بررسی را انجام دهد، اما الزامی ندارد. هنگام استفاده از asm!، حتی یک خطای تایپی ممکن است برای ناسالم شدن برنامه کافی باشد؛ قواعد اسمبلی نیز ممکن است هزاران صفحه از راهنماهای مرجع معماری را در بر بگیرند. برنامه‌نویسان باید دقت لازم را به خرج دهند، زیرا استفاده از این قابلیت unsafe به معنی پذیرفتن مسئولیت رعایت قواعد کامپایلر و معماری است.
