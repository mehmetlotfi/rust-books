    [expr.placeholder]

    عبارت‌های _ (Underscore expressions)

    [expr.placeholder.syntax]

    Syntax
    UnderscoreExpression → _

    [expr.placeholder.intro]

    عبارت‌های زیرخط که با نماد _ نشان داده می‌شوند، برای نمایش جای‌نگهدار در یک انتساب ساختارزدایی به کار می‌روند.

    [expr.placeholder.lhs-assignment-only]

    آن‌ها فقط می‌توانند در سمت چپ انتساب ظاهر شوند.

    [expr.placeholder.pattern]

    توجه کنید که این مفهوم با الگوی عام متفاوت است.

    نمونه‌هایی از عبارت _:

    let p = (1, 2);
let mut a = 0;
(_, a) = p;

struct Position {
    x: u32,
    y: u32,
}

Position { x: a, y: _ } = Position{ x: 2, y: 3 };

// unused result, assignment to `_` used to declare intent and remove a warning
_ = 2 + 2;
// triggers unused_must_use warning
// 2 + 2;

// equivalent technique using a wildcard pattern in a let-binding
let _ = 2 + 2;
