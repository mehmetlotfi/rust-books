    [expr.if]

    عبارت‌های if

    [expr.if.syntax]

    Syntax
    IfExpression →
        if Conditions BlockExpressionNoInnerAttributes
        ( else ( BlockExpressionNoInnerAttributes | IfExpression ) )?

    Conditions →
          Expressionexcept StructExpression
        | LetChain

    LetChain → LetChainCondition ( && LetChainCondition )*

    LetChainCondition →
          Expressionexcept ExcludedConditions
        | OuterAttribute* let Pattern = Scrutineeexcept ExcludedConditions

    ExcludedConditions →
          StructExpression
        | LazyBooleanExpression
        | RangeExpr
        | RangeFromExpr
        | RangeInclusiveExpr
        | AssignmentExpression
        | CompoundAssignmentExpression

    [expr.if.intro]

    نحو عبارت if شامل دنباله‌ای از یک یا چند عملوند شرط جداشده با &&، سپس بلوک نتیجه، هر تعداد شرط و بلوک else if و یک بلوک پایانی اختیاری else است.

    [expr.if.condition]

    عملوندهای شرط باید یا یک Expression با نوع بولی باشند یا یک تطبیق شرطی let.

    [expr.if.condition-true]

    اگر همهٔ عملوندهای شرط به true ارزیابی شوند و همهٔ الگوهای let با بررسی‌شوندههای خود با موفقیت مطابقت کنند، بلوک نتیجه اجرا و همهٔ بلوک‌های بعدی else if یا else رد می‌شوند.

    [expr.if.else-if]

    اگر هر عملوند شرط به false ارزیابی شود یا هر الگوی let با بررسی‌شوندهٔ خود مطابقت نکند، بلوک نتیجه رد و شرط else if بعدی ارزیابی می‌شود.

    [expr.if.else]

    اگر همهٔ شرط‌های if و else if به false ارزیابی شوند، بلوک else در صورت وجود اجرا می‌شود.

    [expr.if.result]

    عبارت if به همان مقدار بلوک اجراشده ارزیابی می‌شود؛ و اگر هیچ بلوکی ارزیابی نشود، مقدار آن () است.

    [expr.if.type]

    عبارت if باید در همهٔ وضعیت‌ها نوع یکسانی داشته باشد.

    let x = 3;
if x == 4 {
    println!("x is four");
} else if x == 3 {
    println!("x is three");
} else {
    println!("x is something else");
}

// `if` can be used as an expression.
let y = if 12 * 15 > 150 {
    "Bigger"
} else {
    "Smaller"
};
assert_eq!(y, "Bigger");
    [expr.if.diverging]

    عبارت if زمانی واگرا می‌شود که عبارت شرط واگرا باشد یا همهٔ بازوها واگرا باشند.

    fn diverging_condition() -> ! {
    // Diverges because the condition expression diverges
    if loop {} {
        ()
    } else {
        ()
    };
    // The semicolon above is important: The type of the `if` expression is
    // `()`, despite being diverging. When the final body expression is
    // elided, the type of the body is inferred to ! because the function body
    // diverges. Without the semicolon, the `if` would be the tail expression
    // with type `()`, which would fail to match the return type `!`.
}

fn diverging_arms() -> ! {
    // Diverges because all arms diverge
    if true {
        loop {}
    } else {
        loop {}
    }
}
    [expr.if.let]

    الگوهای if let

    [expr.if.let.intro]

    الگوهای let در شرط if اجازه می‌دهند هنگامی که الگو با موفقیت مطابقت می‌کند، متغیرهای جدیدی به حوزه متصل شوند.

    نمونه‌های زیر اتصال با الگوهای let را نشان می‌دهند:

    let dish = ("Ham", "Eggs");

// This body will be skipped because the pattern is refuted.
if let ("Bacon", b) = dish {
    println!("Bacon is served with {}", b);
} else {
    // This block is evaluated instead.
    println!("No bacon will be served");
}

// This body will execute.
if let ("Ham", b) = dish {
    println!("Ham is served with {}", b);
}

if let _ = 5 {
    println!("Irrefutable patterns are always true");
}
    [expr.if.let.or-pattern]

    با عملگر | می‌توان چند الگو را مشخص کرد. معنای آن با | در عبارت‌های match یکسان است:

    enum E {
    X(u8),
    Y(u8),
    Z(u8),
}
let v = E::Y(12);
if let E::X(n) | E::Y(n) = v {
    assert_eq!(n, 12);
}
    [expr.if.chains]

    زنجیرهٔ شرط‌ها

    [expr.if.chains.intro]

    چند عملوند شرط را می‌توان با && از هم جدا کرد.

    [expr.if.chains.order]

    همانند یک LazyBooleanExpression نوع &&، هر عملوند از چپ به راست ارزیابی می‌شود تا یک عملوند به false برسد یا تطبیق let شکست بخورد؛ در این حالت عملوندهای بعدی ارزیابی نمی‌شوند.

    [expr.if.chains.bindings]

    اتصال‌های هر الگو وارد حوزه می‌شوند تا در عملوند شرط بعدی و بلوک نتیجه در دسترس باشند.

    نمونهٔ زیر زنجیره‌کردن چند عبارت، ترکیب اتصال‌های let و عبارت‌های بولی، و ارجاع عبارت‌ها به اتصال الگوهای عبارت‌های پیشین را نشان می‌دهد:

    fn single() {
    let outer_opt = Some(Some(1i32));

    if let Some(inner_opt) = outer_opt
        && let Some(number) = inner_opt
        && number == 1
    {
        println!("Peek a boo");
    }
}

    کد بالا بدون استفاده از زنجیرهٔ شرط‌ها معادل کد زیر است:

    fn nested() {
    let outer_opt = Some(Some(1i32));

    if let Some(inner_opt) = outer_opt {
        if let Some(number) = inner_opt {
            if number == 1 {
                println!("Peek a boo");
            }
        }
    }
}
    [expr.if.chains.or]

    اگر هر عملوند شرط یک الگوی let باشد، به‌دلیل ابهام و تقدم با بررسی‌شوندهٔ let، هیچ عملوند شرطی نمی‌تواند عبارت عملگر بولی تنبل از نوع || باشد.

    مثال اگر عبارت || لازم باشد، می‌توان از پرانتز استفاده کرد. برای مثال:

    let foo = Some(123);
let condition1 = true;
let condition2 = false;
if let Some(x) = foo
    // Parentheses are required here.
    && (condition1 || condition2)
{}
    [expr.if.edition2024]

    تفاوت‌های ویرایش 2024 پیش از ویرایش ۲۰۲۴، زنجیره‌های let پشتیبانی نمی‌شوند؛ یعنی دستور زبان LetChain در عبارت if مجاز نیست.
