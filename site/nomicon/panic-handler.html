راست‌نومیکونThe Rustonomicon

    #[panic_handler]

    #[panic_handler] برای تعریف رفتار panic! در برنامه‌های #![no_std] به کار می‌رود. اتریبیوت #[panic_handler] باید روی تابعی با امضای fn(&PanicInfo) -> ! اعمال شود و چنین تابعی باید یک بار در گراف وابستگی یک crate از نوع binary / dylib / cdylib ظاهر شود. API مربوط به PanicInfo را می‌توانید در مستندات API ببینید.

    از آنجا که برنامه‌های #![no_std] خروجی استاندارد ندارند و برخی برنامه‌های #![no_std]، مثلاً برنامه‌های نهفته، برای توسعه و برای انتشار به رفتارهای panic متفاوتی نیاز دارند، می‌تواند مفید باشد که crateهای panic داشته باشیم، یعنی crateهایی که فقط یک #[panic_handler] دارند. به این ترتیب برنامه‌ها می‌توانند رفتار panic را به‌سادگی و تنها با پیوند دادن به یک crate panic دیگر عوض کنند.

    در ادامه مثالی نشان داده شده است که در آن یک برنامه بسته به اینکه با پروفایل dev (cargo build) یا با پروفایل release (cargo build --release) کامپایل شود، رفتار panic متفاوتی دارد.

    crate با نام panic-semihosting – پیام‌های panic را با semihosting در stderr میزبان ثبت می‌کند:

    #![no_std]

use core::fmt::{Write, self};
use core::panic::PanicInfo;

struct HStderr {
    // ..
    _0: (),
}

impl HStderr {
    fn new() -> HStderr { HStderr { _0: () } }
}

impl fmt::Write for HStderr {
    fn write_str(&mut self, _: &str) -> fmt::Result { Ok(()) }
}

#[panic_handler]
fn panic(info: &PanicInfo) -> ! {
    let mut host_stderr = HStderr::new();

    // logs "panicked at '$reason', src/main.rs:27:4" to the host stderr
    writeln!(host_stderr, "{}", info).ok();

    loop {}
}

    crate با نام panic-halt – نخ را هنگام panic متوقف می‌کند؛ پیام‌ها دور ریخته می‌شوند:

    #![no_std]

use core::panic::PanicInfo;

#[panic_handler]
fn panic(_info: &PanicInfo) -> ! {
    loop {}
}

    crate با نام app:

    #![no_std]

// dev profile
#[cfg(debug_assertions)]
extern crate panic_semihosting;

// release profile
#[cfg(not(debug_assertions))]
extern crate panic_halt;

fn main() {
    // ..
}
