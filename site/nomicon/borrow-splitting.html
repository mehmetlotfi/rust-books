    تقسیم قرض‌ها (Splitting Borrows)

    ویژگی انحصار متقابل ارجاع‌های تغییرپذیر هنگام کار با یک ساختار مرکب می‌تواند بسیار محدودکننده باشد. بررسی‌گر قرض (borrow checker، به اختصار borrowck) چند چیز پایه را می‌فهمد، اما خیلی راحت از کار می‌افتد. این بررسی‌گر ساختارها را به اندازه‌ی کافی می‌فهمد تا بداند می‌توان فیلدهای مجزای یک ساختار را هم‌زمان قرض گرفت. بنابراین این کد امروز کار می‌کند:

    struct Foo {
    a: i32,
    b: i32,
    c: i32,
}

let mut x = Foo {a: 0, b: 0, c: 0};
let a = &mut x.a;
let b = &mut x.b;
let c = &x.c;
*b += 1;
let c2 = &x.c;
*a += 10;
println!("{} {} {} {}", a, b, c, c2);

    اما borrowck آرایه‌ها یا slice ها را به هیچ شکلی نمی‌فهمد، بنابراین این کد کار نمی‌کند:

    let mut x = [1, 2, 3];
let a = &mut x[0];
let b = &mut x[1];
println!("{} {}", a, b);
    error[E0499]: cannot borrow `x[..]` as mutable more than once at a time
 --> src/lib.rs:4:18
  |
3 |     let a = &mut x[0];
  |                  ---- first mutable borrow occurs here
4 |     let b = &mut x[1];
  |                  ^^^^ second mutable borrow occurs here
5 |     println!("{} {}", a, b);
6 | }
  | - first borrow ends here

error: aborting due to previous error

    هرچند معقول بود که borrowck بتواند این حالت ساده را بفهمد، اما روشن است که امید بستن به اینکه borrowck در نوع‌های کانتینر عمومی مانند درخت، مجزا بودن را تشخیص دهد بی‌فایده است، به‌ویژه اگر کلیدهای متفاوت واقعاً به یک مقدار نگاشت شوند.

    برای اینکه به borrowck «یاد بدهیم» کاری که می‌کنیم درست است، باید به کد unsafe پایین بیاییم. برای مثال، slice های تغییرپذیر تابعی به نام split_at_mut عرضه می‌کنند که slice را مصرف می‌کند و دو slice تغییرپذیر برمی‌گرداند. یکی برای هر چیز سمت چپ اندیس، و دیگری برای هر چیز سمت راست آن. به‌طور شهودی می‌دانیم این کار امن است چون slice ها هم‌پوشانی ندارند و بنابراین alias نیستند. با این حال پیاده‌سازی آن به مقداری ناامنی (unsafety) نیاز دارد:

    use std::slice::from_raw_parts_mut;
struct FakeSlice<T>(T);
impl<T> FakeSlice<T> {
fn len(&self) -> usize { unimplemented!() }
fn as_mut_ptr(&mut self) -> *mut T { unimplemented!() }
pub fn split_at_mut(&mut self, mid: usize) -> (&mut [T], &mut [T]) {
    let len = self.len();
    let ptr = self.as_mut_ptr();

    unsafe {
        assert!(mid <= len);

        (from_raw_parts_mut(ptr, mid),
         from_raw_parts_mut(ptr.add(mid), len - mid))
    }
}
}

    این در واقع کمی ظریف است. برای اینکه هرگز دو &mut به یک مقدار نسازیم، صراحتاً slice های کاملاً جدیدی را از طریق اشاره‌گرهای خام می‌سازیم.

    اما ظریف‌تر از آن، نحوه‌ی کار iterator هایی است که ارجاع‌های تغییرپذیر تولید می‌کنند. صفت Iterator به شکل زیر تعریف شده است:

    trait Iterator {
    type Item;

    fn next(&mut self) -> Option<Self::Item>;
}

    با این تعریف، Self::Item هیچ ارتباطی با self ندارد. این یعنی می‌توانیم next را چندین بار پشت سر هم صدا بزنیم و همه‌ی نتایج را هم‌زمان نگه داریم. این برای iterator های by-value که دقیقاً چنین معناشناسی‌ای دارند کاملاً مشکلی ندارد. برای ارجاع‌های اشتراکی هم در واقع مشکلی نیست، چون آن‌ها تعداد دلخواهی ارجاع به یک چیز را می‌پذیرند (هرچند iterator باید شیئی جدا از چیزی باشد که به اشتراک گذاشته می‌شود).

    اما ارجاع‌های تغییرپذیر این را به هم می‌ریزند. در نگاه اول ممکن است کاملاً با این API ناسازگار به نظر برسند، زیرا چندین ارجاع تغییرپذیر به یک شیء تولید می‌کند!

    با این حال در واقع کار می‌کند، دقیقاً به این دلیل که iterator ها اشیای یک‌بارمصرف هستند. هر چیزی که یک IterMut تولید کند حداکثر یک بار تولید می‌شود، بنابراین در عمل هرگز چندین ارجاع تغییرپذیر به یک تکه داده تولید نمی‌کنیم.

    شاید جالب باشد که iterator های تغییرپذیر برای بسیاری از نوع‌ها برای پیاده‌سازی نیازی به کد unsafe ندارند!

    برای مثال، این یک لیست پیوندی یک‌طرفه است:

    fn main() {}
type Link<T> = Option<Box<Node<T>>>;

struct Node<T> {
    elem: T,
    next: Link<T>,
}

pub struct LinkedList<T> {
    head: Link<T>,
}

pub struct IterMut<'a, T: 'a>(Option<&'a mut Node<T>>);

impl<T> LinkedList<T> {
    fn iter_mut(&mut self) -> IterMut<T> {
        IterMut(self.head.as_mut().map(|node| &mut **node))
    }
}

impl<'a, T> Iterator for IterMut<'a, T> {
    type Item = &'a mut T;

    fn next(&mut self) -> Option<Self::Item> {
        self.0.take().map(|node| {
            self.0 = node.next.as_mut().map(|node| &mut **node);
            &mut node.elem
        })
    }
}

    این هم یک slice تغییرپذیر:

    fn main() {}
use std::mem;

pub struct IterMut<'a, T: 'a>(&'a mut[T]);

impl<'a, T> Iterator for IterMut<'a, T> {
    type Item = &'a mut T;

    fn next(&mut self) -> Option<Self::Item> {
        let slice = mem::take(&mut self.0);
        if slice.is_empty() { return None; }

        let (l, r) = slice.split_at_mut(1);
        self.0 = r;
        l.get_mut(0)
    }
}

impl<'a, T> DoubleEndedIterator for IterMut<'a, T> {
    fn next_back(&mut self) -> Option<Self::Item> {
        let slice = mem::take(&mut self.0);
        if slice.is_empty() { return None; }

        let new_len = slice.len() - 1;
        let (l, r) = slice.split_at_mut(new_len);
        self.0 = l;
        r.get_mut(0)
    }
}

    و این هم یک درخت دودویی:

    fn main() {}
use std::collections::VecDeque;

type Link<T> = Option<Box<Node<T>>>;

struct Node<T> {
    elem: T,
    left: Link<T>,
    right: Link<T>,
}

pub struct Tree<T> {
    root: Link<T>,
}

struct NodeIterMut<'a, T: 'a> {
    elem: Option<&'a mut T>,
    left: Option<&'a mut Node<T>>,
    right: Option<&'a mut Node<T>>,
}

enum State<'a, T: 'a> {
    Elem(&'a mut T),
    Node(&'a mut Node<T>),
}

pub struct IterMut<'a, T: 'a>(VecDeque<NodeIterMut<'a, T>>);

impl<T> Tree<T> {
    pub fn iter_mut(&mut self) -> IterMut<T> {
        let mut deque = VecDeque::new();
        if let Some(root) = self.root.as_mut() {
            deque.push_front(root.iter_mut());
        }
        IterMut(deque)
    }
}

impl<T> Node<T> {
    pub fn iter_mut(&mut self) -> NodeIterMut<T> {
        NodeIterMut {
            elem: Some(&mut self.elem),
            left: self.left.as_deref_mut(),
            right: self.right.as_deref_mut(),
        }
    }
}

impl<'a, T> Iterator for NodeIterMut<'a, T> {
    type Item = State<'a, T>;

    fn next(&mut self) -> Option<Self::Item> {
        self.left.take().map(State::Node).or_else(|| {
            self.elem
                .take()
                .map(State::Elem)
                .or_else(|| self.right.take().map(State::Node))
        })
    }
}

impl<'a, T> DoubleEndedIterator for NodeIterMut<'a, T> {
    fn next_back(&mut self) -> Option<Self::Item> {
        self.right.take().map(State::Node).or_else(|| {
            self.elem
                .take()
                .map(State::Elem)
                .or_else(|| self.left.take().map(State::Node))
        })
    }
}

impl<'a, T> Iterator for IterMut<'a, T> {
    type Item = &'a mut T;
    fn next(&mut self) -> Option<Self::Item> {
        loop {
            match self.0.front_mut().and_then(Iterator::next) {
                Some(State::Elem(elem)) => return Some(elem),
                Some(State::Node(node)) => self.0.push_front(node.iter_mut()),
                None => {
                    self.0.pop_front()?;
                }
            }
        }
    }
}

impl<'a, T> DoubleEndedIterator for IterMut<'a, T> {
    fn next_back(&mut self) -> Option<Self::Item> {
        loop {
            match self.0.back_mut().and_then(DoubleEndedIterator::next_back) {
                Some(State::Elem(elem)) => return Some(elem),
                Some(State::Node(node)) => self.0.push_back(node.iter_mut()),
                None => {
                    self.0.pop_back()?;
                }
            }
        }
    }
}

    همه‌ی این‌ها کاملاً امن هستند و روی Rust پایدار (stable) کار می‌کنند! این در نهایت از همان حالت ساده‌ی ساختار که پیش‌تر دیدیم نتیجه می‌شود: Rust می‌فهمد که می‌توانید یک ارجاع تغییرپذیر را با امنیت به زیرفیلدها تقسیم کنید. سپس می‌توانیم مصرف دائمی یک ارجاع را از طریق Option ها کدگذاری کنیم (یا در مورد slice ها، با جایگزین کردن آن با یک slice خالی).
