    درج و حذف

    چیزی که slice ارائه نمی‌دهد insert و remove است، پس بیایید در ادامه آن‌ها را بنویسیم.

    Insert باید همهٔ عناصر موجود در اندیس هدف را یک خانه به راست جابه‌جا کند. برای این کار باید از ptr::copy استفاده کنیم، که نسخهٔ ما از memmove زبان C است. این تابع یک تکه از حافظه را از یک مکان به مکان دیگر کپی می‌کند و حالتی را که مبدأ و مقصد همپوشانی دارند (که اینجا قطعاً اتفاق می‌افتد) درست مدیریت می‌کند.

    اگر در اندیس i درج کنیم، می‌خواهیم با استفاده از len قدیمی، [i .. len] را به [i+1 .. len+1] منتقل کنیم.

    pub fn insert(&mut self, index: usize, elem: T) {
    // Note: `<=` because it's valid to insert after everything
    // which would be equivalent to push.
    assert!(index <= self.len, "index out of bounds");
    if self.len == self.cap { self.grow(); }

    unsafe {
        // ptr::copy(src, dest, len): "copy from src to dest len elems"
        ptr::copy(
            self.ptr.as_ptr().add(index),
            self.ptr.as_ptr().add(index + 1),
            self.len - index,
        );
        ptr::write(self.ptr.as_ptr().add(index), elem);
    }

    self.len += 1;
}

    Remove به شیوهٔ معکوس رفتار می‌کند. باید همهٔ عناصر را از [i+1 .. len + 1] به [i .. len] منتقل کنیم و در این کار از len جدید استفاده کنیم.

    pub fn remove(&mut self, index: usize) -> T {
    // Note: `<` because it's *not* valid to remove after everything
    assert!(index < self.len, "index out of bounds");
    unsafe {
        self.len -= 1;
        let result = ptr::read(self.ptr.as_ptr().add(index));
        ptr::copy(
            self.ptr.as_ptr().add(index + 1),
            self.ptr.as_ptr().add(index),
            self.len - index,
        );
        result
    }
}
