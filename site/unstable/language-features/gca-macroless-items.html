    gca_macroless_args

    پیاده‌سازی آیتم‌های const را تحت #![feature(gca_min_const_items)] و #![feature(gca_const_items)] بدون ماکروی gca! ممکن می‌کند.

    issue ردیابی این قابلیت: #162540

    هشدار: این قابلیت ناقص است؛ طراحی و نحو آن ممکن است تغییر کند.

    قابلیت‌های مرتبط:

    • gca_min_const_items. برای اینکه gca! چیست، آن مستند را ببینید. این قابلیت از آرگومان‌های const با نمایش مستقیم، در سمت راست آیتم‌های const و بدون ماکرو، پشتیبانی می‌کند.
    • gca_macroless_args. نسخه‌ای از این قابلیت که برای آرگومان‌های const در موقعیت‌های دیگر کار می‌کند

    نمونه‌ها

    در اینجا نمونه‌ای از gca_min_const_items آمده است:

    #![allow(incomplete_features)]
#![feature(
    gca_min_const_items,
    gca_const_items,
    gca_macroless_args,
    generic_const_items,
)]

trait Trait {
    const ASSOC<const N: usize>: usize;
}

impl Trait for () {
    const ASSOC<const N: usize>: usize = core::gca!(N);
}

fn foo<const N: usize>() {
    let a: [(); <() as Trait>::ASSOC::<N>]
        = [(); N];
}

    با #![feature(gca_macroless_items)] می‌توانید کد بالا را بدون ماکرو بنویسید:

    #![allow(incomplete_features)]
#![feature(
    gca_min_const_items,
    gca_const_items,
    gca_macroless_args,
    gca_macroless_items,
    generic_const_items,
)]

trait Trait {
    const ASSOC<const N: usize>: usize;
}

impl Trait for () {
    const ASSOC<const N: usize>: usize = N;
}

fn foo<const N: usize>() {
    let a: [(); <() as Trait>::ASSOC::<N>]
        = [(); N];
}
