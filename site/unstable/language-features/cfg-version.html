    cfg_version

    issue ردیابی این قابلیت: #64796

    قابلیت cfg_version این امکان را فراهم می‌کند که بسته به نسخه‌ی کامپایلر، کد متفاوتی اجرا شود. اگر نسخه‌ی کامپایلر بزرگ‌تر یا مساوی نسخه‌ی مشخص‌شده باشد، مقدار true برمی‌گرداند.

    مثال‌ها

    #![feature(cfg_version)]

#[cfg(version("1.42"))] // 1.42 and above
fn a() {
    // ...
}

#[cfg(not(version("1.42")))] // 1.41 and below
fn a() {
    // ...
}

fn b() {
    if cfg!(version("1.42")) {
        // ...
    } else {
        // ...
    }
}
