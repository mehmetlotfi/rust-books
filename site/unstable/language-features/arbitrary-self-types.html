    arbitrary_self_types

    issue ردیابی این قابلیت: #44874

    اجازه می‌دهد هر نوعی که core::ops::Receiver<Target=T> را پیاده‌سازی می‌کند، به‌عنوان نوعِ self در متدی که به T تعلق دارد به کار رود.

    برای مثال،

    #![feature(arbitrary_self_types)]

struct A;

impl A {
    fn f(self: SmartPtr<Self>) -> i32 { 1 }  // note self type
}

struct SmartPtr<T>(T);

impl<T> core::ops::Receiver for SmartPtr<T> {
    type Target = T;
}

fn main() {
    let smart_ptr = SmartPtr(A);
    assert_eq!(smart_ptr.f(), 1);
}

    صفت Receiver یک پیاده‌سازی فراگیر (blanket) برای همهٔ T: Deref دارد، بنابراین در واقع چیزهایی از این دست هم کار می‌کنند:

    #![feature(arbitrary_self_types)]

use std::rc::Rc;

struct A;

impl A {
    fn f(self: Rc<Self>) -> i32 { 1 } // Rc implements Deref
}

fn main() {
    let smart_ptr = Rc::new(A);
    assert_eq!(smart_ptr.f(), 1);
}

    جالب اینکه این کد حتی بدون قابلیت arbitrary_self_types هم کار می‌کند

    • اما دلیلش این است که برخی نوع‌ها، از جمله Rc، در عمل به‌صورت ثابت در کامپایلر تعریف شده‌اند. (عبارت «تعریف ثابت» کاملاً دقیق نیست؛ آن‌ها از یک lang-item به نام LegacyReceiver استفاده می‌کنند تا این ویژگی خاص بودنشان را نشان دهند.) با قابلیت arbitrary_self_types، این خاص بودن از میان می‌رود و اشاره‌گرهای هوشمند سفارشی هم می‌توانند به همان قابلیت برسند.

    تغییرات در جست‌وجوی متد

    جست‌وجوی متد پیش‌تر به این صورت کار می‌کرد که زنجیرهٔ Deref را گام‌به‌گام طی می‌کرد و سپس فهرست گام‌های حاصل را به دو شکل متفاوت به کار می‌برد:

    • برای شناسایی نوع‌هایی که ممکن است از طریق بلوک‌های impl خود (متدهای ذاتی یا inherent) یا از طریق صفت‌ها متد فراهم کنند
    • برای شناسایی نوع‌هایی که گیرندهٔ متد (a در مثال‌های بالا) می‌تواند به آن‌ها تبدیل شود.

    با این قابلیت، این فهرست‌ها به‌جای آن از طی کردن زنجیرهٔ Receiver ساخته می‌شوند. با این حال، یادداشتی نگه داشته می‌شود که نشان می‌دهد آیا آن نوع از طریق زنجیرهٔ Deref هم قابل دسترسی است یا نه.

    زنجیرهٔ کامل (از طریق گام‌های Receiver) برای هدف نخست (شناسایی بلوک‌های impl و صفت‌های مرتبط) به کار می‌رود؛ در حالی که فهرست کوتاه‌تر (قابل دسترسی از طریق Deref) برای هدف دوم استفاده می‌شود. دلیل این است که برای تبدیل هدف متد (a در a.b()) به نوع self، Rust ممکن است نیاز داشته باشد بتواند از Deref::deref استفاده کند. بنابراین تبدیل نوع‌ها فقط تا انتهای زنجیرهٔ Deref پیش می‌رود، در حالی که زنجیرهٔ طولانی‌تر Receiver را می‌توان برای کاوش جاهای بیشتری به کار برد که متدهای مفید ممکن است در آن‌ها باشند.

    نوع‌های مناسب برای استفاده به‌عنوان اشاره‌گر هوشمند

    این قابلیت امکان ساخت اشاره‌گرهای هوشمند سفارشی را فراهم می‌کند - مثلاً معادل خودتان برای Rc یا Box با هر قابلیتی که بخواهید. این اشاره‌گرهای هوشمند می‌توانند یا Deref را پیاده‌سازی کنند (اگر ساختن ارجاع به شیء مورد اشاره (referent) ایمن باشد) یا Receiver را (اگر ایمن نباشد).

    در هر صورت، نوع‌های اشاره‌گر هوشمند باید عمدتاً از داشتن متد پرهیز کنند. فراخوانی متد روی یک اشاره‌گر هوشمند باعث ابهام می‌شود که قصدتان متدی روی خود اشاره‌گر است یا روی شیء مورد اشاره.

    بنابراین بهترین روش این است که قابلیت‌های اشاره‌گر هوشمند را به‌جای متد در تابع‌های وابسته (associated function) قرار دهید - این همان کاری است که در همهٔ نوع‌های اشاره‌گر هوشمند کتابخانهٔ استاندارد Rust که Receiver را پیاده‌سازی می‌کنند انجام شده است.

    اگر تصمیم بگیرید به نوع اشاره‌گر هوشمند خود متدی اضافه کنید، کاربرانتان ممکن است با خطاهای ناشی از deshadowing روبه‌رو شوند، که در بخش بعدی توضیح داده شده است.

    پرهیز از سایه‌اندازی (shadowing)

    چه با این قابلیت و چه بدون آن، Rust وقتی دو متد کاندیدا پیدا کند خطا صادر می‌کند، مانند این مثال:

    use std::pin::Pin;
use std::pin::pin;

struct A;

impl A {
    fn get_ref(self: Pin<&A>) {}
}

fn main() {
    let pinned_a: Pin<&A> = pin!(A).as_ref();
    let pinned_a: Pin<&A> = pinned_a.as_ref();
    pinned_a.get_ref(); // error[E0034]: multiple applicable items in scope
}

    (به همین دلیل اشاره‌گرهای هوشمند Rust عمدتاً با دقت طراحی شده‌اند تا اصلاً متد نداشته باشند، و نباید در آینده متد جدیدی به آن‌ها اضافه شود.)

    با arbitrary_self_types، مراقبیم که چند نوع دیگر از تعارض را هم تشخیص دهیم:

    #![feature(arbitrary_self_types)]

use std::pin::Pin;
use std::pin::pin;

struct A;

impl A {
    fn get_ref(self: &Pin<&A>) {}  // note &Pin
}

fn main() {
    let pinned_a: Pin<&mut A> = pin!(A);
    let pinned_a: Pin<&A> = pinned_a.as_ref();
    pinned_a.get_ref();
}

    هدف این است که جلوی موردی گرفته شود که نوع درونی (شیء مورد اشاره) متدی با نامی مشخص دارد که اشاره‌گر هوشمند را به‌صورت ارجاع می‌گیرد، و سپس پیاده‌ساز اشاره‌گر هوشمند متد مشابهی اضافه می‌کند که self را به‌صورت مقدار می‌گیرد. همان‌طور که در بخش قبل گفته شد، گزینهٔ ایمن این است که اصلاً به اشاره‌گرهای هوشمند متد اضافه نکنید؛ آن‌وقت این خطاها رخ نمی‌دهند.
