    cfg_sanitize

    issue ردیابی این قابلیت: #39699

    قابلیت cfg_sanitize این امکان را فراهم می‌کند که بسته به فعال‌بودن یا نبودن یک sanitizer خاص، کد متفاوتی اجرا شود.

    مثال‌ها

    #![feature(cfg_sanitize)]

#[cfg(sanitize = "thread")]
fn a() {
    // ...
}

#[cfg(not(sanitize = "thread"))]
fn a() {
    // ...
}

fn b() {
    if cfg!(sanitize = "leak") {
        // ...
    } else {
        // ...
    }
}
