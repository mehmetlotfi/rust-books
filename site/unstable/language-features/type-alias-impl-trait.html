    type_alias_impl_trait

    issue ردیابی این ویژگی: #63063

    این ویژگی را با trait_alias یا impl_trait_in_assoc_type اشتباه نگیرید.

    impl Trait چیست؟

    impl Trait در موقعیت مقدار بازگشتی برای اعلام نوع‌هایی مفید است که با صفت‌ها محدود شده‌اند، اما نوع مشخص آن‌ها باید پنهان بماند:

    use std::fmt::Debug;

fn new() -> impl Debug {
    42
}

fn main() {
    let thing = new();
    // What actually is a `thing`?
    // No idea but we know it implements `Debug`, so we can debug print it
    println!("{thing:?}");
}

    برای اطلاعات بیشتر دربارهٔ impl Trait در موقعیت مقدار بازگشتی، به مرجع مراجعه کنید.

    type_alias_impl_trait

    با این حال، ممکن است بخواهیم از impl Trait در چند جا استفاده کنیم اما در همه‌جا واقعاً همان نوع مشخص را به کار ببریم و آن را پنهان نگه داریم. این کار می‌تواند در کتابخانه‌هایی مفید باشد که می‌خواهید جزئیات پیاده‌سازی را در آن‌ها پنهان کنید.

    attribute به نام #[define_opaque] باید برای فهرست کردن صریح آیتم‌های مبهم (opaque) که توسط آیتمِ حامل آن محدود می‌شوند استفاده شود.

    #![feature(type_alias_impl_trait)]
#![allow(unused_variables, dead_code)]
trait Trait {}

struct MyType;

impl Trait for MyType {}

type Alias = impl Trait;

#[define_opaque(Alias)] // To constrain the type alias to `MyType`
fn new() -> Alias {
    MyType
}

#[define_opaque(Alias)] // So we can name the concrete type inside this item
fn main() {
    let thing: MyType = new();
}

// It can be a part of a struct too
struct HaveAlias {
    stuff: String,
    thing: Alias,
}

    در این مثال، نوع مشخصی که Alias به آن اشاره دارد تضمین شده است که در هر جایی که Alias ظاهر شود یکسان است.

    در ابتدا این ویژگی نام‌های مستعار نوع را به عنوان نوع وابسته (associated type) یک trait نیز شامل می‌شد. در #110237 این بخش جدا شد و به impl_trait_in_assoc_type منتقل گردید.

    type_alias_impl_trait در موقعیت آرگومان.

    توجه کنید که استفاده از Alias به عنوان نوع آرگومان با impl Trait در موقعیت آرگومان یکسان نیست، زیرا Alias به یک نوع یکتا اشاره می‌کند، در حالی که نوع مشخص برای impl Trait در موقعیت آرگومان را فراخواننده انتخاب می‌کند.

    #![feature(type_alias_impl_trait)]
#![allow(unused_variables)]
pub mod x {
pub trait Trait {}

struct MyType;

impl Trait for MyType {}

pub type Alias = impl Trait;

#[define_opaque(Alias)]
pub fn new() -> Alias {
    MyType
}
}
use x::*;
// this...
pub fn take_alias(x: Alias) {
    // ...
}

// ...is *not* the same as
pub fn take_impl(x: impl Trait) {
    // ...
}
fn main(){}
    #![feature(type_alias_impl_trait)]
#![allow(unused_variables)]
pub mod x {
pub trait Trait {}

struct MyType;

impl Trait for MyType {}

pub type Alias = impl Trait;

#[define_opaque(Alias)]
pub fn new() -> Alias {
    MyType
}
}
use x::*;
pub fn take_alias(x: Alias) {
    // ...
}

pub fn take_impl(x: impl Trait) {
   // ...
}

// a user's crate using the trait and type alias
struct UserType;
impl Trait for UserType {}

fn main(){
let x = UserType;
take_alias(x);
// ERROR expected opaque type, found `UserType`
// this function *actually* takes a `MyType` as is constrained in `new`

let x = UserType;
take_impl(x);
// OK

let x = new();
take_alias(x);
// OK

let x = new();
take_impl(x);
// OK
}

    توجه کنید که کاربر نمی‌تواند از #[define_opaque(Alias)] برای مشخص کردن (reify) نوع مبهم استفاده کند، چون فقط crateای که نام مستعار نوع در آن اعلام شده این اجازه را دارد. اما اگر این اتفاق در همان crate می‌افتاد و نوع مبهم مشخص می‌شد، با خطای آشنای «expected MyType, got UserType» روبه‌رو می‌شدند.
