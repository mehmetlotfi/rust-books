    بازصادرها (Re-exports)

    بیایید با توضیح این‌که بازصادر (re-export) چیست شروع کنیم. برای این کار از مثالی استفاده می‌کنیم که در آن کتابخانه‌ای (با نام lib) می‌نویسیم که نوع‌هایش میان چند زیرماژول پخش شده‌اند:

    pub mod sub_module1 {
    pub struct Foo;
}
pub mod sub_module2 {
    pub struct AnotherFoo;
}

    کاربران می‌توانند آن‌ها را این‌گونه import کنند:

    use lib::sub_module1::Foo;
use lib::sub_module2::AnotherFoo;

    اما اگر بخواهید نوع‌ها مستقیماً در ریشهٔ crate در دسترس باشند، یا نخواهیم ماژول‌ها برای کاربران قابل مشاهده باشند، چه؟ اینجاست که بازصادرها وارد می‌شوند:

    // `sub_module1` and `sub_module2` are not visible outside.
mod sub_module1 {
    pub struct Foo;
}
mod sub_module2 {
    pub struct AnotherFoo;
}
// We re-export both types:
pub use crate::sub_module1::Foo;
pub use crate::sub_module2::AnotherFoo;

    و اکنون کاربران می‌توانند این کار را بکنند:

    use lib::{Foo, AnotherFoo};

    و چون هر دو sub_module1 و sub_module2 خصوصی هستند، کاربران نمی‌توانند آن‌ها را import کنند.

    نکتهٔ جالب این است که مستندات تولیدشده برای این crate هر دو Foo و AnotherFoo را مستقیماً در ریشهٔ crate نشان می‌دهد، یعنی آن‌ها درون‌خط (inline) شده‌اند. چند قاعده وجود دارد که تعیین می‌کند آیا یک آیتم بازصادرشده درون‌خط می‌شود یا نه.

    قواعد درون‌خط شدن

    اگر یک آیتم عمومی از یک ماژول خصوصی آمده باشد، درون‌خط می‌شود:

    mod private_module {
    pub struct Public;
}
pub mod public_mod {
    // `Public` will inlined here since `private_module` is private.
    pub use super::private_module::Public;
}
// `Public` will not be inlined here since `public_mod` is public.
pub use self::public_mod::Public;

    همچنین، اگر آیتمی #[doc(hidden)] را از هر یک از اجدادش به ارث ببرد، درون‌خط می‌شود:

    #[doc(hidden)]
pub mod public_mod {
    pub struct Public;
}
// `Public` be inlined since its parent (`public_mod`) has `#[doc(hidden)]`.
pub use self::public_mod::Public;

    اگر آیتمی #[doc(hidden)] داشته باشد، درون‌خط نمی‌شود (و در مستندات تولیدشده نیز قابل مشاهده نیست):

    // This struct won't be visible.
#[doc(hidden)]
pub struct Hidden;

// This re-export won't be visible.
pub use self::Hidden as InlinedHidden;

    با این حال، اگر باز هم می‌خواهید خود بازصادر قابل مشاهده باشد، می‌توانید attribute #[doc(inline)] را روی آن بگذارید:

    // This struct won't be visible.
#[doc(hidden)]
pub struct Hidden;

#[doc(inline)]
pub use self::Hidden as InlinedHidden;

    در این حالت، در مستندات تولیدشده pub use self::Hidden as InlinedHidden; را خواهید داشت اما هیچ پیوندی به آیتم Hidden وجود نخواهد داشت.

    پس به #[doc(hidden)] برمی‌گردیم: اگر چند بازصادر دارید و برخی از آن‌ها #[doc(hidden)] دارند، فقط همین‌ها در مستندات نمایش داده نمی‌شوند:

    mod private_mod {
    /// First
    pub struct InPrivate;
}

/// Second
#[doc(hidden)]
pub use self::private_mod::InPrivate as Hidden;
/// Third
pub use self::Hidden as Visible;

    در این حالت، InPrivate با نام Visible درون‌خط می‌شود. با این حال، مستندات آن First Third خواهد بود و نه First Second Third، زیرا بازصادری که Second مستندات آن است، #[doc(hidden)] دارد و در نتیجه همهٔ attributeهای آن نادیده گرفته می‌شوند.

    درون‌خط کردن با #[doc(inline)]

    اگر می‌خواهید یک آیتم را مجبور به درون‌خط شدن کنید، می‌توانید از attribute #[doc(inline)] استفاده کنید:

    pub mod public_mod {
    pub struct Public;
}
#[doc(inline)]
pub use self::public_mod::Public;

    با این کد، حتی با این‌که public_mod::Public عمومی است و در مستندات حضور دارد، نوع Public هم در ریشهٔ crate و هم در ماژول public_mod حضور خواهد داشت.

    جلوگیری از درون‌خط شدن با #[doc(no_inline)]

    در نقطهٔ مقابل attribute #[doc(inline)]، اگر می‌خواهید از درون‌خط شدن یک آیتم جلوگیری کنید، می‌توانید از #[doc(no_inline)] استفاده کنید:

    mod private_mod {
    pub struct Public;
}
#[doc(no_inline)]
pub use self::private_mod::Public;

    در مستندات تولیدشده، یک بازصادر را در ریشهٔ crate خواهید دید و نه خود نوع را مستقیماً.

    Attributeها

    وقتی یک آیتم درون‌خط می‌شود، توضیحات مستنداتی و بیشتر attributeهای آن نیز همراه با آن درون‌خط می‌شوند:

    mod private_mod {
    /// First
    #[cfg(a)]
    pub struct InPrivate;
    /// Second
    #[cfg(b)]
    pub use self::InPrivate as Second;
}

/// Third
#[doc(inline)]
#[cfg(c)]
pub use self::private_mod::Second as Visible;

    در این حالت، Visible مستنداتی برابر First Second Third خواهد داشت و همچنین cfg آن این‌ها خواهد بود: #[cfg(a)]، #[cfg(b)] و #[cfg(c)].

    پیوندهای درون‌مستندی نسبت به جایی که توضیح مستنداتی تعریف شده است حل می‌شوند.

    با این حال، چند attribute وجود دارند که درون‌خط نمی‌شوند:

    • #[doc(alias="")]
    • #[doc(inline)]
    • #[doc(no_inline)]
    • #[doc(hidden)] (چون خود بازصادر و attributeهای آن نادیده گرفته می‌شوند).

    همهٔ attributeهای دیگر هنگام درون‌خط شدن به ارث برده می‌شوند، تا مستندات با رفتاری مطابقت داشته باشد که آیتم درون‌خط‌شده مستقیماً در جایی که نمایش داده می‌شود تعریف شده بود.

    این قواعد در صورتی هم اعمال می‌شوند که آیتم با یک بازصادر glob درون‌خط شود:

    mod private_mod {
    /// First
    #[cfg(a)]
    pub struct InPrivate;
}

#[cfg(c)]
pub use self::private_mod::*;

    در غیر این صورت، attributeهای نمایش‌داده‌شده از آیتم بازصادرشده خواهند بود و attributeهای روی خود بازصادر نادیده گرفته می‌شوند:

    mod private_mod {
    /// First
    #[cfg(a)]
    pub struct InPrivate;
}

#[cfg(c)]
pub use self::private_mod::InPrivate;

    در حالت بالا، cfg(c) در مستندات نمایش داده نمی‌شود.
