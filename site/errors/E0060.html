    توابع خارجی C مجاز هستند variadic باشند. با این حال، یک تابع variadic حداقل تعداد مشخصی آرگومان می‌گیرد. برای مثال، تابع variadic به نام printf در زبان C را در نظر بگیرید:

    use std::os::raw::{c_char, c_int};

extern "C" {
    fn printf(_: *const c_char, ...) -> c_int;
}

unsafe { printf(); } // error!

    با این اعلان، تابع باید دست‌کم با یک آرگومان فراخوانی شود، بنابراین صرفاً فراخوانی printf() نامعتبر است. اما استفاده‌های زیر مجاز هستند:

    use std::os::raw::{c_char, c_int};
#[cfg_attr(all(windows, target_env = "msvc"),
           link(name = "legacy_stdio_definitions",
                kind = "static", modifiers = "-bundle"))]
extern "C" { fn printf(_: *const c_char, ...) -> c_int; }
fn main() {
unsafe {
    printf(c"test\n".as_ptr());

    printf(c"number = %d\n".as_ptr(), 3);

    printf(c"%d, %d\n".as_ptr(), 10, 5);
}
}
