    ممنوعیت ارجاع به static mut

    خلاصه

    • سطح lint با نام static_mut_refs اکنون به‌طور پیش‌فرض deny است. این lint بررسی می‌کند که آیا یک ارجاع مشترک یا تغییرپذیر به یک static mut گرفته شده است یا نه.

    جزئیات

    lint با نام static_mut_refs گرفتن ارجاع به یک static mut را شناسایی می‌کند. در ویرایش ۲۰۲۴ این lint به‌طور پیش‌فرض deny شده است تا تأکید شود که باید از ساختن چنین ارجاع‌هایی پرهیز کنید.

    static mut X: i32 = 23;
static mut Y: i32 = 24;

unsafe {
    let y = &X;             // ERROR: shared reference to mutable static
    let ref x = X;          // ERROR: shared reference to mutable static
    let (x, y) = (&X, &Y);  // ERROR: shared reference to mutable static
}

    صرفِ گرفتن چنین ارجاعی، که قاعدهٔ «قابل‌تغییر یا هم‌نام، نه هر دو» (mutability XOR aliasing) در Rust را نقض می‌کند، همیشه بلافاصله رفتار تعریف‌نشده به شمار می‌رفته است، حتی اگر از ارجاع هرگز خوانده یا در آن نوشته نشود. افزون بر این، برقرار نگه‌داشتن این قاعده برای یک static mut مستلزم استدلال دربارهٔ کد شما در مقیاس سراسری است، که در حضور بازورود (reentrancy) و/یا چندنخی (multithreading) می‌تواند به‌ویژه دشوار باشد.

    توجه کنید که حالت‌هایی وجود دارد که در آن‌ها ارجاع‌های ضمنی بدون عملگر & مشهود به‌طور خودکار ساخته می‌شوند. برای مثال، این موقعیت‌ها نیز lint را فعال می‌کنند:

    static mut NUMS: &[u8; 3] = &[0, 1, 2];

unsafe {
    println!("{NUMS:?}");   // ERROR: shared reference to mutable static
    let n = NUMS.len();     // ERROR: shared reference to mutable static
}

    جایگزین‌ها

    تا جایی که ممکن است، اکیداً توصیه می‌شود به‌جای آن از یک static تغییرناپذیر از نوعی استفاده کنید که تغییرپذیری درونی (interior mutability) را پشت یک انتزاع با استدلال موضعی فراهم می‌کند (که پیچیدگی تضمین رعایت قاعدهٔ «قابل‌تغییر یا هم‌نام، نه هر دو» در Rust را به‌شدت کاهش می‌دهد).

    در موقعیت‌هایی که هیچ انتزاع با استدلال موضعی ممکن نیست و ناچارید همچنان دربارهٔ دسترسی‌ها به متغیر static خود به‌صورت سراسری استدلال کنید، اکنون باید از اشاره‌گرهای خام (raw pointer) استفاده کنید؛ مثلاً اشاره‌گرهایی که از عملگرهای &raw const یا &raw mut به دست می‌آیند. اگر به‌جای گرفتن مستقیم ارجاع، ابتدا یک اشاره‌گر خام بگیرید، (الزامات ایمنی) دسترسی‌ها از طریق آن اشاره‌گر برای توسعه‌دهندگان unsafe آشناتر خواهد بود و می‌توان آن‌ها را به بعد موکول کرد یا به بخش‌های کوچک‌تری از کد محدود نمود.

    توجه کنید که مثال‌های زیر صرفاً نمونه‌های توضیحی هستند و قرار نیست پیاده‌سازی‌های کامل باشند. آن‌ها را عیناً کپی نکنید. جزئیاتی مربوط به وضعیت خاص شما وجود دارد که ممکن است برای تطبیق با نیازهایتان به تغییراتی نیاز داشته باشد. هدف از این مثال‌ها کمک به شماست تا راه‌های مختلف برخورد با مسئله‌تان را ببینید.

    توصیه می‌شود مستندات نوع‌های مشخص در کتابخانهٔ استاندارد، مرجع رفتار تعریف‌نشده، و Rustonomicon را مطالعه کنید و اگر پرسشی دارید، در یکی از انجمن‌های Rust مانند انجمن کاربران مطرح کنید.

    از متغیرهای سراسری استفاده نکنید

    این احتمالاً چیزی است که از قبل می‌دانید، اما در صورت امکان بهتر است از حالت سراسری قابل‌تغییر پرهیز کنید. البته گاهی این کار کمی دست‌وپاگیرتر یا دشوارتر است، به‌ویژه اگر لازم باشد یک ارجاع قابل‌تغییر را میان توابع بسیاری دست‌به‌دست کنید.

    اتمی‌ها

    نوع‌های اتمی اعداد صحیح، اشاره‌گرها و مقادیر بولی‌ای را فراهم می‌کنند که می‌توان آن‌ها را در یک static (بدون mut) به کار برد.

    use std::sync::atomic::Ordering;
use std::sync::atomic::AtomicU64;

// Change from this:
//   static mut COUNTER: u64 = 0;
// to this:
static COUNTER: AtomicU64 = AtomicU64::new(0);

fn main() {
    // Be sure to analyze your use case to determine the correct Ordering to use.
    COUNTER.fetch_add(1, Ordering::Relaxed);
}

    Mutex یا RwLock

    وقتی نوع شما پیچیده‌تر از یک نوع اتمی است، استفاده از Mutex یا RwLock را برای تضمین دسترسی درست به مقدار سراسری در نظر بگیرید.

    use std::sync::Mutex;
use std::collections::VecDeque;

// Change from this:
//     static mut QUEUE: VecDeque<String> = VecDeque::new();
// to this:
static QUEUE: Mutex<VecDeque<String>> = Mutex::new(VecDeque::new());

fn main() {
    QUEUE.lock().unwrap().push_back(String::from("abc"));
    let first = QUEUE.lock().unwrap().pop_front();
}

    OnceLock یا LazyLock

    اگر از static mut استفاده می‌کنید چون باید مقداردهی اولیه‌ای یک‌باره انجام دهید که نمی‌تواند const باشد، می‌توانید به‌جای آن از OnceLock یا LazyLock استفاده کنید.

    use std::sync::LazyLock;

struct GlobalState;

impl GlobalState {
    fn new() -> GlobalState {
        GlobalState
    }
    fn example(&self) {}
}

// Instead of some temporary or uninitialized type like:
//     static mut STATE: Option<GlobalState> = None;
// use this instead:
static STATE: LazyLock<GlobalState> = LazyLock::new(|| {
    GlobalState::new()
});

fn main() {
    STATE.example();
}

    OnceLock شبیه LazyLock است، اما اگر لازم باشد اطلاعاتی را به سازنده (constructor) بدهید می‌توان از آن استفاده کرد؛ این ویژگی با نقاط مقداردهی اولیهٔ یگانه (مانند main) یا زمانی که ورودی‌ها در هر جایی که به مقدار سراسری دسترسی دارید در دسترس‌اند، به‌خوبی کار می‌کند.

    use std::sync::OnceLock;

struct GlobalState;

impl GlobalState {
    fn new(verbose: bool) -> GlobalState {
        GlobalState
    }
    fn example(&self) {}
}

struct Args {
    verbose: bool
}
fn parse_arguments() -> Args {
    Args { verbose: true }
}

static STATE: OnceLock<GlobalState> = OnceLock::new();

fn main() {
    let args = parse_arguments();
    let state = GlobalState::new(args.verbose);
    let _ = STATE.set(state);
    // ...
    STATE.get().unwrap().example();
}

    مقداردهی اولیهٔ یک‌باره در no_std

    این مثال از این جهت شبیه OnceLock است که مقداردهی اولیهٔ یک‌بارهٔ یک مقدار سراسری را فراهم می‌کند، اما به std نیاز ندارد، که در یک محیط no_std مفید است. با فرض اینکه هدف (target) شما از عملیات اتمی پشتیبانی می‌کند، می‌توانید از یک مقدار اتمی برای بررسی مقداردهی اولیهٔ مقدار سراسری استفاده کنید. این الگو ممکن است چیزی شبیه این باشد:

    use core::sync::atomic::AtomicUsize;
use core::sync::atomic::Ordering;

struct Args {
    verbose: bool,
}
fn parse_arguments() -> Args {
    Args { verbose: true }
}

struct GlobalState {
    verbose: bool,
}

impl GlobalState {
    const fn default() -> GlobalState {
        GlobalState { verbose: false }
    }
    fn new(verbose: bool) -> GlobalState {
        GlobalState { verbose }
    }
    fn example(&self) {}
}

const UNINITIALIZED: usize = 0;
const INITIALIZING: usize = 1;
const INITIALIZED: usize = 2;

static STATE_INITIALIZED: AtomicUsize = AtomicUsize::new(UNINITIALIZED);
static mut STATE: GlobalState = GlobalState::default();

fn set_global_state(state: GlobalState) {
    if STATE_INITIALIZED
        .compare_exchange(
            UNINITIALIZED,
            INITIALIZING,
            Ordering::SeqCst,
            Ordering::SeqCst,
        )
        .is_ok()
    {
        // SAFETY: The reads and writes to STATE are guarded with the INITIALIZED guard.
        unsafe {
            STATE = state;
        }
        STATE_INITIALIZED.store(INITIALIZED, Ordering::SeqCst);
    } else {
        panic!("already initialized, or concurrent initialization");
    }
}

fn get_state() -> &'static GlobalState {
    if STATE_INITIALIZED.load(Ordering::Acquire) != INITIALIZED {
        panic!("not initialized");
    } else {
        // SAFETY: Mutable access is not possible after state has been initialized.
        unsafe { &*&raw const STATE }
    }
}

fn main() {
    let args = parse_arguments();
    let state = GlobalState::new(args.verbose);
    set_global_state(state);
    // ...
    let state = get_state();
    state.example();
}

    این مثال فرض می‌کند که می‌توانید پیش از مقداردهی اولیه یک مقدار پیش‌فرض در static قرار دهید (در این مثال، سازندهٔ default از نوع const). اگر این ممکن نیست، استفاده از MaybeUninit، یا توزیع پویای trait (dynamic trait dispatch) با یک نوع ساختگی که یک trait را پیاده‌سازی می‌کند، یا هر رویکرد دیگری برای داشتن یک جانگهدار پیش‌فرض را در نظر بگیرید.

    crateهایی نیز توسط جامعه فراهم شده‌اند که مقداردهی اولیهٔ یک‌بارهٔ مشابهی ارائه می‌دهند، مانند crate با نام static-cell (که با استفاده از portable-atomic از هدف‌هایی که عملیات اتمی ندارند نیز پشتیبانی می‌کند).

    اشاره‌گرهای خام

    در برخی موارد می‌توانید همچنان از static mut استفاده کنید، اما از ساختن ارجاع پرهیز کنید. برای مثال، اگر فقط لازم است اشاره‌گرهای خام را به یک کتابخانهٔ C بدهید، ارجاع میانی نسازید. در عوض می‌توانید از عملگرهای قرض خام استفاده کنید، مانند مثال زیر:

    #[repr(C)]
struct GlobalState {
    value: i32
}

impl GlobalState {
    const fn new() -> GlobalState {
        GlobalState { value: 0 }
    }
}

static mut STATE: GlobalState = GlobalState::new();

unsafe extern "C" {
    fn example_ffi(state: *mut GlobalState);
}

fn main() {
    unsafe {
        // Change from this:
        //     example_ffi(&mut STATE as *mut GlobalState);
        // to this:
        example_ffi(&raw mut STATE);
    }
}

    فقط توجه داشته باشید که همچنان باید محدودیت‌های هم‌نامی (aliasing) پیرامون اشاره‌گرهای قابل‌تغییر را رعایت کنید. این ممکن است مستلزم نوعی هم‌گام‌سازی درونی یا بیرونی، یا اثبات‌هایی دربارهٔ نحوهٔ استفاده از آن در نخ‌ها، مدیریت‌کننده‌های وقفه (interrupt handler) و بازورود باشد.

    UnsafeCell همراه با Sync

    UnsafeCell trait با نام Sync را پیاده‌سازی (impl) نمی‌کند، بنابراین نمی‌توان آن را در یک static به کار برد. می‌توانید پوششی (wrapper) دور UnsafeCell بسازید و یک پیاده‌سازی Sync به آن اضافه کنید تا برای پیاده‌سازی تغییرپذیری درونی در یک static قابل استفاده شود. این رویکرد زمانی مفید است که قفل‌های بیرونی یا تضمین‌های دیگری دارید که ناورداهای ایمنی (safety invariants) مورد نیاز اشاره‌گرهای تغییرپذیر را برقرار نگه می‌دارند.

    توجه کنید که این رویکرد تا حد زیادی همان مثال اشاره‌گرهای خام است. پوشش کمک می‌کند نحوهٔ استفادهٔ شما از نوع برجسته شود و بر الزامات ایمنی‌ای که باید مراقبشان باشید تمرکز شود. اما در غیر این صورت این دو تقریباً یکسان‌اند.

    use std::cell::UnsafeCell;

fn with_interrupts_disabled<T: Fn()>(f: T) {
    // A real example would disable interrupts.
    f();
}

#[repr(C)]
struct GlobalState {
    value: i32,
}

impl GlobalState {
    const fn new() -> GlobalState {
        GlobalState { value: 0 }
    }
}

#[repr(transparent)]
pub struct SyncUnsafeCell<T>(UnsafeCell<T>);

unsafe impl<T: Sync> Sync for SyncUnsafeCell<T> {}

static STATE: SyncUnsafeCell<GlobalState> = SyncUnsafeCell(UnsafeCell::new(GlobalState::new()));

fn set_value(value: i32) {
    with_interrupts_disabled(|| {
        let state = STATE.0.get();
        unsafe {
            // SAFETY: This value is only ever read in our interrupt handler,
            // and interrupts are disabled, and we only use this in one thread.
            (*state).value = value;
        }
    });
}

    کتابخانهٔ استاندارد گونه‌ای صرفاً nightly (ناپایدار) از UnsafeCell با نام SyncUnsafeCell دارد. مثال بالا نسخهٔ بسیار ساده‌شده‌ای از نوع کتابخانهٔ استاندارد را نشان می‌دهد، اما تقریباً به همان شیوه به کار می‌رود. این نوع می‌تواند جداسازی حتی بهتری فراهم کند، پس برای جزئیات بیشتر پیاده‌سازی آن را ببینید.

    این مثال شامل تابعی خیالی با نام with_interrupts_disabled است که نمونه‌ای از چیزی است که ممکن است در یک محیط نهفته (embedded) ببینید. برای مثال، crate با نام critical-section قابلیتی مشابه فراهم می‌کند که می‌توان از آن در یک محیط نهفته استفاده کرد.

    ارجاع‌های ایمن

    در برخی موارد ممکن است ساختن ارجاع به یک static mut ایمن باشد. تمام فلسفهٔ lint با نام static_mut_refs این است که انجام درست این کار بسیار دشوار است! با این حال، این به معنای ناممکن بودن آن نیست. اگر در وضعیتی هستید که می‌توانید رعایت الزامات هم‌نامی (aliasing) را تضمین کنید، مثلاً تضمین کنید که دامنهٔ static محدود است (فقط در یک ماژول یا تابع کوچک استفاده می‌شود)، هم‌گام‌سازی درونی یا بیرونی دارد، و وقفه‌ها، بازورود، ایمنی در برابر panic، مدیریت‌کننده‌های drop و مانند آن را در نظر می‌گیرد، آنگاه گرفتن ارجاع ممکن است اشکالی نداشته باشد.

    برای این کار دو رویکرد دارید. یا می‌توانید lint با نام static_mut_refs را مجاز کنید (ترجیحاً تا حد ممکن در محدودهٔ باریک)، یا اشاره‌گرهای خام را به ارجاع تبدیل کنید، مانند &mut *&raw mut MY_STATIC.

    ارجاع‌های کوتاه‌عمر

    اگر ناچارید ارجاعی به یک static mut بسازید، توصیه می‌شود مدتی را که آن ارجاع وجود دارد به حداقل برسانید. از پنهان کردن ارجاع در جایی یا زنده نگه‌داشتن آن در بخش بزرگی از کد پرهیز کنید. کوتاه‌عمر نگه‌داشتن آن به ممیزی و بررسی اینکه دسترسی انحصاری در تمام مدت حفظ می‌شود کمک می‌کند. اشاره‌گرها باید واحد پیش‌فرض شما باشند و فقط در صورت نیاز مطلق و به‌صورت درخواستی اشاره‌گر را به ارجاع تبدیل کنید.

    مهاجرت

    مهاجرت خودکاری برای اصلاح این ارجاع‌ها به static mut وجود ندارد. برای پرهیز از رفتار تعریف‌نشده باید کد خود را طبق رویکردهای توصیه‌شده در بخش جایگزین‌ها بازنویسی کنید تا از روش دیگری استفاده کند.
