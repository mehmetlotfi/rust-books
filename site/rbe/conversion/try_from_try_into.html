    TryFrom و TryInto

    همانند From و Into، صفت‌های TryFrom و TryInto نیز صفت‌هایی عمومی برای تبدیل میان نوع‌ها هستند. برخلاف From/Into، صفت‌های TryFrom/TryInto برای تبدیل‌هایی به کار می‌روند که ممکن است شکست بخورند و به همین دلیل Result‌ها را بازمی‌گردانند.

    use std::convert::TryFrom;
use std::convert::TryInto;

#[derive(Debug, PartialEq)]
struct EvenNumber(i32);

impl TryFrom<i32> for EvenNumber {
    type Error = ();

    fn try_from(value: i32) -> Result<Self, Self::Error> {
        if value % 2 == 0 {
            Ok(EvenNumber(value))
        } else {
            Err(())
        }
    }
}

fn main() {
    // TryFrom

    assert_eq!(EvenNumber::try_from(8), Ok(EvenNumber(8)));
    assert_eq!(EvenNumber::try_from(5), Err(()));

    // TryInto

    let result: Result<EvenNumber, ()> = 8i32.try_into();
    assert_eq!(result, Ok(EvenNumber(8)));
    let result: Result<EvenNumber, ()> = 5i32.try_into();
    assert_eq!(result, Err(()));
}
