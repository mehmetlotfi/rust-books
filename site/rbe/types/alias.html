    نام مستعار

    با دستور type می‌توان نام تازه‌ای به یک نوع موجود داد. نام نوع‌ها باید از شیوهٔ UpperCamelCase پیروی کند؛ در غیر این صورت کامپایلر هشدار می‌دهد. نوع‌های اولیه مانند usize و f32 از این قاعده مستثنا هستند.

    // `NanoSecond`, `Inch`, and `U64` are new names for `u64`.
type NanoSecond = u64;
type Inch = u64;
type U64 = u64;

fn main() {
    // `NanoSecond` = `Inch` = `U64` = `u64`.
    let nanoseconds: NanoSecond = 5 as u64;
    let inches: Inch = 2 as U64;

    // Note that type aliases *don't* provide any extra type safety, because
    // aliases are *not* new types
    println!("{} nanoseconds + {} inches = {} unit?",
             nanoseconds,
             inches,
             nanoseconds + inches);
}

    کاربرد اصلی نام‌های مستعار کاهش کدهای تکراری است؛ برای نمونه، نوع io::Result<T> نام مستعاری برای Result<T, io::Error> است.
