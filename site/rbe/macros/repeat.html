    تکرار (Repeat)

    ماکروها می‌توانند در فهرست آرگومان‌ها از + برای نشان دادن تکرار دست‌کم یک‌بارهٔ آرگومان و از * برای نشان دادن صفر بار یا بیشتر استفاده کنند.

    در نمونهٔ زیر، قرار دادن تطبیق‌دهنده درون $(...),+ با یک یا چند عبارت جداشده با ویرگول تطبیق می‌کند. همچنین توجه کنید که نقطه‌ویرگول در حالت آخر اختیاری است.

    // `find_min!` will calculate the minimum of any number of arguments.
macro_rules! find_min {
    // Base case:
    ($x:expr) => ($x);
    // `$x` followed by at least one `$y,`
    ($x:expr, $($y:expr),+) => (
        // Call `find_min!` on the tail `$y`
        std::cmp::min($x, find_min!($($y),+))
    )
}

fn main() {
    println!("{}", find_min!(1));
    println!("{}", find_min!(1 + 2, 2));
    println!("{}", find_min!(5, 2 * 3, 4));
}
