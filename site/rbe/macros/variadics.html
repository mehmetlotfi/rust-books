    رابط‌های با تعداد آرگومان متغیر (Variadic Interfaces)

    یک رابط با تعداد آرگومان متغیر هر تعداد آرگومان می‌پذیرد. برای مثال، println! بسته به رشتهٔ قالب می‌تواند شمار دلخواهی آرگومان دریافت کند.

    می‌توانیم ماکروی calculate! بخش قبل را گسترش دهیم تا تعداد آرگومان متغیر بپذیرد:

    macro_rules! calculate {
    // The pattern for a single `eval`
    (eval $e:expr) => {
        {
            let val: usize = $e; // Force types to be integers
            println!("{} = {}", stringify!{$e}, val);
        }
    };

    // Decompose multiple `eval`s recursively
    (eval $e:expr, $(eval $es:expr),+) => {{
        calculate! { eval $e }
        calculate! { $(eval $es),+ }
    }};
}

fn main() {
    calculate! { // Look ma! Variadic `calculate!`!
        eval 1 + 2,
        eval 3 + 4,
        eval (2 * 3) + 1
    }
}

    خروجی:

    1 + 2 = 3
3 + 4 = 7
(2 * 3) + 1 = 7
