Rust با مثالRust By Example

    while

    با کلیدواژهٔ while می‌توان تا زمانی که یک شرط درست است، حلقه را اجرا کرد.

    بیایید برنامهٔ مشهور FizzBuzz را با یک حلقهٔ while بنویسیم.

    fn main() {
    // A counter variable
    let mut n = 1;

    // Loop while `n` is less than 101
    while n < 101 {
        if n % 15 == 0 {
            println!("fizzbuzz");
        } else if n % 3 == 0 {
            println!("fizz");
        } else if n % 5 == 0 {
            println!("buzz");
        } else {
            println!("{}", n);
        }

        // Increment counter
        n += 1;
    }
}
