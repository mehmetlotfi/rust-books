    if/else

    انشعاب با if و else شبیه دیگر زبان‌هاست. برخلاف بسیاری از آن‌ها، لازم نیست شرط بولی داخل پرانتز قرار بگیرد و پس از هر شرط یک بلوک می‌آید. شرط‌های if و else عبارت هستند و همهٔ شاخه‌های آن‌ها باید نوع یکسانی را بازگردانند.

    fn main() {
    let n = 5;

    if n < 0 {
        print!("{} is negative", n);
    } else if n > 0 {
        print!("{} is positive", n);
    } else {
        print!("{} is zero", n);
    }

    let big_n =
        if n < 10 && n > -10 {
            println!(", and is a small number, increase ten-fold");

            // This expression returns an `i32`.
            10 * n
        } else {
            println!(", and is a big number, halve the number");

            // This expression must return an `i32` as well.
            n / 2
            // TODO ^ Try suppressing this expression with a semicolon.
        };
    //   ^ Don't forget to put a semicolon here! All `let` bindings need it.

    println!("{} -> {}", n, big_n);
}
