    حلقه‌های تودرتو و برچسب‌ها

    در حلقه‌های تودرتو می‌توان با break یا continue روی حلقهٔ بیرونی اثر گذاشت. برای این کار باید حلقه را با یک 'label نشانه‌گذاری و همان برچسب را به دستور break یا continue ارسال کرد.

    #![allow(unreachable_code, unused_labels)]

fn main() {
    'outer: loop {
        println!("Entered the outer loop");

        'inner: loop {
            println!("Entered the inner loop");

            // This would break only the inner loop
            //break;

            // This breaks the outer loop
            break 'outer;
        }

        println!("This point will never be reached");
    }

    println!("Exited the outer loop");
}
