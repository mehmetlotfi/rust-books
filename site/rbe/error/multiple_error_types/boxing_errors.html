    قراردادن خطاها در Box

    یک راه نوشتن کد ساده با حفظ خطاهای اصلی، قراردادن آن‌ها در Box است. عیب آن این است که نوع زیرین خطا فقط هنگام اجرا معلوم است و به‌صورت ایستا تعیین نمی‌شود.

    کتابخانهٔ استاندارد با اینکه Box تبدیل از هر نوعی را که صفت Error را پیاده‌سازی می‌کند به شیء صفت Box<Error> پیاده‌سازی کرده است، از طریق From به جعبه‌کردن (boxing) خطاهای ما کمک می‌کند.

    use std::error;
use std::fmt;

// Change the alias to use `Box<dyn error::Error>`.
type Result<T> = std::result::Result<T, Box<dyn error::Error>>;

#[derive(Debug, Clone)]
struct EmptyVec;

impl fmt::Display for EmptyVec {
    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
        write!(f, "invalid first item to double")
    }
}

impl error::Error for EmptyVec {}

fn double_first(vec: Vec<&str>) -> Result<i32> {
    vec.first()
        .ok_or_else(|| EmptyVec.into()) // Converts to Box using Into trait.
        .and_then(|s| {
            s.parse::<i32>()
                .map_err(From::from) // Converts to Box using From::from fn pointer.
                .map(|i| 2 * i)
        })
}

fn print(result: Result<i32>) {
    match result {
        Ok(n) => println!("The first doubled is {}", n),
        Err(e) => println!("Error: {}", e),
    }
}

fn main() {
    let numbers = vec!["42", "93", "18"];
    let empty = vec![];
    let strings = vec!["tofu", "93", "18"];

    print(double_first(numbers));
    print(double_first(empty));
    print(double_first(strings));
}
