    راست دو نوع ثابت متفاوت دارد که می‌توان آن‌ها را در هر محدوده‌ای، از جمله محدودهٔ سراسری، اعلان کرد. نوع هر دو باید به‌صراحت مشخص شود:

    • const: مقداری تغییرناپذیر و حالت رایج استفاده از ثابت‌ها.
    • static: متغیری که می‌تواند تغییرپذیر باشد و طول عمر 'static دارد. طول عمر ایستا استنباط می‌شود و نیازی به نوشتن آن نیست. دسترسی به یک متغیر ایستای تغییرپذیر یا تغییر آن عملیاتی unsafe است.
    // Globals are declared outside all other scopes.
static LANGUAGE: &str = "Rust";
const THRESHOLD: i32 = 10;

fn is_big(n: i32) -> bool {
    // Access constant in some function
    n > THRESHOLD
}

fn main() {
    let n = 16;

    // Access constant in the main thread
    println!("This is {}", LANGUAGE);
    println!("The threshold is {}", THRESHOLD);
    println!("{} is {}", n, if is_big(n) { "big" } else { "small" });

    // Error! Cannot modify a `const`.
    THRESHOLD = 5;
    // FIXME ^ Comment out this line
}
