    الگوی نوع جدید

    الگوی newtype در زمان کامپایل تضمین می‌کند که مقدار با نوع درست به برنامه داده شده است.

    برای نمونه، تابعی که فاصله را با مایل اندازه می‌گیرد باید مقداری از نوع Miles دریافت کند.

    struct Miles(f64);

struct Kilometers(f64);

impl Miles {
    pub fn to_kilometers(&self) -> Kilometers {
        Kilometers(self.0 * 1.609344)
    }
}

impl Kilometers {
    pub fn to_miles(&self) -> Miles {
        Miles(self.0 / 1.609344)
    }
}

fn is_a_marathon(distance: &Miles) -> bool {
    distance.0 >= 26.2
}

fn main() {
    let distance = Miles(30.0);
    let distance_km = distance.to_kilometers();
    println!("Is a marathon? {}", is_a_marathon(&distance));
    println!("Is a marathon? {}", is_a_marathon(&distance_km.to_miles()));
    // println!("Is a marathon? {}", is_a_marathon(&distance_km));
}

    آخرین دستور چاپ را از حالت توضیح خارج کنید تا ببینید نوع ورودی حتماً باید Miles باشد.

    برای دریافت مقدار newtype به‌صورت نوع پایه، می‌توانید از نگارش تاپل یا ساختارشکنی استفاده کنید:

    struct Miles(f64);

fn main() {
    let distance = Miles(42.0);
    let distance_as_primitive_1: f64 = distance.0; // Tuple
    let Miles(distance_as_primitive_2) = distance; // Destructuring
}
