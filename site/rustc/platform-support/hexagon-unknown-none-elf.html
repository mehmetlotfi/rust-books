    hexagon-unknown-none-elf

    سطح (Tier): ۳

    Rust برای DSPهای Hexagon در حالت baremetal.

    هدفتوضیحات
    hexagon-unknown-none-elfHexagon ۳۲ بیتی (freestanding، hardfloat)

    نگهدارندگان هدف

    @androm3da

    الزامات

    این هدف cross-compile می‌شود. از std پشتیبانی نمی‌شود. تخصیص‌دهنده (allocator) پیش‌فرضی وجود ندارد، اما با فراهم کردن یک allocator می‌توان از alloc استفاده کرد.

    به‌طور پیش‌فرض، کدی که با این هدف تولید می‌شود باید روی سخت‌افزار Hexagon DSP اجرا شود.

    • -Ctarget-cpu=hexagonv73 پشتیبانی از دستورهای تعریف‌شده تا Hexagon V73 را اضافه می‌کند.

    توابعی که با extern "C" علامت‌گذاری شده‌اند از قرارداد فراخوانی معماری Hexagon استفاده می‌کنند.

    این هدف فایل‌های دودویی ELF از نوع PIC تولید می‌کند.

    لینک کردن

    این هدف به‌طور پیش‌فرض rust-lld را انتخاب می‌کند. گزینه دیگر eld است که همراه toolchain متن‌باز hexagon و Hexagon SDK نیز ارائه می‌شود.

    ساخت هدف

    می‌توانید Rust را با پشتیبانی از این هدف بسازید؛ برای این کار آن را به فهرست target در bootstrap.toml اضافه کنید:

    [build]
build-stage = 1
host = ["<target for your host>"]
target = ["<target for your host>", "hexagon-unknown-none-elf"]

[target.hexagon-unknown-none-elf]

cc = "hexagon-unknown-none-elf-clang"
cxx = "hexagon-unknown-none-elf-clang++"
linker = "hexagon-unknown-none-elf-clang"
ranlib = "hexagon-unknown-none-elf-ranlib"
ar = "hexagon-unknown-none-elf-ar"
llvm-libunwind = 'in-tree'

    <target for your host> را با x86_64-unknown-linux-gnu یا هر مقدار مناسب دیگر برای ماشین میزبان خود جایگزین کنید.

    ساخت برنامه‌های Rust

    Rust هنوز artifactهای از پیش کامپایل‌شده برای این هدف ارائه نمی‌دهد. برای کامپایل برای این هدف، یا باید Rust را با فعال‌بودن این هدف بسازید (بخش «ساخت هدف» را در بالا ببینید)، یا با استفاده از build-std یا روشی مشابه، نسخه خودتان از core را بسازید.

    آزمون

    از آنجا که hexagon-unknown-none-elf از محیط‌های متنوعی پشتیبانی می‌کند و از std پشتیبانی نمی‌کند، این هدف از اجرای مجموعه آزمون‌های Rust پشتیبانی نمی‌کند.

    toolchainهای cross-compilation و کد C

    این هدف با استفاده از qemu-system-hexagon آزمایش شده است.

    یکی از کاربردهای رایج hexagon-unknown-none-elf ساخت کتابخانه‌هایی است که با کد C لینک می‌شوند و می‌توان آن‌ها را در شبیه‌سازی یا روی دستگاهی دارای Hexagon DSP به کار برد.

    Hexagon SDK کتابخانه‌هایی دارد که هنگام اجرا روی دستگاه، لینک کردن با آن‌ها مفید است.

    سیستم‌عامل مستقل (Standalone OS)

    اسکریپت زیر یک فایل اجرایی را در برابر «hexagon standalone OS» می‌سازد که برای شبیه‌سازی یا آزمون bare-metal روی دستگاه مناسب است.

    ابتدا cargo new --bin demo1_hexagon را اجرا کنید، سپس کد منبع زیر را به‌عنوان src/main.rs اضافه کنید. این برنامه خروجی کنسول را از طریق semihosting نشان می‌دهد.

    #![no_std]
#![no_main]

extern "C" {
    fn putchar(ch: i32);
    fn _exit(code: i32) -> !;
}

#[no_mangle]
extern "C" fn main() -> i32 {
    let message = "Hello, this is Rust!";
    for b in message.bytes() {
        unsafe {
            putchar(b as i32);
        }
    }
    0
}

#[panic_handler]
fn panic(_panic: &core::panic::PanicInfo) -> ! {
    unsafe {
        _exit(1);
    }
}

    سپس اسکریپت زیر را با نام build.sh ذخیره کنید و آن را متناسب با محیط خود ویرایش کنید.

    • hex_toolchain در زیر به toolchain مربوط به hexagon که منحصراً از مخازن متن‌باز عمومی ساخته شده است اشاره دارد.
    • cc در زیر به clang اشاره دارد. می‌توانید از clang توزیع خود استفاده کنید، به شرطی که دست‌کم clang-17 باشد. یا می‌توانید از hexagon-unknown-none-elf-clang در یکی از انتشارهای toolchain متن‌باز hexagon استفاده کنید.
    # Hexagon SDK, required for target libraries:
hex_sdk_root=/local/mnt/workspace/Qualcomm/Hexagon_SDK/5.3.0.0
hex_sdk_toolchain=${hex_sdk_root}/tools/HEXAGON_Tools/8.6.06

sdk_libs=${hex_sdk_toolchain}/Tools/target/hexagon/lib
q6_arch=v65
g0_lib_path=${sdk_libs}/${q6_arch}/G0
pic_lib_path=${sdk_libs}/${q6_arch}/G0/pic

build_cfg=release
cargo build --target=hexagon-unknown-none-elf -Zbuild-std --release

# Builds an executable against "hexagon standalone OS" suitable for emulation:
${cc} --target=hexagon-unknown-none-elf -o testit \
    -fuse-ld=lld \
    -m${q6_arch} \
    -nodefaultlibs \
    -nostartfiles \
    ${g0_lib_path}/crt0_standalone.o \
    ${g0_lib_path}/crt0.o \
    ${g0_lib_path}/init.o \
    -L${sdk_libs}/${q6_arch}/ \
    -L${sdk_libs}/ \
    wrap.c \
    target/hexagon-unknown-none-elf/${build_cfg}/libdemo1_hexagon.rlib \
    target/hexagon-unknown-none-elf/${build_cfg}/deps/libcore-*.rlib \
    target/hexagon-unknown-none-elf/${build_cfg}/deps/libcompiler_builtins-*.rlib \
    -Wl,--start-group \
    -Wl,--defsym,_SDA_BASE_=0,--defsym,__sbss_start=0,--defsym,__sbss_end=0 \
    ${g0_lib_path}/libstandalone.a \
    ${g0_lib_path}/libc.a \
    -lgcc \
    -lc_eh \
    -Wl,--end-group \
    ${g0_lib_path}/fini.o \

${hex_toolchain}/x86_64-linux-gnu/bin/qemu-system-hexagon -monitor none -display none -kernel ./testit

    QuRT OS

    ابتدا cargo new --lib demo2_hexagon را اجرا کنید، سپس کد منبع زیر را به‌عنوان src/lib.rs اضافه کنید. این برنامه اسمبلی درون‌خطی و خروجی کنسول از طریق semihosting را نشان می‌دهد.

    #![no_std]
#![no_main]
#![feature(lang_items)]
#![feature(asm_experimental_arch)]

use core::arch::asm;

extern "C" {
    fn putchar(ch: i32);
    fn _exit(code: i32) -> !;
}

fn hexagon_specific() {
    let mut buffer = [0_u8; 128];

    unsafe {
        let mut x = &buffer;
        asm!(
                "{{\n\t",
                "  v0=vmem({addr}+#0)\n\t",
                "  {tmp} = and({tmp}, #1)\n\t",
                "}}\n\t",
                addr = in(reg) x,
                tmp = out(reg) _,
            );
    }
}

#[no_mangle]
extern "C" fn hello() -> i32 {
    let message = "Hello, this is Rust!\n";
    for b in message.bytes() {
        unsafe {
            putchar(b as i32);
        }
    }
    hexagon_specific();
    0
}

#[panic_handler]
fn panic(_panic: &core::panic::PanicInfo) -> ! {
    unsafe {
        _exit(1);
    }
}

#[lang = "eh_personality"]
fn rust_eh_personality() {}

    سپس یک برنامه C را به‌عنوان نقطه ورود بسازید و محتوای زیر را با نام wrap.c ذخیره کنید:

    int hello();

int main() {
    hello();
}

    بعد اسکریپت زیر را با نام build.sh ذخیره کنید و آن را متناسب با محیط خود ویرایش کنید. اسکریپت زیر یک shared object را در برابر سیستم‌عامل بی‌درنگ QuRT می‌سازد که برای شبیه‌سازی یا آزمون روی دستگاه، هنگامی که از طریق fastrpc-shell بارگذاری شود، مناسب است.

    # Hexagon SDK, required for target libraries:
hex_sdk_root=/local/mnt/workspace/Qualcomm/Hexagon_SDK/5.3.0.0
hex_sdk_toolchain=${hex_sdk_root}/tools/HEXAGON_Tools/8.6.06

sdk_libs=${hex_sdk_toolchain}/Tools/target/hexagon/lib
q6_arch=v65
g0_lib_path=${sdk_libs}/${q6_arch}/G0
pic_lib_path=${sdk_libs}/${q6_arch}/G0/pic
runelf=${hex_sdk_root}/rtos/qurt/computev65/sdksim_bin/runelf.pbn
rmohs=${hex_sdk_root}/libs/run_main_on_hexagon/ship/hexagon_toolv86_${q6_arch}/run_main_on_hexagon_sim

# Builds a library suitable for loading into "run_main_on_hexagon_sim" for
# emulation or frpc shell on real target:
${cc} --target=hexagon-unknown-none-elf -o testit.so \
    -fuse-ld=lld \
    -fPIC -shared \
    -nostdlib \
    -Wl,-Bsymbolic \
      -Wl,--wrap=malloc \
      -Wl,--wrap=calloc \
      -Wl,--wrap=free \
      -Wl,--wrap=realloc \
      -Wl,--wrap=memalign \
    -m${q6_arch} \
    wrap.c \
    target/hexagon-unknown-none-elf/${build_cfg}/libdemo2_hexagon.rlib \
    target/hexagon-unknown-none-elf/${build_cfg}/deps/libcore-*.rlib \
    target/hexagon-unknown-none-elf/${build_cfg}/deps/libcompiler_builtins-*.rlib \
    -Wl,-soname=testit \
    ${pic_lib_path}/libc.so

# -Bsymbolic above for memory alloc funcs is necessary to access the heap on
# target, but otherwise not required.

# multi-stage loader: runelf => run_main_on_hexagon_sim => testit.so{`main`}
${hex_toolchain}/x86_64-linux-gnu/bin/qemu-system-hexagon \
    -monitor none \
    -display none \
    -kernel ${runelf} \
    -append "${rmohs} -- ./testit.so"
