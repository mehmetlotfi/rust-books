    *-unknown-linux-ohos

    Tier: ۲ (همراه با ابزارهای میزبان)

    • aarch64-unknown-linux-ohos
    • armv7-unknown-linux-ohos
    • x86_64-unknown-linux-ohos

    Tier: 3

    • loongarch64-unknown-linux-ohos

    هدف‌هایی برای سیستم‌عامل OpenHarmony.

    نگهدارندگان هدف

    @Amanieu @cceerczw

    الزامات

    همهٔ هدف‌های ohos در Tier 2 با ابزارهای میزبان (host tools)، همهٔ ابزارهای گسترش‌یافتهٔ Rust را پشتیبانی می‌کنند. (به‌جز miri؛ پشتیبانی از miri به‌زودی اضافه خواهد شد)

    زنجیرهٔ ابزار میزبان

    این هدف‌ها به یک OpenHarmony SDK نسبتاً به‌روز روی میزبان نیاز دارند.

    این هدف‌ها از cargo پشتیبانی می‌کنند که به ohos-openssl نیاز دارد.

    miri هنوز پشتیبانی نمی‌شود، زیرا وابستگی‌های آن (libffi و tikv-jemalloc-sys) از کامپایل برای هدف‌های OHOS پشتیبانی نمی‌کنند.

    راه‌اندازی

    OpenHarmony SDK در حال حاضر کامپایل Rust را مستقیماً پشتیبانی نمی‌کند، بنابراین مقداری راه‌اندازی لازم است.

    ابتدا باید OpenHarmony SDK را از این صفحه دریافت کنید. نسخه‌ای از OpenHarmony را که برای آن توسعه می‌دهید انتخاب کنید و «Public SDK package for the standard system» را دانلود کنید.

    اسکریپت‌های شل زیر را که Clang موجود در OpenHarmony SDK را بسته‌بندی (wrap) می‌کنند ایجاد کنید:

    aarch64-unknown-linux-ohos-clang.sh

    #!/bin/sh
exec /path/to/ohos-sdk/linux/native/llvm/bin/clang \
  -target aarch64-linux-ohos \
  --sysroot=/path/to/ohos-sdk/linux/native/sysroot \
  -D__MUSL__ \
  "$@"

    aarch64-unknown-linux-ohos-clang++.sh

    #!/bin/sh
exec /path/to/ohos-sdk/linux/native/llvm/bin/clang++ \
  -target aarch64-linux-ohos \
  --sysroot=/path/to/ohos-sdk/linux/native/sysroot \
  -D__MUSL__ \
  "$@"

    armv7-unknown-linux-ohos-clang.sh

    #!/bin/sh
exec /path/to/ohos-sdk/linux/native/llvm/bin/clang \
  -target arm-linux-ohos \
  --sysroot=/path/to/ohos-sdk/linux/native/sysroot \
  -D__MUSL__ \
  -march=armv7-a \
  -mfloat-abi=softfp \
  -mtune=generic-armv7-a \
  -mthumb \
  "$@"

    armv7-unknown-linux-ohos-clang++.sh

    #!/bin/sh
exec /path/to/ohos-sdk/linux/native/llvm/bin/clang++ \

    نسخه‌های آیندهٔ OpenHarmony SDK نیاز به این فرایند را برطرف خواهند کرد.

    ساخت برنامه‌های Rust

    Rustup برای این هدف آرتیفکت‌های از پیش کامپایل‌شده ارائه می‌کند که می‌توانید با دستورهای زیر نصب کنید:

    rustup target add aarch64-unknown-linux-ohos
rustup target add armv7-unknown-linux-ohos
rustup target add x86_64-unknown-linux-ohos

    باید پیوندده‌ای (linker) را که قرار است استفاده شود در ~/.cargo/config.toml پیکربندی کنید:

    [target.aarch64-unknown-linux-ohos]
ar = "/path/to/ohos-sdk/linux/native/llvm/bin/llvm-ar"
linker = "/path/to/aarch64-unknown-linux-ohos-clang.sh"

[target.armv7-unknown-linux-ohos]
ar = "/path/to/ohos-sdk/linux/native/llvm/bin/llvm-ar"
linker = "/path/to/armv7-unknown-linux-ohos-clang.sh"

[target.x86_64-unknown-linux-ohos]
ar = "/path/to/ohos-sdk/linux/native/llvm/bin/llvm-ar"
linker = "/path/to/x86_64-unknown-linux-ohos-clang.sh"

    ساخت هدف از روی کد منبع

    به‌جای استفاده از rustup، می‌توانید یک زنجیرهٔ ابزار Rust را از روی کد منبع بسازید. یک فایل bootstrap.toml با محتوای زیر ایجاد کنید:

    profile = "compiler"
change-id = 115898

[build]
sanitizers = true
profiler = true

[target.aarch64-unknown-linux-ohos]
cc = "/path/to/aarch64-unknown-linux-ohos-clang.sh"
cxx = "/path/to/aarch64-unknown-linux-ohos-clang++.sh"
ar = "/path/to/ohos-sdk/linux/native/llvm/bin/llvm-ar"
ranlib = "/path/to/ohos-sdk/linux/native/llvm/bin/llvm-ranlib"
linker  = "/path/to/aarch64-unknown-linux-ohos-clang.sh"

[target.armv7-unknown-linux-ohos]
cc = "/path/to/armv7-unknown-linux-ohos-clang.sh"
cxx = "/path/to/armv7-unknown-linux-ohos-clang++.sh"
ar = "/path/to/ohos-sdk/linux/native/llvm/bin/llvm-ar"
ranlib = "/path/to/ohos-sdk/linux/native/llvm/bin/llvm-ranlib"
linker  = "/path/to/armv7-unknown-linux-ohos-clang.sh"

[target.x86_64-unknown-linux-ohos]
